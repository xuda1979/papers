\begin{frame}{模型架构解析：MoE（混合专家）vs Dense（密集）}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \begin{block}{什么是MoE？}
                \small
                \textbf{Mixture of Experts}（混合专家）
                \begin{itemize}
                    \item 模型包含多个"专家"子网络
                    \item 每次推理只激活\textbf{部分专家}
                    \item 路由器决定使用哪些专家
                    \item 总参数量大，但计算量小
                \end{itemize}
            \end{block}
            
            \vspace{0.2cm}
            \begin{exampleblock}{MoE优势}
                \scriptsize
                \begin{itemize}
                    \item \textcolor{green!70!black}{\faCheck} 训练效率高（同算力训更大模型）
                    \item \textcolor{green!70!black}{\faCheck} 推理计算少（只用部分参数）
                    \item \textcolor{green!70!black}{\faCheck} 成本更低
                \end{itemize}
            \end{exampleblock}
            
            \begin{alertblock}{MoE劣势}
                \scriptsize
                \begin{itemize}
                    \item \textcolor{red}{\faTimes} 显存占用大（需加载所有专家）
                    \item \textcolor{red}{\faTimes} 微调容易过拟合
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{主流模型架构对比}
                \centering
                \scriptsize
                \begin{tabular}{l|c|l}
                    \toprule
                    \textbf{模型} & \textbf{架构} & \textbf{MoE规格} \\
                    \midrule
                    \multicolumn{3}{c}{\textbf{--- MoE架构 ---}} \\
                    \rowcolor{blue!10}
                    DeepSeek V3.2 & MoE & 671B/37B激活 \\
                    Gemini 3系列 & MoE & 未公开 \\
                    Mixtral 8x7B & MoE & 47B/12B激活 \\
                    Grok 4 & MoE & 推测 \\
                    Qwen3-MoE & MoE & 多版本 \\
                    \midrule
                    \multicolumn{3}{c}{\textbf{--- Dense架构 ---}} \\
                    GPT-5.2 & Dense & -- \\
                    Claude 4.5 & Dense & -- \\
                    Llama 4 & Dense & -- \\
                    \bottomrule
                \end{tabular}
            \end{block}
            
            \vspace{0.2cm}
            \begin{block}{DeepSeek MoE详解}
                \scriptsize
                \begin{itemize}
                    \item \textbf{总参数}：671B（6710亿）
                    \item \textbf{激活参数}：仅37B（5.5\%）
                    \item \textbf{专家数}：256个路由专家 + 1个共享专家
                    \item \textbf{每次激活}：8个专家
                    \item 这解释了DeepSeek的\textbf{超低价格}！
                \end{itemize}

\begin{frame}{推理时计算扩展：成本倒挂现象}
    \begin{columns}

                \textbf{传统：训练时扩展}
                \begin{itemize}
                    \item 更大模型 + 更多数据
                    \item GPT-3 → GPT-4 → GPT-5
                    \item 训练成本极高，推理成本固定
                \end{itemize}

                \textbf{新范式：推理时扩展}
                \begin{itemize}
                    \item 让模型"思考更久"
                    \item 复杂问题多花时间推理
                    \item 训练成本降低，推理按需付费
                \end{itemize}

\begin{frame}{问题根源（一）：幻觉的本质——"鹦鹉学舌"而非"理解"}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{大模型的工作原理：}
            \begin{enumerate}
                \item 输入文本 → 转为Token序列
                \item 根据上文预测\textbf{下一个最可能的Token}
                \item 采样生成 → 重复直到结束
            \end{enumerate}
            
            \vspace{0.3cm}
            \textbf{\textcolor{red}{关键洞察：}}
            \begin{itemize}
                \item 模型在做\textbf{统计模式匹配}
                \item \textbf{不是}在"理解"或"推理"
                \item \textbf{不是}在查询知识库
                \item 只是预测"什么词最可能出现在这里"
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{为什么会幻觉？}
            \begin{itemize}
                \item 训练数据中某些模式频繁出现
                \item 模型学会了"看起来正确"的表达方式
                \item 但无法验证内容的真实性
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{tikzpicture}[scale=0.8]
                \node[draw, rounded corners, fill=blue!20, minimum width=3cm, minimum height=0.8cm] (input) at (0,3) {输入: "爱因斯坦在"};
                \node[draw, rounded corners, fill=orange!20, minimum width=3cm, minimum height=0.8cm] (model) at (0,1.5) {模型: 计算概率分布};
                \node[draw, rounded corners, fill=green!20, minimum width=3cm, minimum height=0.8cm] (output) at (0,0) {输出: "1905年" (P=0.3)};
                
                \draw[->, thick] (input) -- (model);
                \draw[->, thick] (model) -- (output);
                
                \node[right, font=\tiny] at (1.8,1.5) {
                    \begin{tabular}{ll}
                        1905年 & 30\% \\
                        1879年 & 25\% \\
                        普林斯顿 & 15\% \\
                        ... & ...
                    \end{tabular}
                };
            \end{tikzpicture}
            
            \vspace{0.3cm}
            \begin{alertblock}{本质问题}
                \small
                模型选择"统计上最可能"的续写，而非"事实上正确"的答案。当训练数据不足或有噪声时，就会产生幻觉。
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{问题根源（二）：上下文遗忘——有限的"工作记忆"}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Transformer的注意力机制：}
            \begin{itemize}
                \item 每个Token都要关注所有其他Token
                \item 计算复杂度：$O(n^2)$（n为序列长度）
                \item \textbf{上下文窗口}是硬性限制
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{上下文窗口限制（2026年）：}
            \begin{itemize}
                \item GPT-5.2: \textbf{128K} tokens
                \item Claude 4.5: \textbf{200K} tokens
                \item Gemini 3: \textbf{2M} tokens
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{\textcolor{red}{关键问题：}}
            \begin{itemize}
                \item 窗口内的信息\textbf{不是均匀关注}
                \item 存在"Lost in the Middle"现象
                \item 开头和结尾被关注更多
                \item 中间部分容易被"遗忘"
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{tikzpicture}[scale=0.7]
                \begin{axis}[
                    xlabel={文本位置},
                    ylabel={注意力权重},

                };
                \end{axis}
            \end{tikzpicture}
            
            \begin{alertblock}{"Lost in the Middle"现象}
                \small
                研究表明，当重要信息放在长文本中间时，模型检索准确率\textbf{下降40-60\%}。
                
                \vspace{0.1cm}
                {\scriptsize Liu et al., 2024, "Lost in the Middle"}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{问题根源（三）：多轮对话质量衰减}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{多轮对话的技术实现：}
            \begin{enumerate}
                \item 每轮对话都把\textbf{完整历史}发给模型
                \item 历史越长，Token消耗越多
                \item 模型每次都要重新"阅读"全部内容
            \end{enumerate}
            
            \vspace{0.3cm}
            \textbf{质量衰减的原因：}
            \begin{itemize}
                \item \textbf{上下文污染}：历史中的错误会被"学习"
                \item \textbf{注意力稀释}：关注点分散到更多内容
                \item \textbf{模式锁定}：倾向重复已有的表达模式
                \item \textbf{指令遗忘}：原始指令权重降低
            \end{itemize}
            
            \vspace{0.2cm}
            \begin{block}{类比理解}
                \small
                像一个人同时听10个人说话，每多一个人，对每个人的注意力就少一点。
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{tikzpicture}[scale=0.7]
                \begin{axis}[
                    xlabel={对话轮次},
                    ylabel={回答质量（相对值）},

                \end{axis}
            \end{tikzpicture}
            
            \vspace{0.2cm}
            \begin{exampleblock}{技术解释}
                \scriptsize
                \textbf{为什么会重复？}\\
                当历史中某个模式出现多次，模型预测"下一个词"时，这个模式的概率被不断强化，形成"自我循环"。
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{深层原因：为什么这些问题难以根治？}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{1. 训练目标的局限}
            \begin{itemize}
                \item 目标：预测下一个Token
                \item \textbf{不是}：生成正确的事实
                \item \textbf{不是}：保持逻辑一致性
                \item \textbf{不是}：长期记忆管理
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{2. 知识存储方式}
            \begin{itemize}
                \item 知识编码在\textbf{模型参数}中
                \item 不是存储在可查询的数据库
                \item 无法精确检索和验证
                \item 更新知识需要重新训练
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{3. 缺乏真正的"理解"}
            \begin{itemize}
                \item 没有世界模型
                \item 无法进行因果推理
                \item 不理解物理规律
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Stochastic Parrot (随机鹦鹉)}
                \small
                Bender et al. (2021) 的著名论文指出：
                
                \vspace{0.2cm}
                "大语言模型是\textbf{随机鹦鹉}——它们在操纵语言形式，而不理解语言的\textbf{意义}。"
                
                \vspace{0.2cm}
                模型学会了"说人话"的统计模式，但不理解话语背后的世界。
            \end{alertblock}
            
            \vspace{0.3cm}
            \begin{block}{类比}
                \small
                \textbf{传统数据库}：图书馆，可以精确查找
                
                \vspace{0.1cm}
                \textbf{大语言模型}：读过很多书的人，凭印象回答，可能记错细节
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{"Lost in the Middle"现象详解}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{什么是"Lost in the Middle"？}
            \begin{itemize}
                \item 大模型处理长文本时的\textbf{注意力分布不均}现象
                \item 开头和结尾的信息被充分关注
                \item \textcolor{red}{\textbf{中间部分的信息容易被"遗忘"}}
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{实验数据（Liu et al., 2024）：}
            \begin{itemize}
                \item 测试任务：长文档问答
                \item 关键信息在\textbf{开头}：准确率 \textbf{90\%+}
                \item 关键信息在\textbf{中间}：准确率 \textbf{40-50\%}
                \item 关键信息在\textbf{结尾}：准确率 \textbf{85\%+}
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{\textcolor{red}{影响范围：}}
            \begin{itemize}
                \item 所有Transformer架构模型都有此问题
                \item 上下文越长，中间遗忘越严重
                \item 即使是200K+上下文的模型也不例外
            \end{itemize}
        \end{column}

\begin{frame}{什么是校准误差？——模型自知之明的衡量}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \textbf{校准误差（Calibration Error）定义}
            \begin{itemize}
                \item 衡量模型\textbf{置信度}与\textbf{实际准确率}的匹配程度
                \item 公式：$\text{CE} = |\text{模型置信度} - \text{实际正确率}| \times 100$
                \item \textbf{数值越低}，模型越``知道自己知不知道''
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{举例说明：}
            \begin{itemize}
                \item 模型说``我90\%确定答案是A''
                \item 如果100次这样的情况中真的有90次正确\\
                      → \textcolor{green!60!black}{校准误差 = 0（完美校准）}
                \item 如果实际只有60次正确\\
                      → \textcolor{red}{校准误差 = 30（过度自信）}
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{tikzpicture}[scale=0.75]

                \item \textbf{记忆 (Memory)}
                    \begin{itemize}

\begin{frame}{使用技巧：如何获得更详细的回答}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{为什么AI回答有时很简短？}
            \begin{itemize}
                \item 大模型通过预测"下一个词"生成回答
                \item 训练数据决定了\textbf{平均输出长度}
                \item 遇到"结束标记(EOF)"就停止输出
                \item 模型倾向于给出"够用"的回答
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{\textcolor{green!60!black}{获取详细回答的技巧：}}
            \begin{itemize}
                \item \textbf{"详细一点"} - 最简单有效
                \item \textbf{"请展开说明"} - 要求更多细节
                \item \textbf{"请给出完整示例"} - 要求完整代码/步骤
                \item \textbf{"请分步骤解释"} - 结构化输出
                \item \textbf{"至少写1000字"} - 明确长度要求
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{大模型输出原理}
                大模型是\textbf{逐词预测}的：
                \begin{itemize}
                    \item 根据上文预测下一个词的概率
                    \item 通过采样选择输出哪个词
                    \item 重复直到生成结束标记
                \end{itemize}
                因此，\textbf{提示词会影响输出长度}！
            \end{alertblock}
            
            \vspace{0.2cm}
            \begin{exampleblock}{示例对比}
                \small
                \textbf{简单问法：}"解释什么是机器学习"\\
                → 可能只得到2-3句话
                
                \vspace{0.2cm}
                \textbf{详细问法：}"详细解释什么是机器学习，包括定义、主要类型、应用场景和学习建议"\\
                → 会得到完整详细的回答
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\section{AI论文检测与学术诚信}
%========================================

\begin{frame}{AI生成内容检测工具概览}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{主流AI检测工具：}
            \begin{itemize}
                \item \textbf{Turnitin AI Detection}
                    \begin{itemize}
                        \item 学术界最权威
                        \item 已集成至多数高校系统
                        \item 检测准确率约98\%
                    \end{itemize}
                \item \textbf{GPTZero}
                    \begin{itemize}
                        \item 专为教育场景设计
                        \item 提供逐句分析
                        \item 免费版可用
                    \end{itemize}
                \item \textbf{Originality.AI}
                    \begin{itemize}
                        \item 商业级检测
                        \item 支持批量检测
                    \end{itemize}
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{检测原理：}
            \begin{block}{统计特征分析}
                \small
                \begin{itemize}
                    \item \textbf{困惑度(Perplexity)}：AI文本通常更"可预测"
                    \item \textbf{突发性(Burstiness)}：人类写作句长变化大
                    \item \textbf{词汇多样性}：AI倾向使用常见词
                \end{itemize}
            \end{block}
            
            \vspace{0.3cm}
            \begin{alertblock}{重要提醒}
                \small
                \begin{itemize}
                    \item 检测工具存在\textbf{假阳性}风险
                    \item 非母语者文本更易被误判
                    \item 技术文档/代码注释误判率高
                \end{itemize}
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{各大期刊/会议的AI使用政策}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{明确允许（需声明）：}
            \begin{itemize}
                \item \textcolor{green!60!black}{\faCheck\ Nature系列}
                    \begin{itemize}
                        \item 必须在Methods或致谢中声明
                        \item AI不能作为作者
                    \end{itemize}
                \item \textcolor{green!60!black}{\faCheck\ Science系列}
                    \begin{itemize}
                        \item 允许用于润色和编辑
                        \item 需明确说明使用方式
                    \end{itemize}
                \item \textcolor{green!60!black}{\faCheck\ ACL/EMNLP/NeurIPS}
                    \begin{itemize}
                        \item 允许辅助写作
                        \item 禁止AI直接生成核心内容
                    \end{itemize}
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{标准声明模板：}
            \begin{exampleblock}{推荐写法}
                \scriptsize
                \texttt{"During the preparation of this work, the authors used [Tool Name] for [specific purpose, e.g., language editing, code debugging]. After using this tool, the authors reviewed and edited the content as needed and take full responsibility for the content of the publication."}
            \end{exampleblock}
            
            \vspace{0.2cm}
            \textbf{禁止的行为：}
            \begin{itemize}
                \item[$\times$] AI作为论文作者
                \item[$\times$] 未声明的大规模AI生成
                \item[$\times$] AI生成数据/实验结果
                \item[$\times$] AI生成文献引用
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{如何合规使用AI避免学术不端}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{\textcolor{green!60!black}{\faCheck\ 合规使用场景：}}
            \begin{itemize}
                \item 语法和拼写检查
                \item 语言润色（非母语者）
                \item 代码调试和优化
                \item 文献检索辅助
                \item 格式转换（如Markdown→LaTeX）
                \item 数据可视化代码生成
                \item 解释复杂概念（学习用途）
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{\textcolor{red}{\faTimes\ 禁止使用场景：}}
            \begin{itemize}
                \item 生成核心创新内容
                \item 编造实验数据
                \item 生成虚假引用
                \item 代写整篇论文
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{降低AI检测风险的技巧：}
            \begin{block}{合法优化方法}
                \small
                \begin{enumerate}
                    \item \textbf{分段处理}：不要一次性生成长文本
                    \item \textbf{人工改写}：AI输出后大幅修改
                    \item \textbf{混合写作}：AI润色人写的草稿
                    \item \textbf{增加个性}：加入个人观点和例子
                    \item \textbf{使用引用}：增加具体文献支撑
                \end{enumerate}
            \end{block}
            
            \vspace{0.2cm}
            \begin{alertblock}{黄金法则}
                \centering
                \textbf{AI是助手，不是作者}\\
                \small
                你必须能解释和捍卫论文中的每一句话
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{学术诚信自查清单}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{提交前必查项目：}
            \begin{itemize}
                \item[$\square$] 所有引用已在数据库中验证存在
                \item[$\square$] 实验数据来自真实实验
                \item[$\square$] 核心创新点是原创思考
                \item[$\square$] AI使用已按期刊要求声明
                \item[$\square$] 能够解释论文任何部分
                \item[$\square$] 合作者都同意最终版本
                \item[$\square$] 没有一字不改地使用AI输出
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{记录保留建议：}
            \begin{itemize}
                \item 保存所有AI对话记录
                \item 保留论文各版本草稿
                \item 记录AI使用的具体环节
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{被质疑时的应对：}
            \begin{block}{如果收到AI检测警告}
                \small
                \begin{enumerate}
                    \item \textbf{保持冷静}：检测工具有误判
                    \item \textbf{提供证据}：展示写作过程记录
                    \item \textbf{解释内容}：证明你理解所写内容
                    \item \textbf{展示草稿}：提供修改历史
                \end{enumerate}
            \end{block}
            
            \vspace{0.2cm}
            \begin{exampleblock}{预防措施}
                \small
                \begin{itemize}
                    \item 使用Git管理论文版本
                    \item 用Overleaf等协作平台（有历史记录）
                    \item 定期保存写作进度截图
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\section{科研LLM应用：风险与最佳实践}
%========================================

\begin{frame}{科研LLM应用的关键风险}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{1. 幻觉与事实错误}
            \begin{itemize}
                \item 编造不存在的引用
                \item 虚构实验数据
                \item 错误的数学推导
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{2. 可复现性问题}
            \begin{itemize}
                \item 模型输出随机性
                \item API版本变化
                \item Prompt敏感性
            \end{itemize}
            
            \vspace{0.3cm}
            \textbf{3. 学术诚信风险}
            \begin{itemize}
                \item 过度依赖AI写作
                \item 未声明AI使用
                \item 抄袭检测新挑战
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{4. 偏见传播}
            \begin{itemize}
                \item 训练数据偏见
                \item 领域知识不均衡
                \item 语言/文化偏见
            \end{itemize}
            
            \vspace{0.3cm}
            \begin{alertblock}{FLAWS基准发现 (Xi et al. 2025)}
                \small
                LLM在科学论文错误检测中：
                \begin{itemize}
                    \item 能检测明显错误
                    \item 但细微逻辑错误难以发现
                    \item 过度自信问题普遍
                \end{itemize}
            \end{alertblock}
            
            {\scriptsize arXiv:2511.21843}
        \end{column}
    \end{columns}
\end{frame}

                \texttt{2. 添加公式编号标签}\\
                \texttt{3. 符号要有注释说明}
            \end{exampleblock}
            
            \vspace{0.2cm}
            \textbf{复杂公式推导：}
            \begin{exampleblock}{Prompt模板}
                \scriptsize
                \texttt{请用LaTeX写出从公式A推导到公式B的完整过程：}\\
                \texttt{起始：[公式A]}\\
                \texttt{目标：[公式B]}\\
                \texttt{使用align环境，每步加注释}

        \begin{column}{0.5\textwidth}
            \textbf{示例输出：}
            \begin{lstlisting}[basicstyle=\ttfamily\tiny,language=TeX]
\begin{equation}
\label{eq:loss}
\mathcal{L} = -\sum_{i=1}^{N} 
  y_i \log(\hat{y}_i)
\end{equation}
% where:
% $\mathcal{L}$: cross-entropy loss
% $N$: number of samples
% $y_i$: ground truth label
% $\hat{y}_i$: predicted probability
            \end{lstlisting}
            
            \vspace{0.2cm}
            \begin{alertblock}{技巧}
                \small
                \begin{itemize}
                    \item 指定使用的宏包（amsmath等）
                    \item 要求符号与论文其他部分一致
                    \item 让AI解释每个符号的含义
                \end{itemize}
            \end{alertblock}

Feasibility analysis; 3. Potential challenges."}
    \end{exampleblock}
    
    \vspace{0.3cm}
    \textbf{2. 文献调研与总结 (Literature Review)}

Compare their methodological similarities and differences (create a comparison table); 3. \textbf{Identify} and \textbf{discover} the common limitations, and \textbf{explore} possible future improvements."}
    \end{exampleblock}
\end{frame}

\begin{frame}{附录：科研全流程 Prompt 指南 - 高级技巧}
    \begin{block}{何时使用"探索/联想/发现"关键词？}
        \begin{itemize}
            \item \textbf{Explore}：当需要模型发散思考、提出多种可能性时
            \item \textbf{Associate / Connect}：当希望模型找到不同概念之间的联系时
            \item \textbf{Discover / Identify}：当希望模型从数据或文本中挖掘隐藏的规律时
            \item \textbf{Brainstorm}：当需要创意和头脑风暴时
        \end{itemize}
    \end{block}
    
    \vspace{0.3cm}
    \begin{exampleblock}{高级 Prompt 示例 (带探索关键词)}
        \small
        \texttt{"Please \textbf{explore} unconventional approaches to solve [problem]. \textbf{Associate} techniques from [Field A] with methods in [Field B]. \textbf{Discover} potential synergies. Think step-by-step and \textbf{brainstorm} at least 5 creative solutions, even if some seem unconventional."}
    \end{exampleblock}
\end{frame}
