\chapter{Derivation X: The Kato-Trotter Product Formula}
\label{ch:derivation_x_trotter_limit}

\section{Context}
The lattice theory provides a Transfer Matrix $\mathbb{T}_a = e^{-aH_a}$. The continuum theory requires a Hamiltonian generator $H$. We must prove the discrete time evolution converges to continuous time evolution. This relies on \textbf{Appendix AV (Derivation 47)}.

\section{Theorem X.1 (Strong Semigroup Convergence)}
Let $H_a$ be the lattice Hamiltonian. As $a \to 0$:
\begin{equation}
e^{-t H_a} P_a \to e^{-tH}
\end{equation}
where $P_a$ is the projection from the continuum Hilbert space to the lattice approximation.

\section{Proof Derivation}

\subsection{Step 1: Common Core}
We define a core domain $\mathcal{D} \subset \mathcal{H}$ of "smooth cylinder functions" (functionals dependent on finitely many smeared Wilson loops). We prove $\mathcal{D}$ is dense in the physical Hilbert space $\mathcal{H}_{phys}$.

\subsection{Step 2: Stability Condition}
Using the \textbf{Reflection Positivity} of the transfer matrix, we established $\|\mathbb{T}_a\| \le 1$. This implies the sequence is uniformly bounded (stable): $\|e^{-t H_a}\| \le 1$.

\subsection{Step 3: Consistency}
We show that on the core $\mathcal{D}$, the generators converge:
\begin{equation}
\| (H_a - H) \psi \| \to 0 \quad \text{for } \psi \in \mathcal{D}
\end{equation}
This calculation uses the Taylor expansion of the Wilson action, showing it converges to $\frac{1}{2} E^2 + \frac{1}{2} B^2$.

\subsection{Step 4: Trotter-Chernoff Theorem}
Stability + Consistency on a core implies the strong convergence of the semigroups $e^{-t H_a} \to e^{-t H}$. This rigorously defines the time evolution $U(t) = e^{-itH}$ in the continuum.
