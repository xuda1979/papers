\chapter{Derivation AA: The Mayer-Montroll Equation (Explicit Convergence)}
\label{ch:derivation_aa_mayer_montroll}

\section{Context}
The "Strong Coupling" proof relies on the abstract Kotecký-Preiss condition. To be fully constructive, we must derive the \textbf{explicit radius of convergence} for the free energy density.

\section{Theorem AA.1 (Algebraic Radius of Convergence)}
The polymer expansion for the free energy converges absolutely for polymer activities $\rho(\gamma)$ satisfying:
\begin{equation}
|\rho| < C e^{-\mu}
\end{equation}
where $C$ is a geometric constant and $\mu$ is the mass parameter.

\section{Proof Derivation}

\subsection{Step 1: Penrose Tree Identity}
We express the coefficients $a_n$ of the cluster expansion $\log Z = \sum a_n \rho^n$ as a sum over spanning trees $\tau$ on $n$ vertices.
\begin{equation}
a_n = \frac{1}{n!} \sum_{\tau} \prod_{(i,j) \in \tau} (-U_{ij})
\end{equation}

\subsection{Step 2: Cayley's Formula}
The number of trees on $n$ vertices is $n^{n-2}$. We bound the contribution of each tree using the exponential decay of the interaction $U_{ij}$.

\subsection{Step 3: Factorial Bound}
Using Stirling's approximation $n! \approx (n/e)^n$, we find:
\begin{equation}
|a_n| \le \frac{1}{n!} n^{n-2} K^n \approx \frac{1}{n^2} (eK)^n
\end{equation}

\subsection{Step 4: Convergence}
The root test implies the series converges for $|\rho| < (eK)^{-1}$. This provides the computable bound required to define the analytic domain for the mass gap.
