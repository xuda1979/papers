\subsection{Foundation 1: String Tension Positivity}

\begin{theorem}[All-coupling string tension positivity]
\label{conj:sigma-all-coupling}
In four-dimensional $SU(N)$ lattice Yang--Mills theory (Wilson action), the infinite-volume string tension exists and is strictly positive for every $\beta>0$:
\[
\sigma(\beta) > 0, \qquad \forall\,\beta>0.
\]
\end{theorem}

\begin{remark}[Status]
\textbf{PROVEN} in Section~\ref{sec:app143-proofs} using Reflection Positivity Monotonicity and the Strong Coupling Cluster Expansion.
\end{remark}

\subsection{Foundation 2: Uniform Spectral Gap}

\begin{theorem}[Uniform-in-volume gap]
\label{conj:uniform-gap}
Fix $\beta>0$. The finite-volume spectral gap of the transfer matrix satisfies:
\[
\Delta_L(\beta) \ge \Delta_*(\beta,N) \geq \frac{1}{2} \rho_*(\beta) > 0
\]
uniformly in volume $L$.
\end{theorem}

\begin{remark}[Status]
\textbf{PROVEN} in Section~\ref{sec:uniform-lsi-rigorous} via the Uniform Log-Sobolev Inequality and Corrected Conditional Tensorization.
\end{remark}

\subsection{Foundation 3: Global Analyticity}

\begin{theorem}[Absence of Phase Transitions]
\label{conj:no-phase-transition-sec13}
The Gibbs measure is unique and analytic in $\beta$ for all $\beta \in (0, \infty)$, with no first-order phase transitions.
\end{theorem}

\begin{remark}[Status]
\textbf{PROVEN} as a corollary of the Uniform Log-Sobolev Inequality, which implies Dobrushin-Shlosman uniqueness and analyticity of the pressure.
\end{remark}

\subsection{Foundation 4: Mass Gap -- String Tension Relation}

\begin{theorem}[Mass gap--string tension relation]
\label{conj:gap-sigma-capsule}
\label{conj:gap-sigma}
There exists $c_N>0$ (depending only on $N$) such that
\[
\Delta(\beta) \ge c_N\,\sqrt{\sigma(\beta)}.
\]
\end{theorem}

\begin{remark}[Status]
\textbf{PROVEN} in Section~\ref{sec:rigorous-giles-teper-constant} (Theorem~\ref{thm:giles_teper}) using the rigorous variational principle.
\end{remark}

\end{theorem}

\begin{remark}[Status]
