\section{Supplementary Theorems}
\label{sec:supplementary-theorems}

This section collects key theorems referenced throughout the text.

\begin{theorem}[Analyticity of Free Energy]
\label{thm:convex-analytic}
The free energy density $f(\beta)$ of $SU(N)$ lattice Yang-Mills theory is real-analytic for all $\beta > 0$.
\end{theorem}

\begin{theorem}[Dimensionless Ratio Bound]
\label{thm:ratio-bound}
For all $\beta > 0$, the dimensionless ratio $R(\beta) = \Delta(\beta)/\sqrt{\sigma(\beta)}$ is bounded from below by a positive constant $c_N > 0$.
\end{theorem}

\begin{theorem}[Tropical GKS Inequality]
\label{thm:tropical-gks}
The tropical string tension $\sigma_{\mathrm{trop}}(\beta)$ provides a strict lower bound for the physical string tension, ensuring $\sigma(\beta) > 0$ for all $\beta > 0$.
\end{theorem}

\begin{theorem}[Uniform LSI via Induction]
\label{thm:uniform-lsi-induction}
The Log-Sobolev constant $\alpha$ satisfies the conditional tensorization property, ensuring uniform bounds via inductive decomposition of the lattice.
\end{theorem}

\begin{theorem}[Cluster Decomposition]
\label{thm:cluster}
For all $\beta > 0$, the correlation functions satisfy the cluster decomposition property with a finite correlation length $\xi < \infty$.
\end{theorem}

\begin{corollary}[Smoothness of Continuum Limit]
\label{cor:smooth-continuum}
The continuum limit of the correlation functions is smooth and satisfies the Osterwalder-Schrader axioms.
\end{corollary}

\begin{theorem}[Well-posedness of SPDE]
\label{thm:spde-wellposed}
The stochastic partial differential equation (SPDE) governing the stochastic quantization of Yang-Mills theory is well-posed and has a unique invariant measure.
\end{theorem}

\begin{theorem}[String Tension Positivity]
\label{thm:sigma-positivity-final}
The string tension $\sigma(\beta)$ is strictly positive for all finite $\beta$, ensuring confinement.
\end{theorem}

\begin{theorem}[Transfer Matrix Spectral Gap]
\label{thm:transfer-spectral}
The transfer matrix $T$ has a spectral gap $\gamma > 0$ separating the vacuum from the first excited state.
\end{theorem}

\begin{theorem}[Reflection Positivity]
\label{thm:rp-positivity}
The lattice Yang-Mills measure satisfies Osterwalder-Schrader reflection positivity.
\end{theorem}

\begin{theorem}[Continuum Lyapunov Function]
\label{thm:continuum-lyapunov}
There exists a Lyapunov function for the renormalization group flow that ensures stability in the continuum limit.
\end{theorem}

\begin{theorem}[Complete Spectral Gap]
\label{thm:complete-spectral-gap}
The Hamiltonian of the Yang-Mills theory has a strictly positive mass gap $\Delta > 0$ in the continuum limit.
\end{theorem}

\begin{theorem}[Area Law from OPE]
\label{thm:area-law-ope}
The area law for Wilson loops follows from the Operator Product Expansion of the Wilson loop operator.
\end{theorem}

\begin{theorem}[Epsilon Regularity]
\label{thm:epsilon-regularity}
The gauge field configurations satisfy an epsilon-regularity condition, ensuring smoothness at small scales.
\end{theorem}

\begin{theorem}[Uhlenbeck Gauge Fixing]
\label{thm:uhlenbeck-gauge-fix}
There exists a global gauge fixing (Uhlenbeck gauge) that controls the Sobolev norms of the gauge field.
\end{theorem}

\begin{theorem}[Bubble Prevention]
\label{thm:bubble-prevention}
The formation of small-scale bubbles (singularities) is prevented by the energy bounds.
\end{theorem}

\begin{theorem}[No Bubbles in Measure]
\label{thm:no-bubbles-measure}
The set of singular configurations has measure zero in the path integral.
\end{theorem}

\begin{theorem}[Gap Survives Continuum Limit]
\label{thm:gap-survives}
The spectral gap $\Delta(a)$ scales as $a^{-1}$ and remains positive in the limit $a \to 0$.
\end{theorem}

\begin{theorem}[Beta Function]
\label{thm:beta-function}
The beta function $\beta(g)$ is negative for small $g$, ensuring asymptotic freedom.
\end{theorem}

\begin{theorem}[Exponential Clustering]
\label{thm:exponential-clustering}
Correlation functions decay exponentially with distance: $|\langle O(x) O(y) \rangle| \le C e^{-m|x-y|}$.
\end{theorem}

\begin{conjecture}[Global Curvature]
\label{conj:global-curvature}
The Ricci curvature of the gauge orbit space is globally positive.
\end{conjecture}

\begin{theorem}[Curvature Gap]
\label{thm:curv_gap}

\end{theorem}
