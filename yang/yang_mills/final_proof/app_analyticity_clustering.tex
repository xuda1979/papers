\section{Global Analyticity from Exponential Clustering}
\label{app:analyticity_clustering}

This appendix provides a rigorous proof of global analyticity for the infinite-volume pressure and local observables, replacing the earlier reliance on Lee--Yang zero assumptions (Hypothesis J.3). We show that uniform exponential clustering (mass gap), which is established via the geometric/LSI bounds in the main text, is sufficient to prove analyticity on the positive real axis.

\subsection{Setup and Definitions}

Let $\Lambda \subset \mathbb{Z}^4$ be a finite periodic box (torus). Let $E(\Lambda)$ be its oriented edges and $P(\Lambda)$ its plaquettes. A configuration is $U = (U_e)_{e \in E(\Lambda)} \in SU(N)^{E(\Lambda)}$. We consider the Wilson plaquette observable:
\begin{equation}
\phi_p(U) := \frac{1}{N} \mathrm{Re} \, \mathrm{Tr}(U_p) \in [-1, 1], \quad U_p = \prod_{e \in \partial p} U_e^{\pm 1}.
\end{equation}
The action is defined as:
\begin{equation}
S_\Lambda(U) = \sum_{p \in P(\Lambda)} \phi_p(U), \qquad \mu_{\Lambda, \beta}(dU) = \frac{1}{Z_\Lambda(\beta)} e^{\beta S_\Lambda(U)} \prod_{e \in E(\Lambda)} dU_e.
\end{equation}
For a bounded local observable $F$ (depending on finitely many edges), we define $\langle F \rangle_{\Lambda, \beta} = \int F \, d\mu_{\Lambda, \beta}$. The pressure (free-energy density) is:
\begin{equation}
f_\Lambda(\beta) := \frac{1}{|P(\Lambda)|} \log Z_\Lambda(\beta).
\end{equation}
We are interested in the infinite-volume limit $f(\beta) = \lim_{\Lambda \nearrow \mathbb{Z}^4} f_\Lambda(\beta)$ and its analyticity in $\beta \in (0, \infty)$.

\subsection{Theorem A: Analyticity from Exponential Clustering}

\begin{theorem}[Analyticity from Exponential Clustering]
\label{thm:analyticity_clustering}
Assume that for a given $\beta_0 > 0$ there is an infinite-volume Gibbs state $\langle \cdot \rangle_{\beta_0}$ and constants $C_0, m_0 > 0$ such that the following \textbf{uniform exponential clustering bound} holds for all bounded local observables $A, B$:
\begin{equation}
\big| \mathrm{Cov}_{\beta_0}(A, B) \big| := \big| \langle AB \rangle_{\beta_0} - \langle A \rangle_{\beta_0} \langle B \rangle_{\beta_0} \big| \le C_0 \|A\|_\infty \|B\|_\infty e^{-m_0 \, \mathrm{dist}(\mathrm{supp} A, \mathrm{supp} B)}.
\label{eq:clustering_bound}
\end{equation}
Then there exists an explicit radius $\delta(\beta_0) > 0$ such that:
\begin{enumerate}
    \item The thermodynamic pressure $f(\beta)$ exists for all $\beta$ with $|\beta - \beta_0| < \delta(\beta_0)$ (complex $\beta$ allowed), and $\beta \mapsto f(\beta)$ is analytic on that disk.
    \item For every fixed local observable $F$, the map $\beta \mapsto \langle F \rangle_{\beta}$ exists and is analytic on the same disk.
\end{enumerate}
Consequently, if \eqref{eq:clustering_bound} holds for every $\beta_0 \in (0, \infty)$ (i.e., a mass gap exists for all couplings), then $f(\beta)$ and every local expectation $\langle F \rangle_\beta$ are \textbf{analytic on all of $(0, \infty)$}, implying no bulk phase transition along the positive real axis.
\end{theorem}

\subsection{Proof of Theorem A}

\paragraph{Step 1: Derivatives are cumulants.}
For finite $\Lambda$, $Z_\Lambda(\beta)$ is an entire function of $\beta$. Differentiating under the integral gives:
\begin{equation}
\frac{d^n}{d\beta^n} \log Z_\Lambda(\beta) = \kappa_{\Lambda, \beta}^{(n)}(S_\Lambda),
\end{equation}
where $\kappa^{(n)}$ denotes the $n$-th cumulant. Hence,
\begin{equation}
f_\Lambda^{(n)}(\beta) = \frac{1}{|P(\Lambda)|} \kappa_{\Lambda, \beta}^{(n)}(S_\Lambda).
\label{eq:pressure_cumulant}
\end{equation}
Similarly, for a local observable $F$,
\begin{equation}
\frac{d^n}{d\beta^n} \langle F \rangle_{\Lambda, \beta} = \kappa_{\Lambda, \beta}^{(n+1)}(F, S_\Lambda, \ldots, S_\Lambda).
\label{eq:observable_cumulant}
\end{equation}
Analyticity reduces to proving factorial bounds on these cumulants that are uniform in volume.

\paragraph{Step 2: A tree bound for cumulants.}
Let $X_1, \dots, X_n$ be bounded random variables. The standard tree-graph inequality states:
\begin{equation}
|\kappa(X_1, \dots, X_n)| \le 2^{n-1} n! \sum_{T \in \mathcal{T}_n} \prod_{\{i, j\} \in T} |\mathrm{Cov}(X_i, X_j)| \prod_{k=1}^n \|X_k\|_\infty,
\label{eq:tree_bound}
\end{equation}
where the sum is over spanning trees $T$ on $\{1, \dots, n\}$. Applying this to plaquette variables $X_i = \phi_{p_i}$ (with $\|X_i\|_\infty \le 1$) and using \eqref{eq:clustering_bound}:
\begin{equation}
|\kappa_{\beta_0}(\phi_{p_1}, \dots, \phi_{p_n})| \le 2^{n-1} n! \sum_{T} \prod_{\{i, j\} \in T} \left( C_0 e^{-m_0 d(p_i, p_j)} \right).
\end{equation}

\paragraph{Step 3: Summing over plaquettes.}
Using multilinearity, $\kappa_{\Lambda, \beta_0}^{(n)}(S_\Lambda) = \sum_{p_1, \dots, p_n \in P(\Lambda)} \kappa_{\Lambda, \beta_0}(\phi_{p_1}, \dots, \phi_{p_n})$.
Let $B(m_0) := \sup_{p} \sum_{q \in P(\mathbb{Z}^4)} e^{-m_0 d(p, q)} < \infty$. Summing along the edges of a tree $T$ rooted at $p_1$ yields a factor of $B(m_0)^{n-1}$. Summing over the root $p_1$ gives $|P(\Lambda)|$. Since there are $n^{n-2}$ spanning trees, we have:
\begin{equation}
|\kappa_{\Lambda, \beta_0}^{(n)}(S_\Lambda)| \le |P(\Lambda)| \cdot 2^{n-1} \cdot n^{n-2} \cdot C_0^{n-1} B(m_0)^{n-1}.
\end{equation}
Using $n^{n-2} \le e^n n!$, we obtain the bound:
\begin{equation}
|f_\Lambda^{(n)}(\beta_0)| \le n! K_0^n, \quad \text{where } K_0 := 2e C_0 B(m_0).
\end{equation}
A similar bound holds for local observables $\langle F \rangle$.

\paragraph{Step 4: Uniform Taylor convergence.}
The Taylor series for $f_\Lambda(\beta)$ converges absolutely for $|\beta - \beta_0| < 1/K_0$, uniformly in $\Lambda$. This implies that $\{f_\Lambda\}$ is a normal family, and the thermodynamic limit $f(\beta)$ is analytic in this disk. Since $\beta_0$ was arbitrary, analyticity holds on $(0, \infty)$.

\qed
