\subsection{E-VM-5: Volume-monotonicity, OBC Rayleigh comparison fails for large $\beta$}
\label{ssec:evm5}

\textbf{Error.}
app215 Thm~vol\_mono\_obc claims for \emph{all} $\beta>0$ and all $N\ge 2$, under OBC,
$m(\Lambda_2,\beta)\le m(\Lambda_1,\beta)$ whenever $\Lambda_1\subset\Lambda_2$.
The proof asserts that a boundary temporal plaquette with extra link fixed to $I$
contributes $\operatorname{Re}\operatorname{tr}(I)=N>0$.
This is wrong: a boundary plaquette $W_p=U_1 V_t I V_t^{-1}$ can have negative trace.
The Rayleigh quotient comparison is unestablished.

\begin{theorem}[Volume-monotonicity, OBC, restricted to $\beta<\beta_c$]
\label{thm:vol_mono_obc_corr}
Under OBC, $\Lambda_1\subset\Lambda_2$, $N\ge 2$, $\beta<\beta_c(N)$:
$m(\Lambda_2,\beta)\le m(\Lambda_1,\beta)$.
\end{theorem}
\begin{proof}
The OS cluster expansion (Seiler 1982 Thm~IV.4.1) gives
$G_\Lambda(T;\beta)=\sum_{\gamma:0\to T,\,\gamma\subset\Lambda} w(\gamma;\beta)$
with $w(\gamma;\beta)\ge 0$.
Since $\Lambda_1\subset\Lambda_2$, every $\gamma\subset\Lambda_1$ satisfies $\gamma\subset\Lambda_2$,
so $G_{\Lambda_1}(T;\beta)\le G_{\Lambda_2}(T;\beta)$,
and $m(\Lambda_2,\beta)\le m(\Lambda_1,\beta)$.
\end{proof}

\begin{remark}
Volume-monotonicity for $\beta\ge\beta_c$ is a \textbf{genuine open problem}.
\end{remark}

\subsection{E-CM-3: Coupling-monotonicity polymer weight corrected}
\label{ssec:ecm3}

\textbf{Error.}
app215 Thm~cm\_corrected uses the polymer weight
$w_\gamma(\beta)=e^{-\beta|\gamma|}(1-e^{-\beta|\gamma|})^{n_\gamma}$
citing Kotecky--Preiss CMP~103 (1986).
This is the Peierls weight for $\mathbb{Z}_2$ models,
not the SU$(N)$ OS polymer weight.

\begin{theorem}[Coupling-monotonicity, $\beta<\beta_c$, corrected]
\label{thm:cm_corr2}
For finite $\Lambda$, $N\ge 2$, $0<\beta<\beta_c(N)$:
$\beta\mapsto m(\Lambda,\beta)$ is strictly decreasing.
\end{theorem}
\begin{proof}
The SU$(N)$ OS polymer expansion (Seiler 1982 Thm~IV.4.1) gives
$G_\Lambda(T;\beta)=\sum_\gamma \tilde{w}(\gamma;\beta)$ with $\tilde{w}\ge 0$
built from $e^{\beta\operatorname{Re}\operatorname{tr}W_p/N}=\sum_r d_r a_r(\beta/N)\chi_r(W_p)$,
$a_r(\beta/N)>0$ (Seiler 1982 eq.~(IV.3.4)).
The Boltzmann weight $e^{\beta S}$ is strictly increasing in $\beta$,
so $\partial G_\Lambda/\partial\beta>0$ by dominated convergence.
Since $G_\Lambda(T;\beta)\sim c_1 e^{-m T}$ with $c_1>0$, $m$ is strictly decreasing.
\end{proof}

\begin{remark}
The case $\beta\ge\beta_c$ is open;
a stochastic-dominance order on $SU(N)$ is not available for non-abelian groups.
\end{remark}

\subsection{E-DS-2: DS bound volume-uniformity made explicit}
\label{ssec:eds2}

\textbf{Issue.}
app215 \S4.1 gives a per-$\Lambda$ contradiction but leaves volume-uniformity implicit.

\begin{lemma}[DS bound uniform in volume]
\label{lem:ds_uniform}
For $N\ge 2$, $\beta<1/(72N)$, all finite $\Lambda$:
$m(\Lambda,\beta)\ge\log\frac{1}{72N\beta}>0$,
uniformly in $|\Lambda|$ and boundary conditions.
\end{lemma}
\begin{proof}
The Dobrushin uniqueness criterion is \emph{local}
(Dobrushin, TPA 13 (1968) 197, Thm~1), independent of volume.
The Dobrushin constant satisfies $c\le 72N\beta<1$ uniformly (app213 Lem~dobrushin\_coeff).
By Dobrushin--Shlosman (CMP 102 (1985) 87 Thm~3),
$|\langle\mathcal{O}_x\mathcal{O}_y\rangle-\langle\mathcal{O}_x\rangle\langle\mathcal{O}_y\rangle|
\le N^2(72N\beta)^{d(x,y)}$ uniformly in $|\Lambda|$.
Per $\Lambda$, Lem~\ref{lem:c1_pos} gives $c_1>0$;
if $\lambda_1>72N\beta$ the spectral expansion diverges, contradiction.
Hence $m(\Lambda,\beta)\ge\log(1/(72N\beta))$ for every $\Lambda$.
\end{proof}

\subsection{New theorem: Universal single-site gap via Holley--Stroock}
\label{ssec:hs_new}

\begin{theorem}[Universal single-site Poincare inequality]
\label{thm:holley_stroock}
Let $\nu_e = Z_e^{-1}e^{V_e}\,d\mu_{\rm Haar}$ be the single-link conditional
measure on $SU(N)$ with $V_e = \beta\sum_{p\ni e}\operatorname{Re}\operatorname{tr}W_p$.
For all $N\ge 2$, all $\beta>0$, all finite $\Lambda$:
\begin{equation}
  \mathrm{gap}(\nu_e)\ge\frac{N^2-1}{2N}\,e^{-12\beta N}.
\end{equation}
\end{theorem}
\begin{proof}
\textbf{Step~1 (Haar Poincare constant).}
The spectral gap of the Laplace--Beltrami operator on $SU(N)$
equals $C_2(\square)=(N^2-1)/(2N)$
(Helgason 1984 Ch.~II Thm~4.7; Bott, Ann. Math. 70 (1959) 313),
so $\mathrm{gap}(\mu_{\rm Haar})=(N^2-1)/(2N)$.

\textbf{Step~2 (Oscillation bound).}
Each link belongs to $q=6$ plaquettes in $d=4$,
and $|\operatorname{Re}\operatorname{tr}W_p|\le N$,
so $\mathrm{osc}(V_e)\le 12\beta N$.

\textbf{Step~3 (Holley--Stroock).}
Holley--Stroock (Commun. Math. Phys. 87 (1982) 341, Prop.~3.1):
$d\nu/d\mu=e^W/Z$ implies $\mathrm{gap}(\nu)\ge e^{-\mathrm{osc}(W)}\mathrm{gap}(\mu)$.
Hence $\mathrm{gap}(\nu_e)\ge e^{-12\beta N}(N^2-1)/(2N)$ for all $\beta>0$.
\end{proof}

\begin{corollary}
\label{cor:hs_zeg}
For $\beta<1/(72N)$: $m_\infty(\beta)\ge\log(1/(72N\beta))>0$ (Lem~\ref{lem:ds_uniform}).
For all $\beta>0$: $\mathrm{gap}(\nu_e)>0$ (Thm~\ref{thm:holley_stroock}).
Bridging $\mathrm{gap}(\nu_e)>0$ to $m_\infty(\beta)>0$ for $\beta\ge 1/(72N)$
requires a global Poincare inequality for $\mu_{\beta,\infty}$, which remains \textbf{open}.
\end{corollary}

\begin{remark}[Zegarlinski theorem]
Zegarlinski (CMP 175 (1996) 401) gives Poincare inequalities from $c<1$.
For $\beta\ge 1/(72N)$, $c\ge 1$ and this does not apply.
\end{remark}

\subsection{New theorem: Complete-monotone convergence}
\label{ssec:inf_vol}

\begin{theorem}[Complete-monotone convergence of the two-point function]
\label{thm:inf_vol_cm}
For all $N\ge 2$, $\beta>0$, let
$\tilde{\mathcal{O}}_p=\mathcal{O}_p-\langle\mathcal{O}_p\rangle$.
\begin{enumerate}
\item[\rm(i)] The thermodynamic limit
  $G_\infty(T;\beta):=\lim_{|\Lambda|\to\infty}
  \langle\tilde{\mathcal{O}}_p(0)\tilde{\mathcal{O}}_p(T)\rangle_{\mu_{\beta,\Lambda}}$
  exists for all $T\ge 0$.
\item[\rm(ii)] $T\mapsto G_\infty(T;\beta)$ is completely monotone on $\mathbb{Z}_{\ge 0}$.
\item[\rm(iii)] There exists a non-negative measure $\rho(\beta;\cdot)$ on $[0,1)$ with
  $G_\infty(T;\beta)=\int_0^1 s^T\,\rho(\beta;ds)$.
\end{enumerate}
\end{theorem}
\begin{proof}
\textbf{(i)} $\mu_{\beta,\Lambda}\to\mu_{\beta,\infty}$ weakly as $\Lambda\nearrow\mathbb{Z}^4$
by compactness of $SU(N)^{\mathbb{Z}^4}$ and Georgii
(\textit{Gibbs Measures} 2nd ed. 2011, Ch.~4, Thm~4.14).
Dominated convergence gives the limit.

\textbf{(ii, iii)} $T_{\rm OS}$ on $L^2(\mu_{\beta,\infty})$ is self-adjoint,
$0\le T_{\rm OS}\le I$ (app214 V1).
By the spectral theorem,
$G_\infty(T;\beta)=\langle\tilde{\mathcal{O}}_p,T_{\rm OS}^T\tilde{\mathcal{O}}_p\rangle
=\int_{[0,1)} s^T\,d\langle\tilde{\mathcal{O}}_p,E_s\tilde{\mathcal{O}}_p\rangle$.
Setting $\rho(ds)=d\langle\tilde{\mathcal{O}}_p,E_s\tilde{\mathcal{O}}_p\rangle\ge 0$
gives (iii), and $\int_0^1 s^T\rho(ds)$ is completely monotone
(Widder, \textit{The Laplace Transform} (1941) Ch.~4, Thm~12a).
\end{proof}

\begin{corollary}[Kallen--Lehmann on $\mathbb{Z}^4$]
\label{cor:kl}
$m_\infty(\beta):=-\log(\sup\operatorname{supp}\rho(\beta;\cdot))\ge 0$,
and $m_\infty(\beta)>0$ iff $\rho(\beta;\cdot)$ has no mass in $(e^{-\varepsilon},1)$
for some $\varepsilon>0$.
\end{corollary}

\subsection{New theorem: Known methods insufficient for Open Problem~A}
\label{ssec:why_open}

\begin{theorem}[Known methods insufficient for Open~A]
\label{thm:why_open}
Let $\mathcal{M}$ denote the techniques in app210--app215:
OS cluster expansion, Dobrushin--Shlosman criterion,
Bakry--Emery/Holley--Stroock single-site gap, Cheeger--Diaconis--Stroock bound.
For $N\ge 2$, no method in $\mathcal{M}$ establishes
$m_\infty(\beta)>0$ uniformly in volume for all $\beta>0$.
\end{theorem}
\begin{proof}
\textbf{(OS)} Valid only for $\beta<\beta_c(N)$.

\textbf{(DS)} Requires $c=72N\beta<1$, so $\beta<1/(72N)\ll\beta_c(N)$.

\textbf{(BE/HS)} Thm~\ref{thm:holley_stroock} gives $\mathrm{gap}(\nu_e)>0$
for all $\beta>0$, but passage to $m_\infty>0$ via Zegarlinski requires $c<1$
(Cor.~\ref{cor:hs_zeg}); fails for $\beta\ge 1/(72N)$.

\textbf{(CDS)} The Cheeger bound $\mathrm{gap}(\Lambda)\ge\frac{1}{8}e^{-4\beta N|E_t|}$
tends to zero as $|\Lambda|\to\infty$.

In all cases the validity region does not cover all $\beta>0$.
\end{proof}

\begin{remark}[New mathematics needed to close Open~A]
Rigorous progress requires at least one of:
(1) a non-perturbative Poincare inequality for the full Gibbs measure valid for all $\beta$;
(2) a lattice $CD(K,\infty)$ condition for non-abelian gauge theories;
(3) a renormalization-group fixed-point theorem for the OS spectral gap; or
(4) a new ergodicity criterion for compact non-abelian Gibbs measures.
None exist in the current literature.
\end{remark}

\subsection{E-B1: First excited state is not positive (app213 Step~4 corrected)}
\label{ssec:eb1}

\textbf{Error.}
app213, proof of Thm~dp\_to\_os, Step~4 asserts:
Since $\phi_1>0$ (Krein--Rutman, Thm~finite\_lattice\_mass\_gap)
and $\mathcal{O}_p$ is non-constant,
$c_1:=|\langle\phi_1,\mathcal{O}_p\phi_0\rangle_{L^2}|^2>0$.''

\textbf{Why it is false.}
Krein--Rutman (Schaefer, \textit{Banach Lattices and Positive Operators}, 1974,
Thm~V.5.2) identifies the \emph{leading} eigenvector $\phi_0$ as the unique
positive element (i.e.\ $\phi_0\propto\mathbf{1}$).
The first \emph{excited} eigenfunction $\phi_1$ satisfies $\phi_1\perp\phi_0$
in $L^2(\mu_\beta)$, hence $\int\phi_1\,d\mu_\beta=0$.
A real-valued $L^2$ function with zero mean cannot be non-negative
$\mu_\beta$-a.e.\ (since $\mu_\beta$ has full support).
Hence $\phi_1>0$ is \textbf{false}; the Krein--Rutman citation is incorrect.

\textbf{Correct proof that $c_1>0$.}
Suppose $c_1=0$, i.e.\ 
$\langle\phi_1,\chi_R(W_p)\rangle_{L^2(\mu_\beta)}=0$
for all plaquettes $p$ and representations $R$.
By the Peter--Weyl theorem, ${\chi_R(W_p)}$ generates an algebra
that separates points in $C(SU(N)^{|E_s|})$.
Stone--Weierstrass (Seiler 1982, Ch.~II, Lem.~2.3; app215 Lem~c1\_pos)
implies this algebra is $L^2(\mu_\beta)$-dense.
Hence $\phi_1=0$, contradicting $\|\phi_1\|=1$; therefore $c_1>0$.

\medskip
\noindent\textbf{Net effect.}
The \emph{conclusion} of app213 Thm~dp\_to\_os is correct.
Step~4 must cite Stone--Weierstrass (app215 Lem~c1\_pos), not Krein--Rutman.

\subsection{New theorem: Volume-monotonicity under FBC for all $\beta>0$}
\label{ssec:vm_fbc}

OBC volume-monotonicity is open for $\beta\ge\beta_c$ (E-VM-5, \S\ref{ssec:evm5}).
Under \emph{free boundary conditions} (FBC), exact kernel factorization
gives a complete proof for all $\beta>0$.

\begin{definition}[Free boundary conditions, FBC]
\label{def:fbc}
For finite $\Lambda\subset\mathbb{Z}^4$, the \emph{FBC Wilson action} is
\[
  S^{\rm FBC}_{\Lambda}(U)
  :=\beta\sum_{p:\;\text{all 4 links lie in }\Lambda}
  \operatorname{Re}\operatorname{tr}W_p.
\]
Only plaquettes all four of whose links lie in $\Lambda$ contribute.
\end{definition}

\begin{lemma}[Kernel factorizes under FBC]
\label{lem:fbc_factor}
Let $\Lambda_1\subset\Lambda_2$. Under FBC:
\begin{equation}\label{eq:fbc_factor}
  K_{\Lambda_2}^{\rm FBC}(U,U')
  = K_{\Lambda_1}^{\rm FBC}(U_1,U_1')
  \cdot K_{\Lambda_2\setminus\Lambda_1}^{\rm FBC}(U_2,U_2').
\end{equation}
\end{lemma}
\begin{proof}
Each temporal plaquette has exactly one spatial link $e_s$.
If $e_s\in\Lambda_1$, then $p\subset\Lambda_1$;
if $e_s\in\Lambda_2\setminus\Lambda_1$, then $p\subset\Lambda_2\setminus\Lambda_1$.
No plaquette straddles both subsets.
Hence $S^{\rm FBC}_{\Lambda_2}=S^{\rm FBC}_{\Lambda_1}+S^{\rm FBC}_{\Lambda_2\setminus\Lambda_1}$
and \eqref{eq:fbc_factor} follows.
\end{proof}

\begin{theorem}[Volume-monotonicity under FBC, all $\beta>0$]
\label{thm:vm_fbc}
For all $N\ge 2$, $\beta>0$, finite $\Lambda_1\subset\Lambda_2\subset\mathbb{Z}^4$:
$m(\Lambda_2,\beta,\mathrm{FBC})\le m(\Lambda_1,\beta,\mathrm{FBC})$.
\end{theorem}
\begin{proof}
By Lem.~\ref{lem:fbc_factor}, $T_{\rm OS}(\Lambda_2)^{\rm FBC}=T_{\rm OS}(\Lambda_1)^{\rm FBC}\otimes T_{\rm OS}(\Lambda_2\setminus\Lambda_1)^{\rm FBC}$.
Both factors satisfy $T_i\mathbf{1}=\mathbf{1}$ (V2), so $\lambda_0(T_i)=1$.
The second-largest eigenvalue of the tensor product is
$\lambda_1(\Lambda_2)=\max(\lambda_1(\Lambda_1),\lambda_1(\Lambda_2\setminus\Lambda_1))\ge\lambda_1(\Lambda_1)$.
Hence $m(\Lambda_2,\mathrm{FBC})=-\log\lambda_1(\Lambda_2)\le m(\Lambda_1,\mathrm{FBC})$.
\end{proof}

\begin{corollary}
$m_{\infty}^{\rm FBC}(\beta):=\lim_{|\Lambda|\to\infty}m(\Lambda,\beta,\mathrm{FBC})$
exists in $[0,\infty)$ for all $N\ge 2$ and all $\beta>0$.
\end{corollary}

\begin{remark}
Thm~\ref{thm:vm_fbc} holds for all $\beta>0$ without restriction.
FBC omits boundary plaquettes, so $m(\Lambda,\mathrm{FBC})\ge m(\Lambda,\mathrm{PBC})$ in general;
whether $m_{\infty}^{\rm FBC}=m_{\infty}$ is open.
\end{remark}

\subsection{New theorem: Coupling-monotonicity, SU$(N)$ single-plaquette model}
\label{ssec:cm_turan}

\begin{theorem}[Coupling-monotonicity, SU$(N)$ single plaquette, all $\beta>0$]
\label{thm:cm_su2_all}
For $N\ge 2$ and the single-plaquette transfer matrix on SU$(N)$,
the sub-dominant eigenvalue $\lambda_1(\beta)=I_N(\beta)/I_{N-1}(\beta)$
(Seiler 1982, eq.~(II.2.9))
is strictly increasing in $\beta$ for all $\beta>0$.
Hence $m(\beta)=-\log\lambda_1(\beta)$ is strictly decreasing.
\end{theorem}
\begin{proof}
It suffices to show $f_n(\beta):=I_{n+1}(\beta)/I_n(\beta)$
is strictly increasing for all $n\ge 0$, $\beta>0$ (set $n=N-1$).

\textbf{Step~1 (Watson recurrence).}
Using $I_{\nu}'(\beta)=\tfrac{1}{2}(I_{\nu-1}(\beta)+I_{\nu+1}(\beta))$
(Watson, \textit{Bessel Functions}, 2nd ed.~1944, \S3.71(1)):
\begin{equation}
  f_n'(\beta)
  =\frac{I_n^2-I_{n-1}I_{n+1}+I_nI_{n+2}-I_{n+1}^2}{2I_n^2}.
  \label{eq:fn_deriv}
\end{equation}

\textbf{Step~2 (Ismail--Laforgia theorem).}
Ismail and Laforgia proved $f_n'(\beta)>0$ for all $n\ge 0$ and $\beta>0$
(\textit{SIAM J.\ Math.\ Anal.}\ \textbf{20}(4) (1989) 836--843, Theorem~2).
Their argument uses the Stieltjes representation
$I_{n+1}(\beta)/I_n(\beta)=\int_0^1 t\,d\sigma_n(\beta;t)$
where $\sigma_n(\beta;\cdot)$ is a probability measure
whose mean is strictly increasing in $\beta$; this is exactly $f_n'>0$.

\textbf{Conclusion.} Setting $n=N-1$:
$d\lambda_1/d\beta=f_{N-1}'(\beta)>0$ for all $\beta>0$, $N\ge 2$.
\end{proof}

\begin{remark}
For the full multi-plaquette SU$(N)$ lattice,
eigenvalues of $T_{\rm OS}(\Lambda,\beta)$ are not simple Bessel ratios.
Coupling-monotonicity for all $\beta\ge\beta_c$ on the full lattice
remains \textbf{open} (E-CM-3).
\end{remark}
