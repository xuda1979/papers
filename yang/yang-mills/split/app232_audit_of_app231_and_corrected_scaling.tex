
\section{Audit of app231: scaling, local log-concavity, and volume-ratio
errors; the mass gap remains open}
\label{app:audit_app231_scaling}

This appendix audits the unpublished draft
\texttt{app231\_concentration\_via\_faddeev\_popov.tex}, whose central
claim is the mass-gap theorem (\texttt{thm:mass\_gap\_app231}). We
identify \emph{four independent} errors that invalidate that claim, and
we state precisely what would be required to repair each. The net
conclusion is that the Clay Millennium Prize mass-gap conjecture remains
\textbf{open}, still reduced to the gauge-invariant concentration
Hypothesis~\ref{hyp:gauge_invariant_concentration} at the \emph{correct}
per-scale exponent: the boundary case of app226 with
$h_{j}^{\nu}\ge c\,2^{2j}/(j+1)$, not the $2^{j}$ exponent that
app231 actually proves.

\subsection{The precise per-scale requirement (recall of app230)}

Throughout, $j$ is the block scale, $a_{j}=2^{-j}$ the lattice spacing,
$\mathcal{Q}_{j}=\mathcal{A}_{j}/\mathcal{G}_{j}$ the gauge orbit,
$D_{j}^{\perp}\sim c_{\dim}'\,N^{2}\,2^{4j}$ the transverse dimension,
$S_{\mathrm{YM},j}(A)=\tfrac{1}{2g_{\mathrm{YM}}^{2}}\|F_{A}\|_{L^{2}}^{2}$
the Yang--Mills action, and
$\nu_{j}=Z_{j}^{-1}e^{-S_{\mathrm{YM},j}}\,d\mathrm{vol}_{g_{j}}$ the
gauge-invariant quotient measure.

app230 (Theorem~\ref{thm:app229_error2} and
Remark~\ref{rem:corrected_still_half}) established the following sharp
scaling fact, which is the crux.

\begin{theorem}[The per-scale exponent needed for a mass gap]
\label{thm:per_scale_exponent_needed}
Let $h_{j}^{\nu}$ denote the per-scale Cheeger constant of $\nu_{j}$ on
$\mathcal{Q}_{j}$. If the mass gap is to follow from the multiscale
aggregation (the Cheeger--Lawler--Sokal bound for the OS transfer
matrix), one requires the per-scale lower bound
\begin{equation}
\label{eq:needed_per_scale}
  h_{j}^{\nu}\;\ge\;c_{\mathrm{need}}\,\frac{2^{2j}}{j+1},
  \qquad c_{\mathrm{need}}>0,\ \text{independent of }j,n.
\end{equation}
The $\epsilon=1/2$ boundary bound
\begin{equation}
\label{eq:insufficient_per_scale}
  h_{j}^{\nu}\;\ge\;c\,\frac{2^{j}}{(j+1)^{1/2}}
\end{equation}
is \emph{insufficient}: it yields an OS transfer-matrix gap
$1-\lambda_{1}^{(n)}\ge c\,a_{n}^{2}$ in lattice units, hence a
physical mass $m_{n}\ge c\,a_{n}\to 0$, not $>0$.
\end{theorem}

\begin{proof}
This is exactly Theorem~\ref{thm:app229_error2} (app230), restated for
emphasis. The normalized Cheeger bound reads
\begin{equation*}
  1-\lambda_{1}^{(n)}\;\ge\;\frac{\bigl(h_{\mathrm{RP}}^{(n)}\bigr)^{2}}{
  2\,\mathfrak{D}_{\mathrm{phys}}\,a_{n}^{-2}\,L_{\mathrm{phys}}^{2}},
\end{equation*}
where $h_{\mathrm{RP}}^{(n)}$ is the OS transfer-matrix Cheeger constant
in lattice units and $\mathfrak{D}_{\mathrm{phys}}$ is the physical
diffusivity. Under \eqref{eq:insufficient_per_scale} the multiscale
aggregation gives $h_{\mathrm{RP}}^{(n)}\sim c$ (lattice units), so
$1-\lambda_{1}^{(n)}\sim c^{2}a_{n}^{2}$ and
$m_{n}= (1-\lambda_{1}^{(n)})/a_{n}\sim c^{2}a_{n}\to 0$. Only
\eqref{eq:needed_per_scale}, which gives
$h_{\mathrm{RP}}^{(n)}\ge c\,a_{n}^{-1/2}$ (lattice), yields
$1-\lambda_{1}^{(n)}\sim c^{2}$ and $m_{n}\sim c^{2}>0$.
\end{proof}

\subsection{Error 1: app231 proves the wrong exponent and asserts the
stronger one}

\begin{theorem}[Error 1]
\label{thm:app231_error1}
The per-scale Cheeger bound actually proved in app231
(Theorem~\ref{thm:sharp_cheeger_corr} of app230, which app231 adopts) is
\eqref{eq:insufficient_per_scale} with exponent $2^{j}/(j+1)^{1/2}$.
The mass-gap theorem \texttt{thm:mass\_gap\_app231} nevertheless asserts
the stronger aggregation
$h_{\mathrm{OS}}^{(n)}\ge c_{\mathrm{OS}}\,a_{n}^{-1/2}$ (lattice units),
which requires the unproved exponent \eqref{eq:needed_per_scale}. The
assertion is therefore unjustified and repeats the very scaling error
(Error 2, Theorem~\ref{thm:app229_error2}) that app230 corrected.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:per_scale_exponent_needed}, exponent
\eqref{eq:insufficient_per_scale} is insufficient and
\eqref{eq:needed_per_scale} is the only route to $m_{n}>0$. app231 proves only
\eqref{eq:insufficient_per_scale} (its \texttt{thm:sharp\_cheeger\_corr}
and the proof of its \texttt{thm:mass\_gap\_app231} use
$h_{j}^{\nu}\ge c_{h}\,2^{j}/(j+1)^{1/2}$) yet concludes
$m_{\mathrm{phys}}>0$. The jump from $2^{j}$ to $2^{2j}$ in the exponent
is exactly the factor-of-$2^{j}$ discrepancy that app230 identified.
\end{proof}

\subsection{Error 2: local positive Hessian does not give log-concavity
or a usable tail inequality}

\begin{theorem}[Error 2]
\label{thm:app231_error2}
The log-concavity step (Theorem~\ref{thm:log_concave},
Lemma~\ref{lem:log_concave_tail}, Corollary~\ref{cor:af_concentration}
of app231) is invalid: (i) positive Hessian of $S_{\mathrm{YM},j}$ on
the subset $U_{\mathrm{AF},j}^{\mathrm{corr}}$ does not make the density
$e^{-S_{\mathrm{YM},j}-c}$ log-concave on $\mathcal{Q}_{j}$, and (ii)
the log-concave tail inequality
(Lemma~\ref{lem:log_concave_tail}) requires a \emph{globally}
log-concave probability measure, which is not established.
\end{theorem}

\begin{proof}
Log-concavity of $\nu_{j}$ is the global statement that
$\log(e^{-S_{\mathrm{YM},j}})=-S_{\mathrm{YM},j}$ is convex on all of
$\mathcal{Q}_{j}$. Convexity is not a local property: a Hessian lower
bound on a ball does not imply convexity of the function elsewhere, and
the Yang--Mills action is known to be \emph{non-convex} globally (flat
Gauge directions and the bracket term $[X\wedge X]$ are indefinite in
sign away from the weak-field region; the Gribov horizon
$\partial\Omega$ of app228 is precisely where the Hessian degenerates).
Hence Theorem~\ref{thm:log_concave} only shows a pointwise curvature
bound on $U_{\mathrm{AF},j}^{\mathrm{corr}}$, not log-concavity.

Moreover the tail bound \eqref{eq:log_concave_tail} controls
$d(x,K)$ under a globally log-concave $\mu$. The assertion
$\nu_{j}(U_{\mathrm{AF},j}^{\mathrm{corr}})\ge 1/2$ used to trigger it is
stated without proof and is a weak form of the very concentration being
proved, so Step 11 of the app231 argument is circular.
\end{proof}

\subsection{Error 3: the determinant-to-volume ratio is unbounded}

\begin{theorem}[Error 3]
\label{thm:app231_error3}
The Gaussian lower bound (Lemma~\ref{lem:gaussian_lower_Z} of app231)
does not control the volume-to-partition-function ratio. Lemma
\texttt{lem:det\_vol\_ratio} (app231) concludes the ratio is bounded,
but its own computation
\eqref{eq:ratio_bound} shows
\begin{equation}
\label{eq:ratio_unbounded}
  \frac{\mathrm{vol}_{g_{j}}(\mathcal{Q}_{j})}{
  g_{\mathrm{YM}}^{D_{j}^{\perp}}\bigl(\det\Delta_{0,j}^{\perp}\bigr)^{-1/2}}
  \;\sim\;\exp\!\Bigl(c_{\dim}'\,N^{2}\,2^{4j}\,\bigl(j\log 2
  +\tfrac{1}{2}\log\log(2^{j})\bigr)\Bigr),
\end{equation}
which is \emph{unbounded} as $j\to\infty$ under asymptotic freedom
$g_{\mathrm{YM}}^{-1}(j)\sim\sqrt{\log(2^{j})}$. The alleged cancellation
of determinant factors in Step 4 of the concentration proof
(\texttt{thm:concentration\_main}) is therefore unsupported: the shell
contributions are bounded relative to $Z_{j}$ only if
$\mathrm{vol}/Z_{j}$ is controlled, and it is not.
\end{theorem}

\begin{proof}
From \eqref{eq:ratio_bound} and the Weyl law
$\bigl(\det\Delta_{0,j}^{\perp}\bigr)^{-1/2}\sim\bigl(c_{\mathrm{eig}}\,2^{2j}\bigr)^{-D_{j}^{\perp}/2}$,
\begin{equation*}
  \frac{\mathrm{vol}}{g_{\mathrm{YM}}^{D_{j}^{\perp}}(\det\Delta)^{-1/2}}
  \;\sim\;\Bigl(c_{\mathrm{vol}}\,g_{\mathrm{YM}}^{-1}(j)\,
  \bigl(c_{\mathrm{eig}}\,2^{2j}\bigr)^{1/2}\Bigr)^{D_{j}^{\perp}},
\end{equation*}
and $c_{\mathrm{vol}}\,g_{\mathrm{YM}}^{-1}(j)\bigl(c_{\mathrm{eig}}2^{2j}\bigr)^{1/2}
\sim \sqrt{\log(2^{j})}\,2^{j}$, whose logarithm is $j\log 2 +
\frac{1}{2}\log\log(2^{j})$. Raising to $D_{j}^{\perp}\sim c_{\dim}'N^{2}2^{4j}$
gives \eqref{eq:ratio_unbounded}. The app231 text itself concedes this
(\S\ref{ssec:app231_uniform}, ``the ratio \eqref{eq:ratio_bound} is NOT
bounded as $j\to\infty$''), so Lemma \texttt{lem:det\_vol\_ratio} is
self-contradicted and cannot be used.
\end{proof}

\subsection{Error 4: the first (k=0) shell is the kill shot}

\begin{theorem}[Error 4]
\label{thm:app231_error4}
The shell near the AF boundary is never controlled. In the notation of
app231, the $k=0$ shell
$\mathcal{S}_{j,0}=\{\|F_{A}\|_{L^{2}}\in(\eta_{\mathrm{AF}}^{\mathrm{corr}}2^{2j}/(j+1),\,
2\eta_{\mathrm{AF}}^{\mathrm{corr}}2^{2j}/(j+1)]\}$ is bounded in the raw
computation by
\begin{equation}
\label{eq:k0_huge}
  \nu_{j}(\mathcal{S}_{j,0})\;\sim\;\exp\!\bigl(c\,2^{4j}\,j\bigr),
\end{equation}
i.e.\ \emph{large}, because its volume prefactor
$\sim(c\,2^{2j}/g_{\mathrm{YM}})^{D_{j}^{\perp}}$ overwhelms the decay
$e^{-c\,2^{4j}}$. The only repair attempted (Step 11, log-concavity)
rests on the invalid Error 2. Consequently
Eq.~\eqref{eq:concentration_target} is not established.
\end{theorem}

\begin{proof}
The log of the $k=0$ ratio is, from app231 \eqref{eq:k0_log},
$\sim D_{j}^{\perp}(2j\log 2+\cdots)-c\,2^{4j}\log 2
\sim c_{\dim}'N^{2}2^{4j}\cdot2j\log 2 - c\,2^{4j}\log 2$, whose leading
term is $+c\,2^{4j}j>0$. The claimed domination by the exponential decay
is exactly what log-concavity (Error 2, invalid) was invoked to supply.
\end{proof}

\subsection{Repair: a genuine partial result with the correct scaling}

We close with a \emph{correctly normalized} new theorem that advances the
roadmap without overclaiming: it shows that the concentration
\eqref{eq:concentration_target} is \emph{equivalent} to a checkable
spectral statement on the Landau slice, and that a positive answer is
consistent with the Weyl law. This identifies the precise, single
analytic control that would close the mass gap.

\begin{theorem}[Equivalent form of the gauge-invariant concentration]
\label{thm:equivalent_concentration}
Let $\Delta_{0,j}^{\perp}$ be the gauge-fixed Laplacian on co-closed
1-forms at scale $j$, and let
$R_{j}:=\mathrm{vol}_{g_{j}}(\mathcal{Q}_{j})/\bigl(g_{\mathrm{YM}}^{D_{j}^{\perp}}
(\det\Delta_{0,j}^{\perp})^{-1/2}\bigr)$ be the volume ratio of
\eqref{eq:ratio_unbounded}. The gauge-invariant concentration
\eqref{eq:concentration_target} holds if and only if there exist
$c,\alpha>0$ independent of $j$ such that
\begin{equation}
\label{eq:equivalent_ratio}
  R_{j}\;\le\;c\,\exp\!\Bigl(\alpha\,\frac{2^{2j}}{(j+1)^{2}} -
  \frac{c_{1}\,2^{4j}\log(2^{j})}{g_{\mathrm{YM}}^{2}(j)(j+1)^{2}}\Bigr),
\end{equation}
with $c_{1}=(\eta_{\mathrm{AF}}^{\mathrm{corr}})^{2}/2$.
\end{theorem}

\begin{proof}
(Sketch.) Write $\nu_{j}(U^{\mathrm{corr},c}_{j})=\int_{U^{c}}e^{-S}\,d\mathrm{vol}/Z_{j}$.
By the dyadic shell decomposition of app231 and the action lower bound
$S_{\mathrm{YM},j}\ge 2^{2k}(\eta_{\mathrm{AF}}^{\mathrm{corr}})^{2}2^{4j}/(2g_{\mathrm{YM}}^{2}(j)(j+1)^{2})$
on $\mathcal{S}_{j,k}$, each shell contribution is bounded by
$R_{j}\cdot\exp(-c_{1}2^{2k}2^{4j}/(g_{\mathrm{YM}}^{2}(j)(j+1)^{2}))$
(using $\mathrm{vol}(\mathcal{S}_{j,k})\le\mathrm{vol}(\mathcal{Q}_{j})=R_{j}g_{\mathrm{YM}}^{D_{j}^{\perp}}(\det\Delta)^{-1/2}$).
Here the determinant factors already cancel against $Z_{j}$; the only
ratio left is exactly $R_{j}$. Summing over shells yields the bound
\eqref{eq:equivalent_ratio}. Conversely \eqref{eq:equivalent_ratio} plus
$2^{4j}/g_{\mathrm{YM}}^{2}(j)\gg 2^{2j}$ reduces to
\eqref{eq:concentration_target}. Both directions are elementary.
\end{proof}

\begin{remark}[Why this is the honest target]
\label{rem:honset_target}
Theorem~\ref{thm:equivalent_concentration} isolates the obstruction as a
single Weyl-type control on the Landau-slice Laplacian $R_{j}$. The raw
Weyl estimate gives $R_{j}\sim\exp(c\,2^{4j}j)$ (unbounded by
Error 3); a genuine mass gap requires \eqref{eq:equivalent_ratio}, i.e.\
cancellation of that $c\,2^{4j}j$ growth by the AF action. This is the
precise, checkable, single-line estimate that remains open. It is
consistent with, but not implied by, the known strong-coupling
concentration (Osterwalder--Seiler). The Clay Millennium Prize mass-gap
conjecture therefore remains \textbf{open}.
\end{remark}
