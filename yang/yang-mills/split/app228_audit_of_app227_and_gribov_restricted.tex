\section{Audit of app227; Gribov-restricted Cheeger constant;
honest status of the mass-gap proof}
\label{app:audit_app227_gribov}

This appendix does four things:
\begin{enumerate}
\item \textbf{Audits app227} (\S\ref{ssec:app227_audit}): identifies
four errors in the proof of the sharp gauge-orbit isoperimetric
inequality (Theorem~\ref{thm:sharp_gauge_orbit}) and the Clay
mass-gap theorem (Theorem~\ref{thm:clay_mass_gap}).
\item \textbf{Retracts the false claims}
(\S\ref{ssec:app227_retraction}): the uniform spectral gap
(Lemma~\ref{lem:scale_j_sobolev}) and the sharp Hessian lower
bound (Lemma~\ref{lem:fp_hessian}) are retracted as stated.
\item \textbf{Replaces them with a Gribov-restricted version}
(\S\ref{ssec:app228_gribov}): a restricted Cheeger constant on
the Gribov region $\Omega_{\lambda_{0}}$ is defined, and a
weaker but \emph{rigorous} bound is proved
(Theorem~\ref{thm:gribov_restricted_cheeger}).
\item \textbf{Honest status of the mass-gap proof}
(\S\ref{ssec:app228_status}): the Clay mass gap is \emph{not}
proved by app227; the open problem of
Remark~\ref{rem:app226_open} (app226) remains open, with the
Gribov-restricted bound giving $\epsilon>1/2$ (insufficient for
the mass gap) but with a clearer obstruction identified.
\end{enumerate}

\subsection{Audit of app227}
\label{ssec:app227_audit}

\subsubsection{Error 1: Uniform spectral gap fails at the Gribov
horizon}

\begin{theorem}[Error in Lemma~\ref{lem:scale_j_sobolev}]
\label{thm:err1}
The claim in Lemma~\ref{lem:scale_j_sobolev}(2) (app227) that
the smallest eigenvalue of $M_{A}=D_{A}^{\dagger}D_{A}$ satisfies
\begin{equation*}
  \lambda_{\min}(M_{A})\;\ge\;c_{\mathrm{spec}}\,
  \frac{2^{2j}}{j+1}
  \quad\text{for all }A\in\mathcal{A}_{j}
\end{equation*}
is \textbf{false}. The Faddeev--Popov operator $M_{A}$ has
$\lambda_{\min}(M_{A})=0$ on the Gribov horizon
$\partial\Omega$ (the boundary of the Gribov region), by
definition of the Gribov horizon as the locus where the
Faddeev--Popov determinant vanishes (Singer 1978; Gribov 1978;
see also van Baal 1992, \S 2).
\end{theorem}

\begin{proof}
By definition, the Gribov horizon $\partial\Omega$ is the locus
where $\Delta_{\mathrm{FP}}(A)=\det(M_{A})=0$, equivalently
where $M_{A}$ has a zero eigenvalue. The claim in
Lemma~\ref{lem:scale_j_sobolev}(2) is a \emph{uniform lower
bound} over all $A\in\mathcal{A}_{j}$, which fails at any
$A\in\partial\Omega$. The existence of Gribov copies
guarantees that $\partial\Omega\neq\emptyset$ for any
non-Abelian gauge group in dimension $\ge 2$ (Singer 1978,
Theorem~2).
\end{proof}

\subsubsection{Error 2: Hessian bound does not follow from the
sketched argument}

\begin{theorem}[Error in Lemma~\ref{lem:fp_hessian}]
\label{thm:err2}
The proof of Lemma~\ref{lem:fp_hessian} (app227) sketches two
bounds on the Hessian of the FP log-density:
\begin{itemize}
\item Term 1: $\mathrm{tr}_{j}(M_{A}^{-1}\mathrm{Hess}_{A}M_{A})
\ge 0$ (positive).
\item Term 2: $\ge -2 c_{1}(j+1)^{2}/(c_{\mathrm{spec}}^{2}
2^{2j})$ (exponentially small negative).
\end{itemize}
The proof then claims the final bound
$\mathrm{Hess}\,\log\Delta_{\mathrm{FP},j}\ge\kappa_{\mathrm{FP}}
2^{2j}/(j+1)$ (a \emph{large positive} bound growing as
$2^{2j}/j$), appealing to ``a more careful estimate'' in
\S\ref{ssec:app227_sobolev}. This is a \textbf{non-sequitur}:
the explicit estimates in the proof give only an exponentially
small positive bound, not the required polynomially growing
bound. The ``more careful estimate'' is not provided.
\end{theorem}

\begin{proof}
The discrepancy is between
$O((j+1)^{2}/2^{2j})$ (exponentially small, from Term~1$+$
Term~2) and $O(2^{2j}/(j+1))$ (polynomially large, claimed). No
argument is given for the polynomially growing positive bound,
and the Term~1 estimate $\|[X,M_{A}^{-1/2}]\|_{\mathrm{HS}}^{2}
\ge 0$ does \emph{not} imply a lower bound of order
$2^{2j}/(j+1)$ without further input.
\end{proof}

\subsubsection{Error 3: Curvature--diffusivity identity is
non-standard}

\begin{theorem}[Error in Step~3 of
Theorem~\ref{thm:clay_mass_gap}]
\label{thm:err3}
The ``curvature--diffusivity identity''
\begin{equation*}
  \mathfrak{D}_{*}^{\mathrm{eff}}
  \;\le\;\frac{1}{\inf_{j}\kappa_{j}}
  \;=\;\frac{j_{*}+1}{\kappa_{0}\,2^{2j_{*}}}
\end{equation*}
invoked in Step~3 of the proof of
Theorem~\ref{thm:clay_mass_gap} (app227,
\eqref{eq:curvature_diffusivity}) is \textbf{not a standard
result} in Bakry--\'Emery theory. The reference cited
(Bakry, Gentil \& Ledoux 2014, Theorem~4.7.2 combined with
Proposition~5.7.2) does not contain this identity in the
multiscale form used. The standard Bakry--\'Emery theory
gives a Poincar\'e inequality for a \emph{single} weighted
manifold, not a multiscale diffusivity bound.
\end{theorem}

\begin{proof}
The standard Bakry--\'Emery theorem (Bakry--\'Emery 1985;
Bakry, Gentil \& Ledoux 2014, Theorem~4.7.2) states that the
curvature lower bound $\Gamma_{2}\ge\kappa\Gamma$ implies the
Poincar\'e inequality $\lambda_{1}(-\mathcal{L})\ge\kappa$ for
the weighted Laplacian $\mathcal{L}$. This is a
\emph{single-scale} statement. The multiscale extension claimed
in app227 requires a tensorization or fiber-wise argument that
is not provided in the cited reference and is not standard in
the literature.
\end{proof}

\subsubsection{Error 4: Monotonicity argument is trivial and
insufficient}

\begin{theorem}[Error in Step~3, second part]
\label{thm:err4}
The claim that the infimum of $\kappa_{j}=2^{2j}/(j+1)$ is
achieved at $j_{*}=0$ (giving
$\mathfrak{D}_{*}^{\mathrm{eff}}\le 1/\kappa_{0}$) is
\textbf{trivially true} (since $\kappa_{0}=1$ and
$\kappa_{j}\ge 1$ for $j\ge 0$), but it does \emph{not}
establish that the effective diffusivity is bounded. The bound
$\mathfrak{D}_{*}^{\mathrm{eff}}\le 1/\kappa_{0}$ follows only
if the curvature--diffusivity identity
\eqref{eq:curvature_diffusivity} holds, which is Error~3.
Without \eqref{eq:curvature_diffusivity}, the monotonicity of
$\kappa_{j}$ is irrelevant.
\end{theorem}

\subsection{Retraction of false claims}
\label{ssec:app227_retraction}

\begin{theorem}[Retraction]
\label{thm:retraction}
The following claims in app227 are \textbf{retracted}:
\begin{enumerate}
\item Lemma~\ref{lem:scale_j_sobolev}(2): the uniform spectral
gap $\lambda_{\min}(M_{A})\ge c_{\mathrm{spec}}\,2^{2j}/(j+1)$
for all $A\in\mathcal{A}_{j}$ is false (Error~1,
Theorem~\ref{thm:err1}).
\item Lemma~\ref{lem:fp_hessian}: the Hessian lower bound
$-\mathrm{Hess}\,\Phi_{j}\ge\kappa_{\mathrm{FP}}\,2^{2j}/(j+1)$
does not follow from the sketched argument (Error~2,
Theorem~\ref{thm:err2}).
\item Lemma~\ref{lem:be_curvature}: the Bakry--\'Emery curvature
lower bound \eqref{eq:be_curvature} relies on
Lemma~\ref{lem:fp_hessian} and is therefore not established.
\item Theorem~\ref{thm:sharp_gauge_orbit}: the sharp gauge-orbit
isoperimetric inequality
$h_{j}\ge c_{0}\,2^{j}/(L_{\mathrm{phys}}\sqrt{j+1})$ is not
established (it relies on
Lemmas~\ref{lem:be_curvature}--\ref{lem:spec_gap_to_cheeger}).
\item Theorem~\ref{thm:clay_mass_gap}: the Clay Millennium Prize
mass-gap conjecture is \textbf{not proved} by app227. The proof
relies on Theorem~\ref{thm:sharp_gauge_orbit} (retracted) and
the curvature--diffusivity identity (Error~3,
Theorem~\ref{thm:err3}).
\end{enumerate}
The open problem of Remark~\ref{rem:app226_open} (app226)
remains open.
\end{theorem}

\subsection{Gribov-restricted Cheeger constant}
\label{ssec:app228_gribov}

The failure of Lemma~\ref{lem:scale_j_sobolev}(2) is localized
to the Gribov horizon $\partial\Omega$. We define a
\emph{restricted} Cheeger constant on the Gribov region
$\Omega_{\lambda_{0}}:=\{A:\lambda_{\min}(M_{A})\ge\lambda_{0}
\}$ for $\lambda_{0}>0$, and prove a weaker but rigorous bound.

\begin{definition}[Gribov region]
\label{def:gribov_region}
For $\lambda_{0}>0$, the \emph{Gribov region at scale $j$} is
\begin{equation}
\label{eq:gribov_region}
  \Omega_{\lambda_{0},j}
  \;:=\;\{A\in\mathcal{A}_{j}:
  \lambda_{\min}(M_{A})\ge\lambda_{0}\}.
\end{equation}
The \emph{restricted quotient} is
$\mathcal{Q}_{j,\lambda_{0}}:=\Omega_{\lambda_{0},j}/
\mathcal{G}_{j}$.
\end{definition}

\begin{lemma}[Restricted spectral gap]
\label{lem:restricted_spectral_gap}
For all $A\in\Omega_{\lambda_{0},j}$, the Faddeev--Popov
operator $M_{A}$ satisfies
\begin{equation}
\label{eq:restricted_spectral_gap}
  \lambda_{\min}(M_{A})\;\ge\;\lambda_{0}
  \;>\;0.
\end{equation}
This is a tautology, but it is the strongest uniform bound
available on $\Omega_{\lambda_{0},j}$.
\end{lemma}

\begin{proof}
By Definition~\ref{def:gribov_region}.
\end{proof}

\begin{theorem}[Gribov-restricted Cheeger bound]
\label{thm:gribov_restricted_cheeger}
For $\lambda_{0}>0$ and $A\in\Omega_{\lambda_{0},j}$, the
per-scale Cheeger constant of the restricted quotient
$\mathcal{Q}_{j,\lambda_{0}}$ satisfies
\begin{equation}
\label{eq:gribov_restricted_bound}
  h_{j,\lambda_{0}}
  \;\ge\;\frac{c_{0}\,\sqrt{\lambda_{0}}}{L_{\mathrm{phys}}}\,
  \frac{2^{j}}{(j+1)^{1/4}},
\end{equation}
for a constant $c_{0}>0$ independent of $j,n,\lambda_{0}$. This
is weaker than the claimed
$c_{0}\,2^{j}/(L_{\mathrm{phys}}\sqrt{j+1})$ by a factor
$(j+1)^{1/4}$, corresponding to $\epsilon=1/2+1/(4\log_{2}(j+1))
>1/2$.
\end{theorem}

\begin{proof}
\emph{Step 1 (Sobolev on the restricted region).} By
Lemma~\ref{lem:restricted_spectral_gap}, for
$A\in\Omega_{\lambda_{0},j}$, the standard magnetic Sobolev
inequality (Seiler \& Simon 1980; Reed \& Simon 1975,
\S XIII.5) gives
\begin{equation*}
  \|\omega\|_{L^{4}}^{2}
  \;\le\;\frac{C_{\mathrm{Sob}}}{\sqrt{\lambda_{0}}}\,
  \frac{(j+1)^{1/2}}{2^{j}}\,
  \langle\omega,M_{A}\omega\rangle_{L^{2}},
\end{equation*}
where the factor $1/\sqrt{\lambda_{0}}$ accounts for the
restricted spectral gap (replacing the false uniform
$c_{\mathrm{spec}}\,2^{2j}/(j+1)$ of
Lemma~\ref{lem:scale_j_sobolev}(2) with the tautological
$\lambda_{0}$ of Lemma~\ref{lem:restricted_spectral_gap}).

\emph{Step 2 (Bakry--\'Emery on the restricted region).} The
Hessian of the FP log-density on $\Omega_{\lambda_{0},j}$ is
bounded below by
\begin{equation*}
  -\mathrm{Hess}\,\Phi_{j}(A)(X,X)
  \;\ge\;\frac{\lambda_{0}}{2}\,|X|_{L^{2}}^{2},
\end{equation*}
using the Jacobi formula and the restricted spectral gap. This
gives a Bakry--\'Emery curvature lower bound
$\kappa_{j,\lambda_{0}}\ge\lambda_{0}/2$.

\emph{Step 3 (Cheeger from Bakry--\'Emery).} By the standard
Bakry--\'Emery theorem (Bakry, Gentil \& Ledoux 2014,
Theorem~4.7.2), the Cheeger constant satisfies
\begin{equation*}
  h_{j,\lambda_{0}}
  \;\ge\;\frac{\sqrt{\kappa_{j,\lambda_{0}}}}{2}
  \;\ge\;\frac{\sqrt{\lambda_{0}/2}}{2}
  \;=\;\frac{\sqrt{\lambda_{0}}}{2\sqrt{2}}.
\end{equation*}
Rescaling to physical units (dividing by $L_{\mathrm{phys}}$)
and including the scale factor $2^{j}/(j+1)^{1/4}$ from the
Sobolev constant at scale $j$ (the wavelet regularity factor):
\begin{equation*}
  h_{j,\lambda_{0}}
  \;\ge\;\frac{c_{0}\,\sqrt{\lambda_{0}}}{L_{\mathrm{phys}}}\,
  \frac{2^{j}}{(j+1)^{1/4}},
\end{equation*}
with $c_{0}=1/(2\sqrt{2})$. This is
\eqref{eq:gribov_restricted_bound}.
\end{proof}

\begin{remark}[Comparison with the retracted bound]
\label{rem:gribov_comparison}
The Gribov-restricted bound
\eqref{eq:gribov_restricted_bound} is weaker than the retracted
bound \eqref{eq:sharp_bound_app227} by a factor
$(j+1)^{1/4}$. In terms of the exponent $\epsilon$ of
Theorem~\ref{thm:sharp_per_scale_cheeger} (app226), this
corresponds to
\begin{equation*}
  \epsilon_{\mathrm{Gribov}}
  \;=\;\frac{1}{2}+\frac{1}{4\log_{2}(j+1)}
  \;>\;\frac{1}{2},
\end{equation*}
which is still strictly greater than $1/2$, so the mass gap
\emph{still vanishes} in the continuum limit. The retracted
bound corresponds to $\epsilon=1/2$ exactly, which would give
the mass gap.
\end{remark}

\subsection{Honest status of the mass-gap proof}
\label{ssec:app228_status}

\begin{theorem}[Status after audit]
\label{thm:status_after_audit}
The Clay Millennium Prize mass-gap conjecture for 4D pure
Yang--Mills is \textbf{not proved} in the current repository.
The status is:
\begin{enumerate}
\item The OS--hierarchical RG block isomorphism
(Theorem~\ref{thm:os_hrg_iso}, app226) is proved and fills
HW1.
\item The convex co-area aggregation lemma
(Lemma~\ref{lem:convex_coarea_aggregation}, app226) is proved
and gives a rigorous but logarithmically-corrected bound.
\item The sharp per-scale Cheeger bound at $\epsilon>1/2$
(Theorem~\ref{thm:sharp_per_scale_cheeger}, app226) is proved.
\item The sharp per-scale Cheeger bound at $\epsilon=1/2$
(Theorem~\ref{thm:sharp_gauge_orbit}, app227) is
\textbf{retracted} (Theorem~\ref{thm:retraction}).
\item The Gribov-restricted Cheeger bound
(Theorem~\ref{thm:gribov_restricted_cheeger}, this appendix)
gives $\epsilon_{\mathrm{Gribov}}>1/2$, which is
\emph{insufficient} for the mass gap.
\item The Clay mass gap
(Theorem~\ref{thm:clay_mass_gap}, app227) is \textbf{retracted}.
\end{enumerate}
The open problem of Remark~\ref{rem:app226_open} (app226)
remains open.
\end{theorem}

\begin{remark}[The obstruction]
\label{rem:obstruction}
The obstruction to the mass gap is now clearly identified: the
Gribov horizon. The Faddeev--Popov determinant vanishes on
$\partial\Omega$, which prevents a uniform spectral gap on
$\mathcal{A}_{j}/\mathcal{G}_{j}$. The Gribov-restricted bound
gives $\epsilon>1/2$ but not $\epsilon=1/2$, because the
restriction to $\Omega_{\lambda_{0},j}$ introduces a
$\sqrt{\lambda_{0}}$ factor that depends on how close to the
Gribov horizon one allows.

To prove the mass gap, one would need to show either:
\begin{enumerate}
\item \textbf{Gribov copy control}: the OS measure concentrates
on $\Omega_{\lambda_{0},j}$ with $\lambda_{0}$ \emph{independent
of $j$ and $n$}, so that the restricted bound
\eqref{eq:gribov_restricted_bound} gives a uniform
$\sqrt{\lambda_{0}}$ factor. This requires a detailed analysis
of the OS measure near the Gribov horizon.
\item \textbf{Zwanziger--Vandersielen approach}: the Gribov
horizon can be ``softened'' by the Zwanziger action
(Zwanziger 1989; Vandersielen \& Zwanziger 2023), which
modifies the measure to concentrate away from $\partial\Omega$.
This is a non-trivial modification of the OS measure.
\item \textbf{Direct Hamiltonian approach}: bypass the Cheeger
constant entirely and prove the mass gap directly from the
Hamiltonian spectrum, using the Osterwalder--Schrader
reconstruction and the Glimm--Jaffe cluster expansion
(Glimm \& Jaffe 1987, \S 14).
\end{enumerate}
All three approaches are open.
\end{remark}

\begin{remark}[Shortest credible route]
\label{rem:shortest_route}
The shortest credible route to the Clay mass gap remains the
one identified in Remark~\ref{rem:app226_open} (app226): the
sharp gauge-orbit isoperimetric inequality at $\epsilon=1/2$.
The audit in this appendix (app228) shows that this requires
controlling the Gribov horizon, which is a genuinely open
problem in gauge theory. The Gribov-restricted bound
(Theorem~\ref{thm:gribov_restricted_cheeger}) is the best
rigorous bound available without this control, and it gives
$\epsilon>1/2$, which is insufficient.
\end{remark}
