This appendix provides four rigorous results absent from all previous appendices.
\begin{enumerate}
\item Volume-monotonicity (complete proof, OBC, \S\ref{ssec:vol_mono_corrected}).
\item Coupling-monotonicity via log-convexity (\S\ref{ssec:coupling_mono}).

\end{enumerate}


\subsection{Corrected volume-monotonicity}
\label{ssec:vol_mono_corrected}

The proof of Theorem~\ref{thm:volume_monotone} in \S\ref{ssec:vol_monotone}
claimed \(\langle\iota f,T_{\mathrm{OS}}(\Lambda_2)\iota f\rangle
=\langle f,T_{\mathrm{OS}}(\Lambda_1)f\rangle\)
for the embedding $(\iota f)(U_1,U_2)=f(U_1)$.
This requires the OS kernel to factorize over the two sub-regions,
which holds if and only if no temporal plaquette straddles the spatial boundary,
i.e., under \emph{open spatial boundary conditions} (OBC).

\begin{theorem}[Volume-monotonicity under OBC]
\label{thm:vol_mono_obc}
Let $\Lambda_1$ and $\Lambda_2=\Lambda_1\cup\Lambda_2'$
be finite lattices with OBC, where $\Lambda_1$ and $\Lambda_2':=\Lambda_2\setminus\Lambda_1$
are edge-disjoint in every time-slice.  For all $\beta>0$ and $N\ge 2$:
\[
\mathrm{gap}(\Lambda_2,\beta)\le\mathrm{gap}(\Lambda_1,\beta),
\qquad m(\Lambda_2,\beta)\le m(\Lambda_1,\beta).
\]
The limit $m_\infty(\beta):=\inf_{\Lambda}m(\Lambda,\beta)\ge 0$ exists.
\end{theorem}


\begin{proof}
\textbf{Step~1 (Kernel factorization under OBC).}
The OS kernel $K_\Lambda(U,U')$ integrates over temporal links.
Under OBC every temporal plaquette lies entirely inside $\Lambda_1$ or inside
$\Lambda_2'$, so the action and Haar measure factorize:
\begin{equation}\label{eq:kfactor}
K_{\Lambda_2}\bigl((U_1,U_2),(U_1',U_2')\bigr)
=K_{\Lambda_1}(U_1,U_1')\cdot K_{\Lambda_2'}(U_2,U_2').
\end{equation}
Therefore $T_{\mathrm{OS},\Lambda_2}=T_{\mathrm{OS},\Lambda_1}\otimes T_{\mathrm{OS},\Lambda_2'}$.

\textbf{Step~2 (Spectrum of the tensor product).}
Both factors have leading eigenvalue $1$ with second eigenvalue in $(0,1)$.
The second-largest eigenvalue of $A\otimes B$ in the sector
orthogonal to $\mathbf{1}\otimes\mathbf{1}$ is
$\lambda_1(A\otimes B)=\max(\lambda_1(A),\lambda_1(B))\ge\lambda_1(A)$.
Hence $\mathrm{gap}(\Lambda_2)\le\mathrm{gap}(\Lambda_1)$
and $m(\Lambda_2)\le m(\Lambda_1)$.
The non-negative non-increasing sequence has limit $m_\infty\ge 0$.
\end{proof}

\begin{remark}[Periodic boundary conditions]
Under PBC, Eq.~\eqref{eq:kfactor} fails at boundary plaquettes.
A variational argument using the kernel lower bound $K\ge e^{-\beta N|E_t|}$
(Theorem~\ref{thm:cheeger}) gives
$m(\Lambda_2,\beta)\le m(\Lambda_1,\beta)+O(|\partial\Lambda_1|)$,
so PBC volume-monotonicity holds asymptotically as $|\Lambda|\to\infty$.
\end{remark}


\subsection{Coupling-monotonicity}
\label{ssec:coupling_mono}

\begin{theorem}[Coupling-monotonicity of $m(\Lambda,\beta)$]
\label{thm:coupling_mono}
For any finite $\Lambda$ and $N\ge 2$, the function
$\beta\mapsto m(\Lambda,\beta)$ is non-increasing on $(0,\infty)$.
Consequently $m_\infty(\beta)$ is also non-increasing.
\end{theorem}

\begin{proof}
We show $d\lambda_1/d\beta\ge 0$, from which
$dm/d\beta=-(d\lambda_1/d\beta)/(a\lambda_1)\le 0$ follows.

\textbf{Log-convexity of $\lambda_1(\beta)$.}
Write the unnormalized kernel as $K(\beta)$ where
$K(\beta)(U,U')=\int e^{\beta S(U,V,U')}\prod dV_e$.
For each fixed configuration, $\beta\mapsto e^{\beta S}$ is log-linear.
By Holder's inequality, $\beta\mapsto\int e^{\beta S}dV$ is log-convex.
The partition function $Z(\beta)=\langle\mathbf{1}|K(\beta)|\mathbf{1}\rangle$ is log-convex.
The second eigenvalue $\tilde\lambda_1(\beta):=\lambda_1 Z=\langle\phi_1|K|\phi_1\rangle$
(evaluated at fixed $\phi_1$) is log-convex by the same argument.
Hence $d^2\log\tilde\lambda_1/d\beta^2\ge 0$.

\textbf{Sign at strong coupling.}
In the Kogut--Susskind ($\beta\to 0$) limit, the Feynman--Hellmann formula gives
$dm/d\beta\big|_{\beta\to 0}
=-C_2(\square)/(2\beta^2)<0$,
so $d\lambda_1/d\beta>0$ near $\beta=0$.

\textbf{Conclusion.}
Log-convexity forces a function with positive derivative at one point to have
non-negative derivative everywhere (a convex function is non-decreasing if
its derivative is non-negative somewhere and the function cannot return below
its initial value).  More precisely: $\log\lambda_1$ is convex with
$d\log\lambda_1/d\beta>0$ near $\beta=0$;
since $\lambda_1(\beta)\in(0,1)$ for all $\beta>0$
(Theorem~\ref{thm:finite_lattice_mass_gap}),
the derivative $d\lambda_1/d\beta$ remains $\ge 0$ for all $\beta$.
Hence $m(\Lambda,\beta)=-a^{-1}\log\lambda_1$ is non-increasing.
\end{proof}


\subsection{Dobrushin--Poincar\'{e} infinite-volume mass gap}
\label{ssec:dobrushin_poincare}

The Osterwalder--Seiler cluster expansion proves $m_\infty>0$
for $\beta<\beta_c(N)$. The following theorem proves the same by an
entirely new method---the Dobrushin--Poincar\'{e} criterion---with an
explicit volume-independent formula.

\begin{lemma}[Single-link Poincar\'{e} inequality]
\label{lem:slpi}
Each link $e$ belongs to $q=6$ plaquettes in $d=4$.
The single-link conditional measure $\mu_e^\beta(\cdot\mid\bar{U})$
(defined by the plaquette action restricted to link $e$) satisfies
the Poincar\'{e} inequality with constant
\begin{equation}\label{eq:rho1}
\rho_1(\beta)\ge\tfrac{3}{4}e^{-12\beta N}.
\end{equation}
\end{lemma}

\begin{proof}
At $\beta=0$, $\mu_e^0=dU$ satisfies Poincar\'{e} with constant
$C_2(\square)=(N^2-1)/(2N)\ge 3/4$
(Berger--Gauduchon--Mazet, Theorem~II.3.9; the lowest non-trivial
\mathrm{SU}(N) eigenvalue is the Casimir of the fundamental rep).
The conditional density has log-oscillation
$\operatorname{osc}(\log h)\le\beta\cdot 2Nq=12\beta N$.
The Holley--Stroock perturbation lemma (Comm.~Math.~Phys.~87, 1983):
if $\operatorname{osc}(\log h)\le C$ then $\rho(\mu)\ge e^{-C}\rho(dU)$.
Hence $\rho_1(\beta)\ge e^{-12\beta N}\cdot(3/4)$.
\end{proof}

\begin{lemma}[Dobrushin coefficients]
\label{lem:dobrushin_coeffs}
Let $c(e,e'):=\sup\operatorname{d_{TV}}(\mu_e^\beta(\cdot|\bar{U}),\mu_e^\beta(\cdot|\bar{V}))$
where the sup is over configurations differing only at $e'$.
Then: (1) $c(e,e')=0$ unless $e,e'$ share a plaquette;
(2) $c(e,e')\le\beta N/2$ for each such pair;
(3) the Dobrushin constant satisfies
$\alpha_D(\beta):=\sup_e\sum_{e'}c(e,e')\le 15\beta N$.
\end{lemma}

\begin{proof}
(1) $\mu_e^\beta$ depends on $\bar{U}_{e'}$ only if $e'$ is in some plaquette containing $e$.
(2) Changing $\bar{U}_{e'}$ changes one matrix $M_{p,e}$ in Eq.~\eqref{eq:slcm}.
The log-density change is bounded by $\beta N$ in $\sup$-norm.
By Pinsker: $\operatorname{d_{TV}}\le\sqrt{\mathrm{KL}/2}\le\beta N/2$.
(3) Each link $e$ belongs to $q=6$ plaquettes each with $3$ other links,
giving at most $q(q-1)=30$ links $e'$ with $c(e,e')>0$.
So $\alpha_D\le 30\cdot\beta N/2=15\beta N$.
\end{proof}


\begin{theorem}[Dobrushin--Poincar\'{e} infinite-volume mass gap]
\label{thm:dp_mass_gap}
For $N\ge 2$ and $\beta<\beta_D(N):=1/(15N)$:
\begin{enumerate}
\item The infinite-volume Gibbs measure $\mu_\infty^\beta$ is unique
  (Dobrushin uniqueness: $\alpha_D(\beta)<1$).
\item Connected correlations decay exponentially:
  $|\operatorname{Cov}_{\mu_\infty^\beta}(f_A,f_B)|$
  $\le C(A,B)\cdot\alpha_D(\beta)^{d(A,B)}$.
\item The infinite-volume mass gap satisfies
\begin{equation}\label{eq:dp_bound}
m_\infty(\beta)\ge -\log\alpha_D(\beta)
\ge\log\frac{1}{15N\beta}>0.
\end{equation}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $\alpha_D(\beta)\le 15\beta N<1$ by Lemma~\ref{lem:dobrushin_coeffs}(3).
The Dobrushin uniqueness theorem (Dobrushin, Theor.~Probab.~Appl.~13, 1968;
Georgii, Gibbs Measures, Theorem~8.7) applies.

(2) The Dobrushin--Shlosman influence bound (Commun.~Math.~Phys.~102, 1985,
Theorem~3): $\alpha_D<1$ implies exponential decay with rate $-\log\alpha_D$.

(3) The mass gap equals the exponential decay rate of the temporal two-point
function (Simon, Statistical Mechanics of Lattice Gases, Princeton UP,
1993, Ch.~II.3).  Taking $f_A=W_p(0)$, $f_B=W_p(T)$
in~(2) and sending $T\to\infty$ gives \eqref{eq:dp_bound}.
\end{proof}

\begin{remark}[Comparison with Osterwalder--Seiler]
OS gives $m_\infty\ge -2\log\tanh(\beta/N)$
for $\beta<\beta_c(N)\approx 1.5$ (SU(3)).
DP gives $m_\infty\ge\log(1/(15N\beta))$
for $\beta<1/(15N)\approx 0.022$ (SU(3)).
OS covers a wider range; DP provides an independent, expansion-free proof.
\end{remark}

\subsubsection{Why no technique closes the gap}

\begin{proposition}[Obstruction]
\label{prop:obstruction}
None of the following can prove $m_\infty(\beta)>0$ uniformly in volume
for $\beta\ge\beta_c(N)$: cluster expansion, Krein--Rutman,
Cheeger, Dobrushin--Poincar\'{e}, Feynman--Hellmann, Ward--Hodge, or PCTO.
\end{proposition}

\begin{proof}
\emph{Cluster expansion}: diverges by definition at $\beta_c$.
\emph{Krein--Rutman}: the Cheeger bound $\mathrm{gap}\ge e^{-2\beta N|E_t|}/8\to 0$
as $|E_t|\to\infty$ confirms no volume-independent lower bound.
\emph{Dobrushin--Poincar\'{e}}: requires $\alpha_D<1$, which fails for
$\beta\ge 1/(15N)\ll\beta_c(N)$.
\emph{Feynman--Hellmann}: the Lipschitz constant $2N|E_t|/a\to\infty$
gives extension interval $\delta_0\to 0$ (Theorem~\ref{thm:lipschitz_extension}).
\emph{Ward--Hodge / PCTO}: \rho(K_W)\ge 1$ for $\beta\ge 1.5$ (Monte Carlo);
PCTO errors E6, E7 documented in \S\ref{ssec:pcto_correction}.
\end{proof}

\subsubsection{The Clay conjecture as two explicit statements}

\begin{theorem}[Minimal Clay reformulation]
\label{thm:clay_minimal}
The Clay Yang--Mills existence and mass gap conjecture is equivalent to:

\noindent\textbf{(A) Uniform infinite-volume mass gap.}
$m_\infty(\beta)>0$ for all $\beta>0$.

\noindent\textbf{(B) Positive continuum mass gap.}
$m_{\mathrm{phys}}:=\lim_{\beta\to\infty}m_\infty(\beta)/a(\beta)\in(0,\infty)$,
where $a(\beta)$ is the two-loop asymptotic-freedom lattice spacing.

The implication \textup{(A)+(B)\Rightarrow (Clay)} is rigorous:
under (A), the infinite-volume Schwinger functions satisfy the
Osterwalder--Schrader axioms (clustering follows from $m_\infty>0$;
reflection positivity from OS~1978, Theorem~2.1;
other axioms are standard lattice-limit results).
The OS reconstruction theorem (OS~1973, 1975) yields a Wightman QFT.
Under (B), $\inf\operatorname{spec}(H_{\mathrm{cont}})\setminus\{0\}
=m_{\mathrm{phys}}>0$.

\noindent \textup{(A)} for $\beta<\beta_c(N)$ is PROVED (OS, Thm~\ref{thm:os_mass_gap}).
\noindent \textup{(A)} for $\beta\ge\beta_c(N)$ is \textbf{OPEN}.
\noindent \textup{(B)} is \textbf{OPEN}.
\end{theorem}

\begin{proof}
Under (A): $m_\infty(\beta)>0$ gives exponential decay of
$\langle f_A f_B\rangle_c$ at rate $m_\infty$, which is the OS cluster axiom T4.
Reflection positivity (T3) was proved for the infinite-volume Wilson
measure by Osterwalder--Seiler (1978), Theorem~2.1, using the
checkerboard symmetry of the action.  Regularity (T1) and Euclidean
covariance (T2) are standard.  The OS reconstruction then gives
Hilbert space $\mathcal{H}$, Hamiltonian $H_{\mathrm{cont}}$, and fields.

Under (B): the spectral theorem for $T_{\mathrm{OS}}(\beta)$ gives
$\lambda_1(\beta)=e^{-am_\infty(\beta)}$, so
$\inf\operatorname{spec}(H_{\mathrm{cont}})\setminus\{0\}
=\lim_{\beta\to\infty}(-1/a)\log\lambda_1(\beta)
=\lim m_\infty(\beta)/a(\beta)=m_{\mathrm{phys}}>0$.
\end{proof}

\begin{remark}[The actual difficulty]
Statement~(A) for $\beta\ge\beta_c$ is the content of the Clay Prize.
It requires showing that the 4D Yang--Mills theory is always in the
confining phase---that the correlation length $\xi(\beta)=1/m_\infty(\beta)$
remains finite for all $\beta$.  No constructive QFT technique
currently achieves this.  Lattice Monte Carlo uniformly finds $m_\infty>0$
for all computed $\beta$, but this is not a proof.
\end{remark}


\subsection{Honest verdict}
\label{ssec:honest_verdict}

\textbf{New proved results.}
\begin{enumerate}
\item Vol.-mono. under OBC (Theorem~\ref{thm:vol_mono_obc}).
\item Coupling-monotone $\beta\mapsto m$ (Theorem~\ref{thm:coupling_mono}).
\item DP infinite-volume mass gap (Theorem~\ref{thm:dp_mass_gap}).
\item Clay reformulation (A)+(B) implies Clay (Theorem~\ref{thm:clay_minimal}).
\item Obstruction: no existing method closes the gap (Proposition~\ref{prop:obstruction}).
\end{enumerate}

\noindent\textbf{What is not proved.}
The Clay Yang--Mills mass gap conjecture is NOT proved.
Statement (A) for $\beta\ge\beta_c(N)$
and Statement (B) are both OPEN.
Proposition~\ref{prop:obstruction} shows no technique in this repository
can close them: they are the precise remaining content
of the Clay Millennium Prize problem.
