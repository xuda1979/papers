\subsection{E-F1: Vacuum preservation}
\label{ssec:f1}

\textbf{Error.} app210 (3) claims $\int K(U,U')\,d\mu_{\rm Haar}(U')=1$. This equals $Z_{\rm strip}(U)$, which depends on $U$.

\textbf{Correct proof.} $T_{\rm OS}^*\mu_\beta=\mu_\beta$ by construction, so $T_{\rm OS}^*\mathbf{1}=\mathbf{1}$ in $L^2(\mu_\beta)$; self-adjointness gives $T_{\rm OS}\mathbf{1}=\mathbf{1}$.

\subsection{E-F2: Dunford--Schwartz citation}
\label{ssec:f2}

\textbf{Error.} Part~II Thm~VI.8.6 is spectral, not compactness.

\textbf{Correct.} Part~I (1958) Thm~VI.8.3: integral operator with $L^2(\mu\otimes\mu)$ kernel is Hilbert--Schmidt, compact. Since $\tilde{K}\in L^\infty\subseteq L^2(\mu_\beta\otimes\mu_\beta)$, Part~I Thm~VI.8.3 applies.

\subsection{E-F4: Analyticity base case}
\label{ssec:f4}

\textbf{Error.} At $\beta=0$ the spectrum of $T_{\rm OS}(0)$ is $\{0,1\}$ (Peter--Weyl), not distinct Casimir values.

\textbf{Corrected proof.} Fix $\beta_0>0$. By Krein--Rutman, $\lambda_1(\beta_0)\in(0,1)$ is simple so $\lambda_1(\beta_0)\ne\lambda_2(\beta_0)$. By Kato Thm~VII.1.8, eigenvalues are locally analytic. The analytic function $\lambda_1-\lambda_2$ is nonzero at $\beta_0$, so its zero set is discrete. Away from zeros, $m(\Lambda,\beta)=-a^{-1}\log\lambda_1(\beta)$ is real-analytic.

\subsection{E-F6: Coupling-monotonicity, corrected proof}
\label{ssec:f6}

\textbf{Error.} app212: evaluates $\langle\phi_1|K(\beta)|\phi_1\rangle$ at fixed $\phi_1$ and calls this $\lambda_1(\beta)$. Since $\phi_1=\phi_1(\beta)$ varies, this is a Rayleigh-quotient lower bound, not $\lambda_1(\beta)$.

\begin{theorem}[Coupling-monotonicity, corrected]
\label{thm:cm_corr}
For finite $\Lambda$, $N\ge2$: $\beta\mapsto m(\Lambda,\beta)$ is non-increasing.
\end{theorem}
\begin{proof}
Feynman--Hellmann (app211 Lem~fh, correctly proved there):
\begin{equation}
\label{eq:fh_again}
  \frac{d\lambda_1}{d\beta}=\lambda_1\Bigl(\langle\phi_1|\hat{S}_W|\phi_1\rangle_{\mu_\beta}-\langle\hat{S}_W\rangle_{\mu_\beta}\Bigr).
\end{equation}
Sign$(d\lambda_1/d\beta)=$sign$(\Delta S(\beta))$ where $\Delta S:=\langle\phi_1|\hat{S}_W|\phi_1\rangle_{\mu_\beta}-\langle\hat{S}_W\rangle_{\mu_\beta}$.

\emph{Claim: $\Delta S>0$.} If $\Delta S=0$, $\phi_1$ is an eigenstate of $\hat{S}_W$ with eigenvalue $\langle\hat{S}_W\rangle_{\mu_\beta}$. By reflection positivity (OS 1978, Thm~2.1) and GKS (Seiler--Simon; Dunlop--Newman CMP~44 (1975) 223), the only such eigenstate in the positive cone is $\phi_0\equiv\mathbf{1}$ (Krein--Rutman uniqueness). Since $\phi_1\perp\phi_0$: contradiction. Hence $\Delta S>0$, $d\lambda_1/d\beta>0$, $dm/d\beta<0$.
\end{proof}

\subsection{E-F7: Corrected Dobrushin coefficients}
\label{ssec:f7}

\textbf{Error~(a): Pinsker.} app212 writes $d_{\rm TV}\le\sqrt{KL/2}\le\beta N/2$. With log-ratio $\le\beta N$, Pinsker gives $d_{\rm TV}\le\sqrt{\beta N/2}$, not $\beta N/2$.

\textbf{Error~(b): Neighbor count.} Each link is in $q=6$ plaquettes with $3$ other links: at most $3q=18$ neighbors, not $q(q-1)=30$.

\begin{lemma}[Dobrushin coefficients, corrected]
\label{lem:dob_corr}
In 4D SU$(N)$, $q=6$: $C_{ee'} \le 4N\beta n_{ee'}$ and $\max_e\sum_{e'}C_{ee'}\le72N\beta$.
\end{lemma}
\begin{proof}
Changing $\bar{U}_{e'}$: log-ratio $\le 4N\beta n_{ee'}$ (each of $n_{ee'}$ shared plaquettes changes $\operatorname{Re}\operatorname{tr}W_p$ by $\le 2N$; normalisation adds another $\le 2N\beta n_{ee'}$). By $1-e^{-x}\le x$: $C_{ee'}\le 4N\beta n_{ee'}$. Summing: $\sum_{e'}n_{ee'}\le 3q=18$ in $d=4$, giving $\sum_{e'}C_{ee'}\le 72N\beta$.
\end{proof}

\subsection{E-F8: Completed OS mass gap from Dobrushin--Shlosman}
\label{ssec:f8}

app213 Thm~dp\_to\_os stated only Step~1. Complete proof:

\begin{theorem}[Dobrushin--Shlosman $\Rightarrow$ OS mass gap]
\label{thm:dp_complete}
For $N\ge2$, $\beta<1/(72N)$: $m(\Lambda,\beta)\ge\log(1/(72N\beta))>0$ uniformly in $|\Lambda|$.
\end{theorem}

\begin{proof}
\textbf{Step~1.} Lem~\ref{lem:dob_corr}: $c:=72N\beta<1$. Dobrushin uniqueness (TPA~13 (1968) 197, Thm~1).

\textbf{Step~2.} Dobrushin--Shlosman (CMP~102 (1985) 87, Thm~3):
\begin{equation}
\label{eq:ds_decay}
  |\operatorname{Cov}_{\mu_\beta}(f_A,f_B)|\le\|f_A\|_\infty\|f_B\|_\infty|A|c^{d(A,B)},
\end{equation}
uniformly in $|\Lambda|$ and boundary conditions.

\textbf{Step~3.} Take $f_A=f_B=\mathcal{O}_p:=\operatorname{Re}\operatorname{tr}W_p$ at times $0$ and $T$: $G(T)\le N^2(72N\beta)^T$.

\textbf{Step~4.} Spectral expansion: $G(T)=\sum_{n\ge1}c_n\lambda_n^T$, $c_1=|\langle\phi_1|\mathcal{O}_p|\phi_0\rangle|^2>0$ (plaquette not in vacuum sector; Seiler 1982, Ch.~3, Prop.~3.2). If $\lambda_1>72N\beta$: $G(T)\ge c_1\lambda_1^T$ dominates the bound, contradiction. Hence $\lambda_1\le72N\beta$ and $m\ge\log(1/(72N\beta))>0$.
\end{proof}
