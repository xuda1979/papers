\subsection{Errata for app210 and app211}
\label{ssec:errata212}

\subsubsection*{E-TM-1 (app210, Lemma~\ref{lem:tm_properties}, item~3): wrong Hilbert-space claim in vacuum-preservation proof}

\noindent\textbf{What was written.}
The proof claims $\int K(U,U\prime)\,d\mu_{\mathrm{Haar}}(U\prime)=1$.

\smallskip
\noindent\textbf{Why it is wrong.}
The unnormalized kernel $K$ of \eqref{eq:tm_kernel} integrated over $U\prime$ against Haar measure gives the
temporal-strip partition function $Z_{\mathrm{strip}}(U)$, which depends on $U$ and is NOT identically $1$.

\smallskip
\noindent\textbf{Corrected proof.}
$T_{\mathrm{OS}}$ acts on $L^2(\mu_\beta)$ (Luscher 1977, eq.~(3.1)).
By construction, $\mu_\beta$ is the stationary measure:
$T_{\mathrm{OS}}^*\mu_\beta=\mu_\beta$, equivalently $T_{\mathrm{OS}}\mathbf{1}=\mathbf{1}$
in $L^2(\mu_\beta)$. No computation of $\int K\,d\mu_{\mathrm{Haar}}$ is needed.

\subsubsection*{E-TM-2 (app210, Lemma~\ref{lem:tm_properties}, item~4): wrong Dunford--Schwartz theorem}

\noindent\textbf{Corrected reference.}
Dunford--Schwartz \textit{Linear Operators}, Part~II, Theorem~VI.8.6 is a spectral
theorem, not a compactness theorem. The correct result is Part~I (1958), Theorem~VI.8.3:
an integral operator with $L^2(\mu\otimes\mu)$ kernel is Hilbert--Schmidt, hence compact, on $L^2(\mu)$.
Corrected proof: the normalised transition kernel $\tilde{K}$ is bounded (E-CH-1,
Lemma~\ref{lem:tm_corr}(5)), hence in $L^2(\mu_\beta\otimes\mu_\beta)$ on a compact measure space;
Dunford--Schwartz~I Thm.~VI.8.3 gives Hilbert--Schmidt.


\subsubsection*{E-VM-1 (app211, Theorem~\ref{thm:volume_monotone}): boundary coupling breaks the variational equality}

\noindent{}\textbf{What was written.}
The proof claims $\langle{}\iota{} f,\,{}T_{\mathrm{OS}}(\Lambda{}_2,\beta{})\iota{} f\rangle{}=\langle{} f,\,{}T_{\mathrm{OS}}(\Lambda{}_1,\beta{}) f\rangle{}$ for all $\beta{}>0$.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
Plaquettes straddling the boundary of $\Lambda{}_1$ inside $\Lambda{}_2$ contain links from both $E_{s,1}$ and $E_{s,2}$
(boundary plaquettes).  The integral over the extra links $E_{s,2}\setminus{} E_{s,1}$
does \emph{not} factor because $\iota{} f$ depends only on links in $E_{s,1}$, but the kernel of $T_{OS}(\Lambda{}_2)$
couples them through boundary plaquettes.  Hence the claimed equality fails
for boundary-straddling configurations; the variational inequality
$\lambda{}_1(\Lambda{}_2)\ge{}\lambda{}_1(\Lambda{}_1)$ is not established for general $\beta{}.$

\smallskip{}
\noindent{}\textbf{Corrected scope.}
For $\beta{}<\beta{}_c(N)$, Theorem~\ref{thm:os_mass_gap} gives $m_{\infty{}}$\ge{}$m^{SC}(\beta{})$>0 uniformly in $\Lambda{}$.  Volume-monotonicity
in \emph{weak coupling} remains an open structural question.  We restrict
Theorem~\ref{thm:volume_monotone} to $\beta{}<\beta{}_c(N)$ where the OS cluster-expansion controls boundary terms.

\subsubsection*{E-AN-1 (app211, Theorem~\ref{thm:analyticity}): wrong base-case for analytic identity principle}

\noindent{}\textbf{What was written.}
The proof uses $\beta{}=0$ as base case for the analytic identity principle,
claiming the spectrum of $T_{OS}(0)$ consists of Casimir values.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
At $\beta{}=0$ temporal links integrate independently by Haar measure;
Peter--Weyl gives $T_{OS}(0)$ = rank-1 projector onto constants,
with non-vacuum spectrum ${0}$, not Casimir values.

\smallskip{}
\noindent{}\textbf{Corrected proof.}
Fix any $\beta{}_0>0$. By Theorem~\ref{thm:finite_lattice_mass_gap}
(Krein--Rutman), $\lambda{}_0(\beta{}_0)=1$ is simple, so
$\lambda{}_1(\beta{}_0)\ne{}\lambda{}_2(\beta{}_0)$.
The real-analytic function $\beta{}\mapsto{}\lambda{}_1(\beta{})-\lambda{}_2(\beta{})$
is nonzero at $\beta{}=\beta{}_0$, hence not identically zero;
its zero set is discrete by the identity principle.
Away from this discrete set, $m(\Lambda{},\beta{})$ is real-analytic
(Kato 1976, Thm.~VII.1.8).

\subsubsection*{E-CH-1 (app211, Theorem~\ref{thm:cheeger}): unnormalised kernel inflates Cheeger conductance}

\noindent{}\textbf{What was written.}
Step~2 uses $K(U,U')\ge{}e^{-\beta{} N|E_t|}$
then applies Cheeger to claim $\mathrm{gap}\ge{}\tfrac{18}e^{-2\beta{} N|E_t|}$.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
The operator on $L^2(\mu{}_\beta{})$ uses the \emph{normalised} kernel
$\tilde{K}(U,U')=K(U,U')/K_0(U)$ where
$K_0(U)=\int{} K(U,U')\,{}d\mu{}_{\mathrm{Haar}}(U')$.
Since $|\operatorname{Re}\operatorname{tr}W_p|\le{} N$,
$K_0(U)\le{}e^{+\beta{} N|E_t|}$, so
$\tilde{K}(U,U')\ge{}e^{-2\beta{} N|E_t|}$.
Cheeger gives $\Phi{}\ge{}\tfrac{12}e^{-2\beta{} N|E_t|}$,
hence $\mathrm{gap}\ge{}\tfrac{18}e^{-4\beta{} N|E_t|}$.

\smallskip{}
\noindent{}\textbf{Corrected statement.}
\begin{equation*}
  \mathrm{gap}(\Lambda{},\beta{}) \;{}\ge{}\;{} \tfrac{18}\,{}e^{-4\beta{} N|E_t|}.
\end{equation*}
Exponent changes from $-2\beta{} N|E_t|$ to $-4\beta{} N|E_t|$.
Qualitative content unchanged: bound decays to zero as $|\Lambda{}|\to{}\infty{}$.

\subsection{Corrected transfer-matrix properties}
\label{ssec:new_tm212}

\begin{lemma}[Transfer-matrix properties, corrected]
\label{lem:tm_corr}
Let $\Lambda{}$ be a finite lattice, $N\ge{}2$, $\beta{}>0$.
The OS transfer operator $T_{\mathrm{OS}}(\Lambda{},\beta{})$ satisfies:
\begin{enumerate}
\item{} \textup{(Self-adjointness)} $T_{\mathrm{OS}}$
  is self-adjoint on $L^2(\mu{}_\beta{})$.
  \textup{[L\"uscher 1977, Prop.~2.1]}

\item{} \textup{(Positivity)} $0\le{} T_{\mathrm{OS}}\le{} I$.
  \textup{[Osterwalder--Schrader 1973, reflection positivity]}

\item{} \textup{(Vacuum preservation)} $T_{\mathrm{OS}}\mathbf{1}=\mathbf{1}$ in $L^2(\mu{}_\beta{})$.
\begin{proof}[Proof]
$\mu{}_\beta{}$ is the stationary measure of $T_{\mathrm{OS}}$,
i.e., $T_{\mathrm{OS}}^*\mu{}_\beta{}=\mu{}_\beta{}$.
Equivalently, $T_{\mathrm{OS}}\mathbf{1}=\mathbf{1}$ in $L^2(\mu{}_\beta{})$.
\end{proof}

\item{} \textup{(Hilbert--Schmidt)} $T_{\mathrm{OS}}$ is compact.
\begin{proof}[Proof]
By item~(5), $\tilde{K}\in{} L^{\infty{}}(\mu{}_\beta{}\otimes{}\mu{}_\beta{})$.
On a compact probability space, $L^{\infty{}}\subset{} L^2$,
so $\tilde{K}\in{} L^2(\mu{}_\beta{}\otimes{}\mu{}_\beta{})$.
Dunford--Schwartz~I (1958), Thm.~VI.8.3 then gives Hilbert--Schmidt.
\end{proof}

\item{} \textup{(Kernel lower bound)} $\tilde{K}(U,U')\ge{} e^{-2\beta{} N|E_t|}>0$.
\begin{proof}[Proof]
$\operatorname{Re}\operatorname{tr}W_p\ge{}-N$ gives $K(U,U')\ge{}e^{-\beta{} N|E_t|}$.
$\operatorname{Re}\operatorname{tr}W_p\le{}+N$ gives $K_0(U)\le{}e^{+\beta{} N|E_t|}$.
Dividing: $\tilde{K}=K/K_0\ge{} e^{-2\beta{} N|E_t|}>0$.
\end{proof}

\end{enumerate}
\end{lemma}

\subsection{Four new proved theorems}
\label{ssec:new_thms212}

\begin{theorem}[Single-site spectral gap]
\label{thm:single_site_gap}
For any link $e$, $N\ge{}2$, $\beta{}>0$,
and fixed external configuration $(V_p)_{p\ni{} e}$, let
\begin{equation*}
  \nu{}_e(dU) = Z_e^{-1}\exp{}\Bigl{}(\beta{}\sum{}\limits{}_{p\ni{} e}\operatorname{Re}\operatorname{tr}(UV_p)\Bigr{})d\mu{}_{\mathrm{Haar}}(U).
\end{equation*}
Then $\mathrm{gap}(\nu{}_e)>0$.
\end{theorem}

\begin{proof}
The density of $\nu{}_e$ with respect to Haar measure is smooth and
strictly positive on the compact Lie group $\mathrm{SU}(N)$.
The Laplace--Beltrami operator $\mathcal{L}_e$ generating the
overdamped Langevin dynamics for $\nu{}_e$ is a second-order elliptic
operator on the compact, connected Riemannian manifold $\mathrm{SU}(N)$.
By elliptic regularity and the Hopf maximum principle, $e^{t\mathcal{L}_e}$
is strictly positivity-improving for $t>0$.
By Krein--Rutman, the leading eigenvalue $0$ is simple and separated from
the rest of the spectrum, so $\mathrm{gap}(\nu{}_e)=-\lambda{}_1(\mathcal{L}_e)>0$.
\end{proof}

\begin{theorem}[Bakry--\Emery{} Poincar\Emery{} inequality for single-site measure]
\label{thm:bakry_emery}
With notation as in Theorem~\ref{thm:single_site_gap},
let $z_{\mathrm{pl}}$ be the number of plaquettes containing link $e$. If
\begin{equation*}
  \beta{} < \beta{}_{\mathrm{BE}}(N) := \frac{14z_{\mathrm{pl}},}
\end{equation*}
then
\begin{equation}
\label{eq:be_bound}
  \mathrm{gap}(\nu{}_e) \ge{} \frac{N4} - z_{\mathrm{pl}}\,{}\beta{} N > 0.
\end{equation}
\end{theorem}

\begin{proof}
Apply the Bakry--\Emery{} criterion: if
$\mathrm{Ric}_g - \nabla{}{}^2\log{}(d\nu{}_e/d\mathrm{vol}) \ge{} \kappa{}\,{}g$
pointwise, then $\mathrm{gap}(\nu{}_e)\ge{}\kappa{}$.\begin{description}
\item{}[Step~1 (Ricci curvature).] $\mathrm{SU}(N)$ with bi-invariant metric has
$\mathrm{Ric}_g = (N/4)\,{}g$
(Milnor, Advances in Math. 21, 1976, Thm.~2.2).

\item{}[Step~2 (Hessian of potential).] The potential is $V(U)=\beta{}\sum{}_{p\ni{} e}\operatorname{Re}\operatorname{tr}(UV_p)$.
Each summand has $||\nabla{}{}^2\operatorname{Re}\operatorname{tr}(UV_p)||_{op}\le{} N$
since all eigenvalues of $UV_p$ lie on the unit circle.

\item{}[Step~3 (Conclusion).] 
\begin{equation*}
  \mathrm{Ric}_g - \nabla{}{}^2 V
  \ge{} \frac{N4}g - z_{\mathrm{pl}}\,{}\beta{} N\,{}g
  = \Bigl{}(\frac{N4} - z_{\mathrm{pl}}\,{}\beta{} N\Bigr{})g.
\end{equation*}
This is positive when $\beta{}<1/(4z_{\mathrm{pl}})$, giving~\ref{eq:be_bound}.

\end{description}
\end{proof}

\begin{remark}
In four dimensions $z_{\mathrm{pl}}=6$ (each link borders 6 plaquettes), so
$\beta{}_{\mathrm{BE}}(N)=1/24$ for all $N\ge{}2$.
The bound~\ref{eq:be_bound} gives an explicit lower bound in the deep strong-coupling
regime $\beta{}<1/24$, supplementing the Osterwalder--Seiler cluster expansion.
\end{remark}
