\subsection{Errata for app210 and app211}
\label{ssec:errata212}

\subsubsection*{E-TM-1 (app210, Lemma~\ref{lem:tm_properties}, item~3): wrong vacuum-preservation proof}

\noindent{}\textbf{What was written.}
The proof claims $\int{} K(U,U\prime{})\,{}d\mu{}_{\mathrm{Haar}}(U\prime{})=1$.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
Integrating the unnormalised kernel $K$ over $U\prime{}$ against Haar measure gives
the temporal-strip partition function $Z_{\mathrm{strip}}(U)$, which depends on $U$ and is NOT identically 1.

\smallskip{}
\noindent{}\textbf{Corrected proof.}
$T_{\mathrm{OS}}$ acts on $L^2(\mu{}_\beta{})$ (L\'uscher 1977, eq.~(3.1)).
By construction, $T_{\mathrm{OS}}^*\mu{}_\beta{}=\mu{}_\beta{}$, equivalently $T_{\mathrm{OS}}\mathbf{1}=\mathbf{1}$ in $L^2(\mu{}_\beta{})$. No Haar integration is needed.

\subsubsection*{E-TM-2 (app210, Lemma~\ref{lem:tm_properties}, item~4): wrong Dunford--Schwartz theorem}

\noindent{}\textbf{Corrected reference.}
Dunford--Schwartz \textit{Linear Operators}, Part~II, Theorem~VI.8.6 is a spectral
theorem, not a compactness theorem. The correct result is Part~I (1958), Theorem~VI.8.3:
an integral operator with $L^2(\mu{}\otimes{}\mu{})$ kernel is Hilbert--Schmidt, hence compact.
Corrected proof: $\tilde{K}$ is bounded (Lemma~\ref{lem:tm_corr}(5)),
hence in $L^2(\mu{}_\beta{}\otimes{}\mu{}_\beta{})$ on a compact probability space; Dunford--Schwartz~I, Thm.~VI.8.3 applies.

\subsubsection*{E-VM-1 (app211, Theorem~\ref{thm:volume_monotone}): boundary coupling breaks the variational equality}

\noindent{}\textbf{What was written.}
The proof claims $\langle{}\iota{} f, T_{\mathrm{OS}}(\Lambda{}_2,\beta{})\iota{} f\rangle{}=\langle{} f, T_{\mathrm{OS}}(\Lambda{}_1,\beta{}) f\rangle{}$ for all $\beta{}>0$.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
Plaquettes straddling the boundary of $\Lambda{}_1$ inside $\Lambda{}_2$ contain links from
both $E_{s,1}$ and $E_{s,2}$.
The integral over extra spatial links $E_{s,2}\setminus{} E_{s,1}$ does \emph{not} factor
because the kernel of $T_{\mathrm{OS}}(\Lambda{}_2)$ couples them through boundary plaquettes.
Hence the claimed equality fails; $\lambda{}_1(\Lambda{}_2)\ge{}\lambda{}_1(\Lambda{}_1)$ is not established for general $\beta{}$.

\smallskip{}
\noindent{}\textbf{Corrected scope.}
For $\beta{}<\beta{}_c(N)$, Theorem~\ref{thm:os_mass_gap} gives $m_{\infty{}}(\beta{})\ge{} m^{SC}(\beta{})>0$ uniformly in $\Lambda{}$.
Volume-monotonicity at weak coupling is an open question.
Theorem~\ref{thm:volume_monotone} is restricted to $\beta{}<\beta{}_c(N)$ where the OS cluster expansion controls boundary terms.

\subsubsection*{E-AN-1 (app211, Theorem~\ref{thm:analyticity}): wrong base-case for analytic identity principle}

\noindent{}\textbf{What was written.}
The proof uses $\beta{}=0$ as base case: claims the spectrum of $T_{OS}(0)$ consists of Casimir values.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
At $\beta{}=0$ temporal links integrate independently by Haar measure; Peter--Weyl gives
$T_{OS}(0)$ = rank-1 projector onto constants, with non-vacuum spectrum $\{{}0\}{}$, not Casimir values.

\smallskip{}
\noindent{}\textbf{Corrected proof.}
Fix $\beta{}_0>0$. By Krein--Rutman (Theorem~\ref{thm:finite_lattice_mass_gap}), $\lambda{}_0(\beta{}_0)=1$ is simple, so $\lambda{}_1(\beta{}_0)\ne{}\lambda{}_2(\beta{}_0)$.
The analytic function $\beta{}\mapsto{}\lambda{}_1(\beta{})-\lambda{}_2(\beta{})$ is nonzero at $\beta{}=\beta{}_0$, hence not identically zero;
its zero set is discrete. Away from this set, $m(\Lambda{},\beta{})$ is real-analytic (Kato 1976, Thm.~VII.1.8).

\subsubsection*{E-CH-1 (app211, Theorem~\ref{thm:cheeger}): unnormalised kernel inflates Cheeger conductance}

\noindent{}\textbf{What was written.}
Step~2 lower-bounds the unnormalised kernel $K(U,U')\ge{}e^{-\beta{} N|E_t|}$,
then applies Cheeger to claim $\mathrm{gap}\ge{}\tfrac{1}{8}e^{-2\beta{} N|E_t|}$.

\smallskip{}
\noindent{}\textbf{Why it is wrong.}
The transfer matrix on $L^2(\mu{}_\beta{})$ uses the \emph{normalised} kernel $\tilde{K}=K/K_0$ where
$K_0(U)=\int{} K(U,U')\,{}d\mu{}_{\mathrm{Haar}}(U')$.
Since $|\operatorname{Re}\operatorname{tr}W_p|\le{} N$,
$K_0(U)\le{}e^{+\beta{} N|E_t|}$, so $\tilde{K}(U,U')\ge{}e^{-2\beta{} N|E_t|}$.
Cheeger gives $\Phi{}\ge{}\tfrac{1}{2}e^{-2\beta{} N|E_t|}$, hence $\mathrm{gap}\ge{}\tfrac{1}{8}e^{-4\beta{} N|E_t|}$.

\smallskip{}
\noindent{}\textbf{Corrected statement.}
\begin{equation*}
  \mathrm{gap}(\Lambda{},\beta{}) \;{}\ge{}\;{} \tfrac{1}{8}\,{}e^{-4\beta{} N|E_t|}.
\end{equation*}
Exponent changes from $-2\beta{} N|E_t|$ to $-4\beta{} N|E_t|$.
The qualitative conclusion (bound decays to zero as $|\Lambda{}|\to{}\infty{}$) is unchanged.

\subsection{Corrected transfer-matrix properties}
\label{ssec:new_tm212}

\begin{lemma}[Transfer-matrix properties, corrected]
\label{lem:tm_corr}
Let $\Lambda{}$ be a finite lattice, $N\ge{}2$, $\beta{}>0$.
The OS transfer operator $T_{\mathrm{OS}}(\Lambda{},\beta{})$ satisfies:
\begin{enumerate}
\item{} \textup{(Self-adjointness)} $T_{\mathrm{OS}}$ is self-adjoint on $L^2(\mu{}_\beta{})$. \textup{[L\'uscher 1977, Prop.~2.1]}

\item{} \textup{(Positivity)} $0\le{} T_{\mathrm{OS}}\le{} I$. \textup{[Osterwalder--Schrader 1973, reflection positivity]}

\item{} \textup{(Vacuum preservation)} $T_{\mathrm{OS}}\mathbf{1}=\mathbf{1}$ in $L^2(\mu{}_\beta{})$.
\begin{proof}[Proof]
$\mu{}_\beta{}$ is the stationary measure: $T_{\mathrm{OS}}^*\mu{}_\beta{}=\mu{}_\beta{}$,
equivalently $T_{\mathrm{OS}}\mathbf{1}=\mathbf{1}$ in $L^2(\mu{}_\beta{})$. No Haar integration needed.
\end{proof}

\item{} \textup{(Hilbert--Schmidt)} $T_{\mathrm{OS}}$ is Hilbert--Schmidt, hence compact.
\begin{proof}[Proof]
By item~(5), $\tilde{K}\in{} L^{\infty{}}(\mu{}_\beta{}\otimes{}\mu{}_\beta{})$. Since $(\mathrm{SU}(N)^{|E_s|},\mu{}_\beta{})$ is compact,
$L^{\infty{}}\subset{} L^2$, so $\tilde{K}\in{} L^2(\mu{}_\beta{}\otimes{}\mu{}_\beta{})$. Dunford--Schwartz~I, Thm.~VI.8.3 then gives Hilbert--Schmidt.
\end{proof}

\item{} \textup{(Kernel lower bound)} $\tilde{K}(U,U')\ge{} e^{-2\beta{} N|E_t|}>0$ for all $U,U'$.
\begin{proof}[Proof]
$\operatorname{Re}\operatorname{tr}W_p\ge{}-N$ gives $K(U,U')\ge{}e^{-\beta{} N|E_t|}$;
$\operatorname{Re}\operatorname{tr}W_p\le{}+N$ gives $K_0(U)\le{}e^{+\beta{} N|E_t|}$. Dividing: $\tilde{K}=K/K_0\ge{} e^{-2\beta{} N|E_t|}>0$.
\end{proof}

\end{enumerate}
\end{lemma}

\subsection{Four new proved theorems}
\label{ssec:new_thms212}

\begin{theorem}[Single-site spectral gap]
\label{thm:single_site_gap}
For any link $e$, $N\ge{}2$, $\beta{}>0$, and fixed external configuration $(V_p)_{p\ni{} e}$, let
\begin{equation*}
  \nu{}_e(dU) = Z_e^{-1}\exp{}\Bigl(\beta{}\sum\limits_{p\ni{} e}\operatorname{Re}\operatorname{tr}(UV_p)\Bigr)d\mu{}_{\mathrm{Haar}}(U).
\end{equation*}
Then $\mathrm{gap}(\nu{}_e)>0$.
\end{theorem}

\begin{proof}
The density of $\nu{}_e$ with respect to Haar measure is smooth and strictly positive
on the compact Lie group $\mathrm{SU}(N)$.
The Laplace--Beltrami operator $\mathcal{L}_e$ generating the overdamped Langevin
dynamics for $\nu{}_e$ is a second-order elliptic operator on the compact,
connected Riemannian manifold $\mathrm{SU}(N)$.
By elliptic regularity and the Hopf maximum principle, $e^{t\mathcal{L}_e}$ is strictly
positivity-improving for all $t>0$.
By Krein--Rutman, the leading eigenvalue $0$ is simple and separated from the rest,
so $\mathrm{gap}(\nu{}_e)=-\lambda{}_1(\mathcal{L}_e)>0$.
\end{proof}

\begin{theorem}[Bakry--\'Emery Poincar\'e inequality for the single-site measure]
\label{thm:bakry_emery}
With notation as in Theorem~\ref{thm:single_site_gap},
let $z_{\mathrm{pl}}$ be the number of plaquettes containing link $e$. If
\begin{equation*}
  \beta{} < \beta{}_{\mathrm{BE}}(N) := \frac{1}{4z_{\mathrm{pl}}},
\end{equation*}
then
\begin{equation}
\label{eq:be_bound}
  \mathrm{gap}(\nu{}_e) \ge{} \frac{N}{4} - z_{\mathrm{pl}}\,{}\beta{} N > 0.
\end{equation}
\end{theorem}

\begin{proof}
Apply the Bakry--\'Emery criterion: if
$\mathrm{Ric}_g - \nabla{}{}^2\log{}(d\nu{}_e/d\mathrm{vol}) \ge{} \kappa{}\,{}g$ pointwise,
then $\mathrm{gap}(\nu{}_e)\ge{}\kappa{}$.
\begin{description}
\item[Step~1 (Ricci curvature).] $\mathrm{SU}(N)$ with bi-invariant metric has $\mathrm{Ric}_g = \frac{N}{4}\,{}g$
(Milnor, Advances in Math. 21, 1976, Thm.~2.2).

\item[Step~2 (Hessian of potential).] The potential is $V(U)=\beta{}\sum{}_{p\ni{} e}\operatorname{Re}\operatorname{tr}(UV_p)$.
Each summand satisfies $\|{}\|{}\nabla{}{}^2\operatorname{Re}\operatorname{tr}(UV_p)\|{}\|{}_{op}\le{} N$,
since eigenvalues of $UV_p$ lie on the unit circle.

\item[Step~3 (Conclusion).]
\begin{equation*}
  \mathrm{Ric}_g - \nabla{}{}^2 V
  \ge{} \frac{N}{4}g - z_{\mathrm{pl}}\,{}\beta{} N\,{}g
  = \Bigl(\frac{N}{4} - z_{\mathrm{pl}}\,{}\beta{} N\Bigr)g.
\end{equation*}
This is positive when $\beta{}<\frac{1}{4z_{\mathrm{pl}}}$, giving~\eqref{eq:be_bound}.

\end{description}
\end{proof}

\begin{remark}
In four dimensions $z_{\mathrm{pl}}=6$, so $\beta{}_{\mathrm{BE}}(N)=1/24$ for all $N\ge{}2$.
The bound~\eqref{eq:be_bound} gives an explicit lower bound for $\beta{}<1/24$,
supplementing the Osterwalder--Seiler cluster expansion.
\end{remark}

\begin{theorem}[Corrected Cheeger bound]
\label{thm:cheeger_corr}
For any finite $\Lambda{}$, $\beta{}>0$, $N\ge{}2$:
\begin{equation}
\label{eq:cheeger_corr}
  \mathrm{gap}(\Lambda{},\beta{}) \;{}\ge{}\;{} \tfrac{1}{8}\,{}e^{-4\beta{} N|E_t|}.
\end{equation}
\end{theorem}

\begin{proof}
\begin{description}
\item[Step~1 (Cheeger inequality).]
$T_{\mathrm{OS}}$ is reversible w.r.t. $\mu{}_\beta{}$ (Lemma~\ref{lem:tm_corr}(1)).
By Lawler--Sokal (Trans. AMS 309, 1988, Thm.~2.1): $\mathrm{gap}(T_{\mathrm{OS}}) \ge \tfrac{1}{2}\Phi^2$.

\item[Step~2 (Kernel lower bound).]
By Lemma~\ref{lem:tm_corr}(5): $\tilde{K}(U,U')\ge{} e^{-2\beta{} N|E_t|}>0$.

\item[Step~3 (Conductance).]
For measurable $S$ with $\mu{}_\beta{}(S)\le{}\tfrac{1}{2}$:
\begin{equation*}
  \Phi{}(S) = \frac{\int{}_S\int{}_{S^c}\tilde{K}\,{}d\mu{}_\beta{}\,{}d\mu{}_\beta{}}{\mu{}_\beta{}(S)}
  \ge{} e^{-2\beta{} N|E_t|}\,{}\mu{}_\beta{}(S^c)
  \ge{} \tfrac{1}{2}e^{-2\beta{} N|E_t|}.
\end{equation*}
Hence $\Phi{}\ge{}\tfrac{1}{2}e^{-2\beta{} N|E_t|}$ and $\mathrm{gap}\ge{}\tfrac{1}{8}e^{-4\beta{} N|E_t|}$.

\end{description}
\end{proof}

\begin{theorem}[Clay equivalence via two-loop RG]
\label{thm:clay_equiv}
Define the RG-rescaled mass
\begin{equation}
\label{eq:xi_def}
  \Xi{}(\beta{}) := m_{\infty{}}(\beta{})\,{}\exp{}\Bigl(\frac{\beta{}}{2Nb_0'}\Bigr)\,{}\Bigl(\frac{2Nb_0'}{\beta{}}\Bigr)^{b_1'/(2b_0'^2)},
\end{equation}
where $b_0'=(11N)/(48\pi{}^2)$ and $b_1'=(34N^2)/(3(16\pi{}^2)^2)$ are the first two
universal Yang--Mills $\beta{}$-function coefficients (Gross--Wilczek; Politzer 1973),
and $m_{\infty{}}(\beta{})=\lim{}_{\Lambda{}\nearrow{}\mathbb{Z}^4}m(\Lambda{},\beta{})$.

The Clay mass-gap problem for $\mathrm{SU}(N)$ is equivalent to:
\begin{equation}
\label{eq:clay_equiv}
  \exists{}\,{}\Xi{}_*>0 \quad{}\text{such that}\quad{}\lim{}_{\beta{}\to{}\infty{}}\Xi{}(\beta{})=\Xi{}_*\in{}(0,\infty{}).
\end{equation}
When \eqref{eq:clay_equiv} holds, the physical mass is $m_{\mathrm{phys}}=\Lambda{}_L\,{}\Xi{}_*$.
\end{theorem}

\begin{proof}
The two-loop lattice RG gives (Hasenfratz--Hasenfratz, Phys. Lett. B 93, 1980):
\begin{equation}
\label{eq:rg_a}
  a(\beta{})\Lambda{}_L
  = \Bigl(\frac{2Nb_0'}{\beta{}}\Bigr)^{-b_1'/(2b_0'^2)}
    \exp{}\Bigl(-\frac{\beta{}}{2Nb_0'}\Bigr)(1+O(1/\beta{})).
\end{equation}
The physical mass in lattice units is $m_{\infty{}}(\beta{})$, so in continuum units:
\begin{align*}
  m_{\mathrm{phys}}(\beta{})
  &= \frac{m_{\infty{}}(\beta{})}{a(\beta{})}
  = m_{\infty{}}(\beta{})\,{}\Lambda{}_L\,{}\Bigl(\frac{2Nb_0'}{\beta{}}\Bigr)^{b_1'/(2b_0'^2)}
    e^{\beta{}/(2Nb_0')}(1+O(1/\beta{}))
  = \Lambda{}_L\,{}\Xi{}(\beta{})(1+O(1/\beta{})).
\end{align*}
A positive continuum mass gap means $m_{\mathrm{phys}}(\beta{})$ has a finite positive limit as $\beta{}\to{}\infty{}$.
Since $\Lambda{}_L>0$ is fixed, this is exactly~\eqref{eq:clay_equiv}.
\begin{description}
\item[Proved lower bound.] For $\beta{}<\beta{}_c(N)$,
Theorem~\ref{thm:os_mass_gap} gives $m_{\infty{}}(\beta{})\ge{} m^{SC}(\beta{})>0$, so $\Xi{}(\beta{})>0$.
\item[Open.] $\Xi{}(\beta{})$ remaining bounded away from 0 and $\infty{}$ as $\beta{}\to{}\infty{}$
is hypothesis~(H') = the Clay problem.
\end{description}
\end{proof}

\medskip{}
\noindent{}\textbf{Conclusion.}
The proof chain establishes: (i)~positive spectral gap on every finite lattice for all $\beta{}$ and $N$;
(ii)~strictly positive infinite-volume mass gap for $\beta{}<\beta{}_c(N)$ via Osterwalder--Seiler;
(iii)~new single-site, Bakry--\'Emery, and corrected Cheeger estimates with explicit constants;
and (iv)~an RG equivalence identifying the Clay problem as convergence of $\Xi{}(\beta{})$.
The remaining open problem is hypothesis~(H'): that $\Xi{}(\beta{})$ has a finite positive limit as $\beta{}\to{}\infty{}$.
\textbf{No claim is made that the Clay Millennium Prize Problem is resolved.}
