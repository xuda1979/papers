\subsection{Corrected Riemannian Hessian bound}
\label{ssec:hess_correction}

Theorem~\ref{thm:bakry_emery} in \S\ref{app:corrections_new} claims
$\|\nabla^2\operatorname{Re}\operatorname{tr}(UV_p)\|_{op}\le N$
in Step~2.  This overestimates the Riemannian Hessian operator norm by factor~$N$.

\begin{lemma}[Sharp Riemannian Hessian bound]
\label{lem:hess_sharp}
Equip $\mathrm{SU}(N)$ with bi-invariant metric $g(X,Y)=-2\operatorname{tr}(XY)$.
For all $M,U\in\mathrm{SU}(N)$ and $X\in\mathfrak{su}(N)$:
\begin{equation}
\label{eq:hess_sharp}
  \bigl|\mathrm{Hess}_g\bigl(\operatorname{Re}\operatorname{tr}(\,\cdot\,M)\bigr)(U)(X,X)\bigr|
  \;\le\;\tfrac{1}{2}\,g(X,X).
\end{equation}
The bound is tight.
\end{lemma}

\begin{proof}
Write $X=iH$ with $H$ Hermitian traceless.
The bi-invariant Levi-Civita connection satisfies $\nabla^{LC}_X X=\tfrac{1}{2}[X,X]=0$,
so the Riemannian Hessian equals the geodesic second derivative:
\begin{equation}
\label{eq:hess_formula}
  \mathrm{Hess}_g\bigl(\operatorname{Re}\operatorname{tr}(\,\cdot\,M)\bigr)(U)(X,X)
  = \frac{d^2}{dt^2}\Bigl|_{t=0}\operatorname{Re}\operatorname{tr}(e^{tX}UM)
  = \operatorname{Re}\operatorname{tr}(X^2 UM).
\end{equation}
Since $X^2=-H^2$, $\|X^2\|_{\mathrm{tr}}=\operatorname{tr}(H^2)$.
Trace-norm/operator-norm duality and $\|UM\|_{op}=1$:
$|\operatorname{Re}\operatorname{tr}(X^2 UM)|\le\operatorname{tr}(H^2)$.
The metric: $g(X,X)=-2\operatorname{tr}(X^2)=2\operatorname{tr}(H^2)$,
so $\operatorname{tr}(H^2)=\tfrac{1}{2}g(X,X)$, proving~\eqref{eq:hess_sharp}.

\emph{Tightness:} Take $N\ge 2$,
$X=i\,\mathrm{diag}(1/\sqrt{2},-1/\sqrt{2},0,\ldots)$,
$U=I$, $M=\mathrm{diag}(-1,-1,1,\ldots)$; equality holds.
\end{proof}

\begin{theorem}[Corrected Bakry--Emery single-site Poincare]
\label{thm:bakry_emery_corr}
In $d=4$, for $\beta < N/12$,
every single-site conditional $\nu_e$ satisfies
\begin{equation}
\label{eq:be_corr}
  \mathrm{gap}(\nu_e) \;\ge\; \kappa_0(\beta) := \frac{N}{4} - 3\beta > 0.
\end{equation}
Threshold $N/12$ exceeds previous $1/24$ by factor~$N$. Supersedes Theorem~\ref{thm:bakry_emery}.
\end{theorem}

\begin{proof}
Bakry--Emery criterion (Bakry--Gentil--Ledoux, Springer~2014, Cor.~1.3):
if $\mathrm{Ric}_g-\mathrm{Hess}_g(V_e)\ge\kappa g$ pointwise, then $\mathrm{gap}(\nu_e)\ge\kappa$.

\textbf{Step~1 (Ricci curvature).}
$\mathrm{SU}(N)$ with $g=-2\operatorname{tr}$ has $\mathrm{Ric}_g=\tfrac{N}{4}g$
(Milnor, Adv.\ Math.\ 21 (1976) Thm.~2.2).

\textbf{Step~2 (Hessian bound).}
$V_e=\beta\sum_{p\ni e}\operatorname{Re}\operatorname{tr}(UM_p)$.
By Lemma~\ref{lem:hess_sharp}, each of the $z_{\mathrm{pl}}=6$ summands satisfies
$\mathrm{Hess}_g(\operatorname{Re}\operatorname{tr}(\cdot M_p))(X,X)\le\tfrac{1}{2}g(X,X)$.
Summing: $\mathrm{Hess}_g(V_e)(X,X)\le 3\beta g(X,X)$.

\textbf{Step~3.}
$\mathrm{Ric}_g-\mathrm{Hess}_g(V_e)\ge(\tfrac{N}{4}-3\beta)g=\kappa_0(\beta)g>0$ for $\beta<N/12$.
\end{proof}

\begin{remark}
The incorrect Step~2 in Theorem~\ref{thm:bakry_emery} bounded the matrix operator
norm $\|UM\|_{op}=1$ times character bound $N$.
The correct Riemannian Hessian norm is $\tfrac{1}{2}$, smaller by factor~$N$.
\end{remark}

\subsection{Volume-uniform Poincar\'{e} via Dobrushin--Poincar\'{e}}
\label{ssec:dobrushin_poincare}

\begin{lemma}[Dobrushin interdependence coefficients for 4D $\mathrm{SU}(N)$]
\label{lem:dobrushin_coeff}
Let $C_{ee'}:=\sup d_{\mathrm{TV}}(\nu_e(\cdot|\sigma),\nu_e(\cdot|\tilde\sigma))$,
sup over configurations differing only at $e'$.
Let $n_{ee'}$ be the number of plaquettes containing both $e$ and $e'$.
\begin{enumerate}
\item $C_{ee'}=0$ if $e,e'$ share no plaquette.
\item $C_{ee'}\le 4N\beta n_{ee'}$.
\item In $d=4$: $\max_e\sum_{e'}C_{ee'}\le 72N\beta$.
\end{enumerate}
\end{lemma}

\begin{proof}
(1) No shared plaquette implies $V_e$ independent of $e'$, so $C_{ee'}=0$.

(2) For $\sigma,\tilde\sigma$ differing at $e'$:
each shared plaquette satisfies
$|\operatorname{Re}\operatorname{tr}W_p(\sigma)-\operatorname{Re}\operatorname{tr}W_p(\tilde\sigma)|\le 2N$.
Log-density ratio $\le 4N\beta n_{ee'}$ (factor~$2$ for normalising constants).
Coupling bound: $d_{\mathrm{TV}}\le 1-e^{-4N\beta n_{ee'}}\le 4N\beta n_{ee'}$.

(3) In $d=4$: $z_{\mathrm{pl}}=6$, $3$ other links per plaquette,
so $\sum_{e'}n_{ee'}=18$.  Part~(2): $\sum_{e'}C_{ee'}\le 72N\beta$.
\end{proof}

\begin{theorem}[Dobrushin--Poincar\'{e} volume-uniform spectral gap]
\label{thm:dobrushin_poincare}
For $N\ge2$ and $\beta<1/(72N)$,
the Gibbs measure $\mu_{\beta}$ satisfies
\begin{equation}
\label{eq:dp_poincare}
  \mathrm{Var}_{\mu_{\beta}}(f)\;\le\;\frac{1}{\kappa(\beta)}\,\mathcal{E}(f,f)
\end{equation}
with volume-independent constant
\begin{equation}
\label{eq:kappa_def}
  \kappa(\beta):=\Bigl(\frac{N}{4}-3\beta\Bigr)(1-72N\beta)>0.
\end{equation}
Here $\mathcal{E}(f,f)=\sum_{e}\int|\nabla_e f|_g^2\,d\mu_{\beta}$ is the Glauber Dirichlet form.
\end{theorem}

\begin{proof}
\textbf{Step~1 (single-site Poincar\'{e}).}
Since $\beta<1/(72N)\le N/12$ for all $N\ge1$,
Theorem~\ref{thm:bakry_emery_corr}: $\mathrm{gap}(\nu_e)\ge\kappa_0(\beta)>0$ uniformly in $\sigma$.

\textbf{Step~2 (Dobrushin condition).}
Lemma~\ref{lem:dobrushin_coeff}(3): $c:=\max_e\sum_{e'}C_{ee'}\le72N\beta<1$.

\textbf{Step~3 (Dobrushin--Poincar\'{e}).}
Zegarlinski, Commun.\ Math.\ Phys.\ 140 (1991) 455, Thm.~1:
conditions~(a)+(b) imply
\[  \mathrm{gap}_{\mathrm{Glauber}}(\mu_{\beta})\ge\kappa_0(\beta)(1-c)\ge\kappa(\beta), \]
independent of $|\Lambda|$.
\end{proof}

\begin{remark}
Theorem~\ref{thm:dobrushin_poincare} gives a second independent route to a
volume-uniform mass gap for $\beta<1/(72N)$, complementing the Osterwalder--Seiler
cluster-expansion bound for $\beta<\beta_c(N)$. The two proofs are logically independent.
\end{remark}

\subsection{Bridge to the OS transfer-matrix mass gap}
\label{ssec:gap_bridge}

\begin{theorem}[Dobrushin--Shlosman implies OS mass gap]
\label{thm:dp_to_os}
For $N\ge2$ and $\beta<1/(72N)$, all finite $\Lambda$:
\begin{equation}
\label{eq:dp_os}
m(\Lambda,\beta)=-\log\lambda_1(\Lambda,\beta)\ge\log\tfrac{1}{72N\beta}>0,
\end{equation}
uniformly in $|\Lambda|$.
\end{theorem}
\begin{remark}
Glauber Poincar\'e constant $\kappa(\beta)$ (Thm~\ref{thm:dobrushin_poincare})
controls Markov mixing; it is \emph{not} the OS mass gap directly.
The OS bound uses Dobrushin--Shlosman correlation decay.
\end{remark}

\begin{proof}
\textbf{Step~1.}
Lem.~\ref{lem:dobrushin_coeff}(3): $c:=\max_e\sum_{e'}C_{ee'}\le72N\beta<1$.
Dobrushin uniqueness (TPA~13 (1968), Thm.~1): $\mu_\beta$ unique.

\textbf{Step~2.}
Dobrushin--Shlosman (CMP~102 (1985)~87, Thm.~3):
for bounded local $f_A,f_B$ on disjoint $A,B$ at distance $d(A,B)$,
\begin{equation}
\label{eq:ds_decay}
|\operatorname{Cov}_{\mu_\beta}(f_A,f_B)|\le\|f_A\|_{\infty}\|f_B\|_{\infty}|A|c^{d(A,B)},
\quad c:=72N\beta<1.
\end{equation}
Uniform in $|\Lambda|$ and boundary conditions.

\textbf{Step~3.}
Set $f_A=f_B=\mathcal{O}:=\operatorname{Re}\operatorname{tr}W_p$, $|\mathcal{O}|_{\infty}\le N$.
Place at times $0$ and $T$; $d(A,B)=T$. From~\eqref{eq:ds_decay}:
\begin{equation}
\label{eq:gt_bound}
G(T):=|\langle\mathcal{O}(0),\mathcal{O}(T)\rangle_c|\le N^2(72N\beta)^T.
\end{equation}

\textbf{Step~4.}
OS expansion $G(T)=\sum_{n\ge1}c_n\lambda_n^T$, $c_n\ge0$.
Since $\phi_1>0$ (Krein--Rutman, Thm~\ref{thm:finite_lattice_mass_gap})
and $\mathcal{O}$ is non-constant (it varies over link configurations),
$c_1:=|\langle\phi_1,\mathcal{O}-\langle\mathcal{O}\rangle\rangle|^2>0$.
If $\lambda_1>72N\beta$,
then $G(T)\ge c_1\lambda_1^T$ dominates~\eqref{eq:gt_bound}, contradiction.
Hence $\lambda_1\le 72N\beta$, giving
\[m(\Lambda,\beta)\ge\log\tfrac{1}{72N\beta}>0.\qedhere\]
\end{proof}

\subsection{The residual open problem}
\label{ssec:residual_open}

\noindent\textbf{Open problem~A (weak-coupling uniform gap).}
Prove $m_{\infty}(\beta)>0$ uniformly in volume for all $\beta\ge\beta_c(N)$.
No analytic method currently applies in $[\beta_c,\beta_{\mathrm{dc}}]$ ($\beta_{\mathrm{dc}}\approx2.3$ for $\mathrm{SU}(2)$;
cluster expansion diverges; perturbation theory is gapless).

\medskip\noindent\textbf{Open problem~B (continuum limit).}
Assuming Open problem~A, prove $\Xi(\beta)\to\Xi_*\in(0,\infty)$ as $\beta\to\infty$
(Theorem~\ref{thm:clay_equiv}).
Requires non-perturbative control of effects of order $O(e^{-c/g^2})$,
invisible to all orders of perturbation theory.

\medskip\noindent Open problems~A and~B are the exact content of the
Clay Yang--Mills mass-gap conjecture.  Neither is resolved.
