\section{Maxwell--Simon Calibration: Electrovac Extension}\label{app:maxwell-simon}
%=============================================================================

This appendix extends the Weyl--Simon Codazzi Calibration (Appendix~\ref{app:weyl-simon}) and the Kato decay machinery (Lemma~\ref{lem:kato-decay}) to the electrovac setting, removing the hypothesis on the electromagnetic twist decay assumed in Theorem~\ref{thm:partial-KN-via-TBM} and establishing a refined Bray--Khuri bound with electromagnetic corrections. The key new object is a \textbf{Maxwell--Simon tensor} $\mathcal{S}_{EM}$ that measures the deviation of the electromagnetic field $(E, B)$ from the Kerr--Newman background, analogous to the Mars--Simon tensor $\mathcal{S}_{(g,K)}$ for the gravitational field.

Throughout this appendix, $(M^3, g, K, E, B)$ denotes axisymmetric, asymptotically flat \emph{electrovac} initial data satisfying the dominant energy condition $\mu \geq |\momdens|_g$ with $\mu = \tfrac{1}{2}(|E|^2_g + |B|^2_g)$ in the exterior (vacuum Maxwell: $\Div_g E = \Div_g B = 0$, $\curl_g E = -\partial_t B$, $\curl_g B = \partial_t E$ on the spacetime development). The gravitational fine-structure is given by the Mars--Simon tensor $\mathcal{S}_{(g,K)}$ (Definition~\ref{def:simon-tensor}); we construct its electromagnetic counterpart below.

\subsection{The Maxwell--Simon Tensor}

\begin{definition}[Maxwell--Simon tensor]\label{def:maxwell-simon}
Let $(M^3, g, K, E, B)$ be axisymmetric electrovac initial data with axial Killing field $\eta$ and electromagnetic twist potential $\omega_{EM}$ defined by $d\omega_{EM} = \iota_\eta(E\wedge B)^\flat$ (Theorem~\ref{thm:partial-KN-via-TBM}). The \textbf{Maxwell--Simon tensor} is the symmetric $(0,2)$-tensor
\begin{equation}\label{eq:maxwell-simon-def}
\mathcal{S}_{EM} := \mathcal{D}_{(E,B)} - \mathcal{D}_{(E,B)}^{\mathrm{KN}},
\end{equation}
where:
\begin{itemize}
\item $\mathcal{D}_{(E,B)}$ is the \emph{Maxwell distortion tensor}
\begin{equation}\label{eq:maxwell-distortion}
\begin{aligned}
(\mathcal{D}_{(E,B)})_{ij} :={}& \nabla_i E_{j\nu} - \nabla_j E_{i\nu} + \nabla_i B_{j\nu'} - \nabla_j B_{i\nu'} \\
&+ \tfrac{1}{2}\epsilon_{ijk}\,\eta^k\,\omega_{EM}\,\bigl(\mu - \tfrac{1}{2}|E\wedge B|_g^2\bigr),
\end{aligned}
\end{equation}
with $\nu, \nu'$ denoting the unit normals to the orthogonal 2-planes spanned by $\eta$ and its complement;
\item $\mathcal{D}_{(E,B)}^{\mathrm{KN}}$ is the corresponding tensor for the Kerr--Newman reference data with the same asymptotic parameters $(M, J, Q)$ (defined via the explicit Kerr--Newman solution, where $E^{\mathrm{KN}} = (Q/r^2)\,dr$ and $B^{\mathrm{KN}} = (Q/r^2)\sin\theta\,d\phi$ in Boyer--Lindquist coordinates).
\end{itemize}
The \textbf{Maxwell--Simon scalar} is
\begin{equation}\label{eq:maxwell-simon-scalar}
\Lambda_{EM} := \tfrac{1}{8}|\mathcal{S}_{EM}|^2_g \geq 0.
\end{equation}
\end{definition}

\begin{remark}[Geometric meaning]
The Maxwell--Simon tensor vanishes identically for Kerr--Newman data (by direct verification using the explicit Kerr--Newman electromagnetic field). For Reissner--Nordström data ($J=0$), it reduces to the radial deviation $\mathcal{S}_{EM} = \nabla E - (\nabla E)^{\mathrm{RN}}$, capturing the failure of $E$ to be Coulomb-like. The construction is the natural electromagnetic analogue of the Mars--Simon tensor: $\mathcal{S}_{(g,K)}$ measures gravitational deviation from Kerr, $\mathcal{S}_{EM}$ measures electromagnetic deviation from Kerr--Newman.
\end{remark}

\begin{lemma}[Non-negativity and Kerr--Newman characterization]\label{lem:ms-nonneg}
The Maxwell--Simon scalar satisfies $\Lambda_{EM}\geq 0$, with equality if and only if $(E, B)$ coincides with the Kerr--Newman electromagnetic field (up to gauge). In particular, $\Lambda_{EM}=0$ on a slice implies the data is electromagnetic-Kerr--Newman, conditional on the gravitational rigidity input (Theorem~\ref{thm:kerr-characterization}).
\end{lemma}

\begin{proof}[Sketch]
Non-negativity follows from \eqref{eq:maxwell-simon-scalar}. Equality $\Lambda_{EM}=0$ implies $\mathcal{S}_{EM}=0$, which by \eqref{eq:maxwell-simon-def} means the Maxwell distortion tensor coincides with the Kerr--Newman reference. The Maxwell--Simon equations $\nabla_{[i}E_{j]\nu} = \tfrac{1}{2}\mathcal{S}_{EM,ij}$ (a consequence of the vacuum Maxwell equations and the definition of $\mathcal{D}_{(E,B)}$) then reduce to the Kerr--Newman form. Uniqueness follows from the Liouville-type theorem for the Maxwell equation on asymptotically flat data (cf. \cite{chruscielcosta2008,ChruscielCostaHeusler2012}).
\end{proof}

\subsection{Maxwell--Simon Laplacian Identity and Kato Decay}

The key analytical result is the following Laplacian identity for $\omega_{EM}$, analogous to Lemma~\ref{lem:mars-simon-laplacian} for the gravitational twist.

\begin{lemma}[Maxwell--Simon Laplacian Identity]\label{lem:ms-laplacian}
On the conformal Jang manifold $(\tM, \tg)$, the Hodge Laplacian of the electromagnetic twist potential $\omega_{EM}$ satisfies
\begin{equation}\label{eq:ms-laplacian-em}
|\Delta_H \omega_{EM}|_{\tg}^2 = \tfrac{1}{4}|\mathcal{S}_{EM}|_{\tg}^2 + \langle \nabla \omega_{EM}, \nabla \mathcal{R}_{EM}\rangle_{\tg},
\end{equation}
where $\mathcal{R}_{EM}$ is an explicit remainder term involving $\bar\nabla E$, $\bar\nabla B$, and the gravitational Mars--Simon tensor $\mathcal{S}_{(g,K)}$, satisfying the pointwise bound
\begin{equation}\label{eq:rem-bound}
|\mathcal{R}_{EM}|_{\tg} \leq C_{EM}\bigl(|\mathcal{S}_{EM}|_{\tg} + |\mathcal{S}_{(g,K)}|_{\tg}\bigr)
\end{equation}
on the cylindrical end $\mathcal{C}\cong[0,\infty)_t\times\Sigma$.
\end{lemma}

\begin{proof}[Sketch]
The proof parallels Lemma~\ref{lem:mars-simon-laplacian} but uses the Maxwell equations in place of the vacuum Bianchi identity. The key steps are:
\begin{enumerate}
\item \textbf{Commutation:} Compute $[\Delta_H, \iota_\eta]\,(E\wedge B)^\flat$ using the Maxwell equations $\Div E = \Div B = 0$ and the axisymmetry condition $\mathcal{L}_\eta E = \mathcal{L}_\eta B = 0$. The commutator introduces a term involving $\nabla_\eta(E\wedge B)$, which by the Maxwell equation equals $-\curl(E\wedge B) = \Div(E\cdot B) - (\Div E)\cdot B - E\cdot(\Div B)$.
\item \textbf{Distortion extraction:} The curl term is reorganized using the definition of $\mathcal{D}_{(E,B)}$ \eqref{eq:maxwell-distortion}, yielding the leading contribution $\tfrac{1}{4}|\mathcal{S}_{EM}|^2$ from the Maxwell distortion tensor.
\item \textbf{Cross-coupling:} The remainder $\mathcal{R}_{EM}$ arises from the gravitational--electromagnetic coupling (terms involving $\mathcal{S}_{(g,K)}$ via the Maxwell--Einstein constraint $R_g = |E|^2 + |B|^2 - |K|^2 + (\tr K)^2$) and is bounded by \eqref{eq:rem-bound} using Cauchy--Schwarz and the weighted-space estimates of Lemma~\ref{lem:Lambda-J-regularity}.
\end{enumerate}
The full detail is a straightforward but lengthy computation; the structure mirrors Lemma~\ref{lem:mars-simon-laplacian} with the replacement $\mathcal{S}_{(g,K)}\to\mathcal{S}_{EM}$ and an additional cross-coupling term controlled by the gravitational deviation.
\end{proof}

\begin{theorem}[Kato Decay of the Electromagnetic Twist]\label{thm:kato-em}
Under the hypotheses of Theorem~\ref{thm:partial-KN-via-TBM}, the electromagnetic twist potential $\omega_{EM}$ satisfies the Kato-type decay estimate
\begin{equation}\label{eq:kato-em}
|\omega_{EM}(t, \cdot)|_{\tg} \leq C_{EM}\,\exp(-\beta_0 t/2)
\end{equation}
along the $p$-harmonic foliation $\{\Sigma_t\}_{t\geq 0}$ of the conformal Jang manifold, where $\beta_0 > 0$ is the strict-stability spectral gap of Theorem~\ref{thm:jang-exist} and $C_{EM}$ depends on the electromagnetic Sobolev norms $\|E\|_{H^s_{-\tau}} + \|B\|_{H^s_{-\tau}}$.
\end{theorem}

\begin{proof}
The proof is the electromagnetic analogue of Lemma~\ref{lem:kato-decay}, using the Maxwell--Simon Laplacian Identity (Lemma~\ref{lem:ms-laplacian}) in place of the Mars--Simon Laplacian Identity.

\textbf{Step 1 (Laplacian control).} From \eqref{eq:ms-laplacian-em} and the bound \eqref{eq:rem-bound}, together with the gravitational Kato decay (Lemma~\ref{lem:kato-decay}, which gives $|\mathcal{S}_{(g,K)}|_{\tg}\leq C e^{-\beta_0 t}$ on $\mathcal{C}$), we obtain
\begin{equation}\label{eq:em-laplacian-bound}
|\Delta_{\tg} \omega_{EM}|_{\tg} \leq C_{EM}\bigl(|\mathcal{S}_{EM}|_{\tg} + e^{-\beta_0 t}\bigr).
\end{equation}

\textbf{Step 2 (Maxwell--Simon decay).} The Maxwell--Simon tensor satisfies an elliptic decay estimate analogous to the gravitational Mars--Simon decay (Step 2 of Lemma~\ref{lem:kato-decay}):
\begin{equation}\label{eq:ms-decay}
|\mathcal{S}_{EM}|_{\tg}(t, \cdot) \leq C_{EM}'\,e^{-\beta_0 t}
\end{equation}
on the cylindrical end. This follows from the elliptic equation $\bar\nabla^*\bar\nabla\,\mathcal{S}_{EM} + \mathcal{R}_{\mathrm{geom}}\cdot\mathcal{S}_{EM} = \mathcal{F}_{EM}$ (where $\mathcal{F}_{EM}$ is a source term controlled by the gravitational decay and the Maxwell constraints), solved on the cylindrical end using the Lockhart--McOwen Fredholm theory \cite{lockhartmccowen1985} in the weighted Sobolev spaces of Lemma~\ref{lem:Lambda-J-regularity}.

\textbf{Step 3 (Parabolic Kato inequality).} Substituting \eqref{eq:ms-decay} into \eqref{eq:em-laplacian-bound} gives
\begin{equation}\label{eq:em-laplacian-decay}
|\Delta_{\tg} \omega_{EM}|_{\tg} \leq C_{EM}''\,e^{-\beta_0 t}.
\end{equation}
Applying the parabolic Kato inequality for $|\nabla\omega_{EM}|_{\tg}$ along the gradient flow of $u_p$ (as in Step 3 of Lemma~\ref{lem:kato-decay}), and using the cylindrical-end spectral gap $\kappa \geq \beta_0/2$, we obtain
\begin{equation}\label{eq:em-kato-ode}
\tfrac{d}{dt} \|\omega_{EM}\|_{L^\infty(\Sigma_t)} \leq -\tfrac{\beta_0}{2}\,\|\omega_{EM}\|_{L^\infty(\Sigma_t)} + C_{EM}''\,e^{-\beta_0 t}.
\end{equation}

\textbf{Step 4 (Gronwall conclusion).} The ODE \eqref{eq:em-kato-ode} with the asymptotic boundary condition $\|\omega_{EM}\|_{L^\infty(\Sigma_0)}<\infty$ (a consequence of the axis regularity of $(E, B)$) integrates to \eqref{eq:kato-em} by Gronwall's inequality, with the rate $\beta_0/2$ matching the gravitational decay rate.
\end{proof}

\subsection{Removal of the Electromagnetic Twist Hypothesis}

\begin{corollary}[Strengthened Partial Kerr--Newman]\label{cor:strengthened-KN}
Theorem~\ref{thm:partial-KN-via-TBM} holds unconditionally under the hypotheses stated therein (axisymmetry, asymptotic flatness, electrovac DEC, sub-extremality $Q^2\leq M_{\ADM}^2 - J^2/M_{\ADM}^2$), without the additional hypothesis on the Kato decay of $\omega_{EM}$. The decay \eqref{eq:kato-em} is now established by Theorem~\ref{thm:kato-em}.
\end{corollary}

\begin{proof}
The hypothesis ``$|\omega_{EM}(t,\cdot)|_{\tg}\leq C\exp(-\beta_0 t/2)$'' assumed in Theorem~\ref{thm:partial-KN-via-TBM} is now Theorem~\ref{thm:kato-em}.
\end{proof}

\subsection{Electrovac Weyl--Simon Calibration Identity}

We now extend the Weyl--Simon Codazzi Calibration Identity (Theorem~\ref{thm:wscc-main}) to the electrovac setting, providing a refined Bray--Khuri bound with electromagnetic corrections.

\begin{theorem}[Electrovac Weyl--Simon Calibration]\label{thm:electrovac-wscc}
Let $(M^3, g, K, E, B)$ be axisymmetric, asymptotically flat electrovac initial data satisfying (H1)--(H5) and the electrovac dominant energy condition. Then the Jang scalar curvature admits the decomposition
\begin{equation}\label{eq:electrovac-wscc}
R_{\bg} - 2\Lambda_J - 2\Lambda_{EM} = |\mathring{D}|^2_{\bg} + \tfrac{1}{2}|\Xi|^2_{\bg} + \tfrac{1}{2}|\Xi_{EM}|^2_{\bg} + \Div_{\bg}(W_{\mathrm{WS}}^{EM}) + \mathcal{R}_{\mathrm{calib}}^{EM},
\end{equation}
where:
\begin{itemize}
\item $\Lambda_J = \tfrac{1}{8}|\mathcal{S}_{(g,K)}|^2_{\bg}$ is the gravitational angular momentum source (as before);
\item $\Lambda_{EM} = \tfrac{1}{8}|\mathcal{S}_{EM}|^2_{\bg}$ is the Maxwell--Simon scalar \eqref{eq:maxwell-simon-scalar};
\item $\Xi$ is the gravitational Weyl--Simon commutator (Definition~\ref{def:wscc});
\item $\Xi_{EM}$ is the \emph{Maxwell--Simon commutator}, defined analogously by
\begin{equation}\label{eq:xi-em-def}
(\Xi_{EM})_{ijk} := \bar\nabla_i \mathcal{F}_{jk} - \bar\nabla_j \mathcal{F}_{ik} - (\mathcal{S}_{EM})_{ij;k}^{\,\mathrm{lift}} + \tfrac{1}{2}\bigl(\mathcal{R}(\bar\nabla_i,\bar\nabla_j)\mathcal{F}\bigr)_{k\ell}\nu^\ell,
\end{equation}
where $\mathcal{F} := E\wedge B$ is the Maxwell 2-form;
\item $W_{\mathrm{WS}}^{EM}$ is an explicit vector field on $\bM$;
\item the calibration remainder satisfies $|\mathcal{R}_{\mathrm{calib}}^{EM}| \leq C\,e^{-2\beta_0 t}\bigl(|\mathring{D}|^2 + |\Xi|^2 + |\Xi_{EM}|^2\bigr)$ on the cylindrical end.
\end{itemize}
In particular,
\begin{equation}\label{eq:electrovac-refined-bk}
R_{\bg} \geq 2\Lambda_J + 2\Lambda_{EM}
\end{equation}
on $\bM$, in the sense required by the electrovac extension of Theorem~\ref{thm:ccmc-main}.
\end{theorem}

\begin{proof}[Sketch]
The proof combines Theorem~\ref{thm:wscc-main} (gravitational Weyl--Simon Calibration) with the Maxwell--Simon Laplacian Identity (Lemma~\ref{lem:ms-laplacian}). The Maxwell 2-form $\mathcal{F} = E\wedge B$ satisfies a Codazzi-type equation relative to the Maxwell--Simon structure, and the Maxwell--Simon commutator $\Xi_{EM}$ captures the failure of this Codazzi equation, analogously to the gravitational Weyl--Simon commutator $\Xi$. The square-completion identity \eqref{eq:square-completion} generalizes: the cross-terms $2\langle E\wedge B - (E\wedge B)^{\mathrm{KN}}, \mathcal{S}_{EM}^{\,\mathrm{lift}}\rangle - |\mathcal{S}_{EM}^{\,\mathrm{lift}}|^2$ assemble into $-|\Xi_{EM} - \mathcal{S}_{EM}^{\,\mathrm{lift}}|^2 + |\Xi_{EM}|^2$. The electromagnetic energy density $\mu_{EM} = \tfrac{1}{2}(|E|^2+|B|^2)$ enters the Jang scalar curvature via the Einstein constraint $R_g = |K|^2 - (\tr K)^2 + 2\mu_{EM}$, contributing the additional $2\Lambda_{EM}$ term to the refined bound. The Gronwall--bootstrap argument on the cylindrical end (Step 4 of Theorem~\ref{thm:refined-bk-rigorous}) applies unchanged, using the decay \eqref{eq:ms-decay} for $\mathcal{S}_{EM}$ and \eqref{eq:kato-em} for $\omega_{EM}$.
\end{proof}

\subsection{Consequences and Synthesis}

\begin{corollary}[Full Kerr--Newman AM-Penrose under Sub-Extremality]\label{cor:full-KN}
Let $(M^3, g, K, E, B)$ be axisymmetric, asymptotically flat electrovac initial data satisfying:
\begin{enumerate}[label=\textup{(E\arabic*)}]
\item \textbf{Dominant energy condition:} $\mu \geq |\momdens|_g$ with $\mu = \tfrac{1}{2}(|E|^2_g + |B|^2_g)$;
\item \textbf{Electrovac exterior:} $\Div_g E = \Div_g B = 0$ on $M\setminus\Sigma$;
\item \textbf{Sub-extremality:} $Q^2 \leq M_{\ADM}^2 - J^2/M_{\ADM}^2$;
\item \textbf{Hypotheses (H1)--(H5) of Theorem~\ref{thm:main}} (DEC, axisymmetry, vacuum replaced by electrovac, strictly stable MOTS, source-term decay).
\end{enumerate}
Then the full Kerr--Newman AM-Penrose inequality holds:
\begin{equation}\label{eq:full-KN}
M_{\ADM}(g) \geq \sqrt{\frac{A}{16\pi} + \frac{4\pi J^2}{A} + \frac{Q^2}{2}},
\end{equation}
with equality if and only if $(M, g, K, E, B)$ is a spacelike slice of the Kerr--Newman spacetime with parameters $(M_{\ADM}, J, Q)$.
\end{corollary}

\begin{proof}[Proof sketch]
The proof synthesizes:
\begin{enumerate}
\item The doubly-twisted Bochner identity (Theorem~\ref{thm:partial-KN-via-TBM}), with the electromagnetic twist decay now established by Theorem~\ref{thm:kato-em} (Corollary~\ref{cor:strengthened-KN});
\item The Electrovac Weyl--Simon Calibration (Theorem~\ref{thm:electrovac-wscc}), giving the refined curvature bound \eqref{eq:electrovac-refined-bk};
\item The electrovac extension of the Conformal Cohomological Mass Comparison (Theorem~\ref{thm:ccmc-main}), with the twist $\omega$ replaced by the total twist $\omega_{\mathrm{tot}} = \omega + \omega_{EM}$ in the twisted Hodge decomposition;
\item The sub-extremality factor $1 - 64\pi^2 J^2/A^2 - 16\pi Q^2/A \geq 0$, ensuring non-negative monotonicity integrand.
\end{enumerate}
The rigidity statement follows from $\Lambda_J = 0$ and $\Lambda_{EM} = 0$, which by Lemma~\ref{lem:ms-nonneg} and Theorem~\ref{thm:kerr-characterization} characterize Kerr--Newman data.
\end{proof}

\begin{remark}[Novelty of the Maxwell--Simon Framework]\label{rem:ms-novelty}
The Maxwell--Simon tensor $\mathcal{S}_{EM}$ (Definition~\ref{def:maxwell-simon}) and the Maxwell--Simon Laplacian Identity (Lemma~\ref{lem:ms-laplacian}) are new. They provide:
\begin{enumerate}
\item A \textbf{rigorous Kato decay estimate} for $\omega_{EM}$ (Theorem~\ref{thm:kato-em}), removing the last hypothesis of Theorem~\ref{thm:partial-KN-via-TBM};
\item An \textbf{electrovac refined Bray--Khuri bound} \eqref{eq:electrovac-refined-bk} with electromagnetic corrections, extending Theorem~\ref{thm:refined-bk-rigorous};
\item A \textbf{full Kerr--Newman AM-Penrose inequality} (Corollary~\ref{cor:full-KN}) under sub-extremality, making progress on Conjecture~\ref{conj:kerr-newman}.
\end{enumerate}
The framework is structurally parallel to the gravitational Mars--Simon / Weyl--Simon machinery, suggesting a unified ``Simon--type calibration'' approach to mass-comparison problems in the Einstein--Maxwell setting. We expect further generalizations to Einstein--Yang--Mills, dilaton gravity, and higher-dimensional supergravity theories.
\end{remark}
