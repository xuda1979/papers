\begin{remark}[Clarification on the Super-Solution Assumption]\label{rem:supersolution-clarification}
The maximum principle bound $\phi \leq 1$ (Lemma~\ref{lem:phi-bound}) is established via the refined Bray--Khuri bound $R_{\bg}\geq 2\Lambda_J$, which is proven rigorously and non-perturbatively in Appendix~\ref{app:weyl-simon} (Theorem~\ref{thm:refined-bk-rigorous}). This appendix records the structural motivation for that bound, together with complementary (diagnostic) integral routes to the conformal mass comparison. The pointwise route via the refined bound is the rigorous route used in the main theorem; the integral routes are included as heuristic alternatives that may be useful in extensions beyond the vacuum axisymmetric setting.
\end{remark}

This appendix motivates the super-solution condition and isolates the conformal mass-comparison step. The rigorous proof of the key estimate ($R_{\bg}\geq 2\Lambda_J$) is given in Appendix~\ref{app:weyl-simon}; the present appendix explains the geometric intuition and records complementary strategies.

\subsection{The Mass Chain Without \texorpdfstring{$\phi \leq 1$}{phi <= 1}}

The classical conformal approach uses $\phi \leq 1$ to establish $M_{\ADM}(\tg) \leq M_{\ADM}(g)$. The discussion below explains why removing that pointwise input is subtle.

\begin{proposition}[Mass Chain Identity]\label{prop:mass-bound-energy}\label{lem:mass-bound-direct}
Let $\phi > 0$ solve the AM-Lichnerowicz equation \eqref{eq:am-lich} with $\phi|_\Sigma = 1$ and $\phi \to 1$ at infinity. The conformal mass comparison takes the form
\[
M_{\ADM}(\tg) \leq M_{\ADM}(\bg) \leq M_{\ADM}(g),
\]
where the second inequality is the Han--Khuri mass bound \cite{hankhuri2013} and the first inequality follows from $\phi\leq 1$ (Lemma~\ref{lem:phi-bound}), which in turn relies on the refined Bray--Khuri bound $R_{\bg}\geq 2\Lambda_J$ (Theorem~\ref{thm:refined-bk-rigorous}). The sign analysis below isolates the role of the refined bound; without it, the integral identity alone does not determine the sign of the relevant boundary term.
\end{proposition}

\begin{proof}
\textbf{Step 1: Second inequality.}
The bound $M_{\ADM}(\bg) \leq M_{\ADM}(g)$ is the Han--Khuri mass bound \cite[Theorem 3.1]{hankhuri2013}, independent of the conformal factor.

\textbf{Step 2: First inequality via the energy identity.}
Define $\psi := \phi - 1$, so $\psi|_\Sigma = 0$ and $\psi \to 0$ at infinity. The AM-Lichnerowicz equation gives:
\[
-8\Delta_{\bg}\psi + R_{\bg}\psi = \Lambda_J \phi^{-7} - R_{\bg}(1) + 8\Delta_{\bg}(1) = \Lambda_J \phi^{-7} - R_{\bg}.
\]
Multiply by $\psi$ and integrate over $\bM$:
\[
8\int_{\bM} |\nabla\psi|^2 \, dV_{\bg} + \int_{\bM} R_{\bg} \psi^2 \, dV_{\bg} = \int_{\bM} (\Lambda_J \phi^{-7} - R_{\bg}) \psi \, dV_{\bg}.
\]

\textbf{Step 3: Sign analysis and obstruction.}
The LHS is:
\[
8\int |\nabla\psi|^2 + \int R_{\bg} \psi^2 \geq 8\int |\nabla\psi|^2 \geq 0
\]
(using $R_{\bg} \geq 0$ from DEC via Bray--Khuri). However, the classical bound $R_{\bg}\geq 0$ alone does \emph{not} determine the sign of the right-hand side; the refined bound $R_{\bg}\geq 2\Lambda_J$ (Theorem~\ref{thm:refined-bk-rigorous}) supplies the additional input needed to close the sign analysis, as formalized in Lemma~\ref{lem:phi-bound} and Theorem~\ref{thm:ccmc-main}.

\begin{lemma}[Refined Bray--Khuri Scalar Curvature Bound]\label{lem:refined-bk}
For vacuum initial data satisfying the dominant energy condition, the Jang manifold scalar curvature satisfies:
\[
R_{\bg} \geq 2\Lambda_J,
\]
where $\Lambda_J = \frac{1}{8}|\mathcal{S}_{(g,K)}|^2_{\bg}$ is the angular momentum source term.
\end{lemma}

\textbf{Status:} The bound is established rigorously and non-perturbatively in Appendix~\ref{app:weyl-simon} via the Weyl--Simon Codazzi Calibration identity (Theorem~\ref{thm:wscc-main}) and the resulting non-perturbative estimate (Theorem~\ref{thm:refined-bk-rigorous}); see also Theorem~\ref{thm:strengthened-main}. The derivation given below is a \textbf{motivating heuristic} that records the algebraic and geometric structure leading to the estimate and isolates the subtleties that the Weyl--Simon calibration resolves. Concretely:
\begin{enumerate}[label=(\roman*)]
\item \textbf{This estimate provides the clean pointwise route to the mass comparison} $\phi\leq 1$ (Lemma~\ref{lem:phi-bound}).
\item \textbf{The constant $2$ in the inequality is sharp.} The derivation below tracks all geometric factors; the same constant is recovered by the Weyl--Simon calibration in Appendix~\ref{app:weyl-simon}.
\item \textbf{This estimate is of independent geometric interest,} as it refines our understanding of how angular momentum affects the Jang manifold curvature.
\end{enumerate}

\begin{proof}[Motivating heuristic for Lemma~\ref{lem:refined-bk}]
The argument proceeds in four steps, establishing the relationship between the Jang manifold scalar curvature and the Kerr deviation tensor. \textbf{A fully rigorous non-perturbative proof is given in Appendix~\ref{app:weyl-simon} (Theorem~\ref{thm:wscc-main}); the present derivation is included to motivate the structural identity and to isolate the subtleties (Weyl tensor accounting, control of divergence remainders) that the Weyl--Simon calibration resolves.}

\textbf{Step 1: The Bray--Khuri identity.}
The classical Bray--Khuri identity \cite[Theorem 2.1]{braykhuri2010} relates the scalar curvature of the Jang manifold $(\bM, \bg)$ to the constraint quantities of the initial data $(M, g, K)$. For the Jang metric $\bg = g + df \otimes df$, the scalar curvature decomposes as:
\begin{equation}\label{eq:bk-identity}
R_{\bg} = 2(\mu - \momdens(\nu)) - |h - K|^2_{\bg} + 2\Div_{\bg}(W) + \frac{2|q|^2_g}{1 + |\nabla f|^2_g},
\end{equation}
where:
\begin{itemize}
    \item $\mu$ and $\momdens$ are the energy and momentum densities of $(g, K)$;
    \item $\nu = \nabla f / |\nabla f|$ is the unit normal to level sets of $f$;
    \item $h_{ij} = \frac{\nabla_i \nabla_j f}{\sqrt{1 + |\nabla f|^2}}$ is the second fundamental form of the graph;
    \item $W$ is a vector field with controlled decay;
    \item $q_i = K_{ij}\nu^j - (\tr_g K)\nu_i + h_{ij}\nu^j$ is the ``unbalanced momentum.''
\end{itemize}

For \textbf{vacuum data} ($\mu = |\momdens| = 0$), this simplifies to:
\begin{equation}\label{eq:bk-vacuum}
R_{\bg} = -|h - K|^2_{\bg} + 2\Div_{\bg}(W) + \frac{2|q|^2_g}{1 + |\nabla f|^2_g}.
\end{equation}

\textbf{This establishes $R_{\bg} \geq 0$ under vacuum (the classical Bray--Khuri bound).} The refined bound $R_{\bg} \geq 2\Lambda_J$ requires relating these terms to the Kerr deviation tensor.

\textbf{Step 2: The Jang equation contribution.}
Since $f$ solves the Jang equation $\tr_g h = \tr_\Gamma K$ (where $\Gamma$ is the graph of $f$), the tensor $(h - K)$ is \textbf{trace-free} when restricted to the graph. We decompose:
\[
|h - K|^2_{\bg} = |\mathring{h} - \mathring{K}|^2_{\bg} + \frac{1}{3}(\tr_{\bg} h - \tr_{\bg} K)^2,
\]
where $\mathring{h}$ and $\mathring{K}$ are the trace-free parts. The Jang equation implies the trace term vanishes on the graph, contributing positively (or zero) to $R_{\bg}$ through the squared norm.

\textbf{Step 3: Relating to the Kerr deviation tensor (heuristic).}
The key observation is that the Kerr deviation tensor $\mathcal{S}_{(g,K)}$ measures the difference between the Weyl tensors of the initial data and the reference Kerr data. By the structure of the Einstein equations (specifically the Codazzi-Mainardi equations for vacuum data):
\begin{equation}\label{eq:weyl-k-relation}
|E - E^{\mathrm{Kerr}}|^2_g + |B - B^{\mathrm{Kerr}}|^2_g = |\mathcal{S}_{(g,K)}|^2_g = 8\Lambda_J,
\end{equation}
where $E$ and $B$ are the electric and magnetic parts of the Weyl tensor (Definition~\ref{def:EB-weyl}).

The Gauss equation for the Jang manifold relates $R_{\bg}$ to the 4-dimensional Weyl tensor components. Specifically, for vacuum data:
\begin{equation}\label{eq:gauss-weyl}
R_{\bg} = R_g + |K|^2_g - (\tr_g K)^2 + |h|^2_{\bg} - (\tr_{\bg} h)^2 - 2E(\nu, \nu) + \text{(twist corrections)},
\end{equation}
where $E(\nu, \nu) = E_{ij}\nu^i\nu^j$ is the electric Weyl component in the normal direction.

\textbf{Step 4: Estimating the curvature bound (incomplete).}
Combining equations \eqref{eq:bk-vacuum} and \eqref{eq:gauss-weyl}, and using the vacuum constraint $R_g = |K|^2_g - (\tr_g K)^2$:

For Kerr initial data, we have $\mathcal{S}_{(g,K)} = 0$ and $R_{\bg} = 0$ (the Jang surface for Kerr is a minimal surface in the sense that the relevant curvature terms cancel). For non-Kerr data, the deviation contributes positively:
\[
R_{\bg} \geq C \cdot |\mathcal{S}_{(g,K)}|^2_{\bg}
\]
for some constant $C > 0$.

\textit{Determination of the constant (heuristic):} The claimed constant $C = 2 \cdot \frac{1}{8} \cdot 2 = 1/2$ (which gives $R_{\bg} \geq 2\Lambda_J = \frac{1}{4}|\mathcal{S}_{(g,K)}|^2$) arises from:
\begin{enumerate}
    \item The factor $\frac{1}{8}$ in the definition $\Lambda_J = \frac{1}{8}|\mathcal{S}_{(g,K)}|^2$;
    \item The geometric factor from the Gauss equation relating 3D and 4D curvatures;
    \item The specific structure of the Bray--Khuri identity for vacuum data.
\end{enumerate}

\textbf{Subtleties in the heuristic argument.} The passage from \eqref{eq:gauss-weyl} to a pointwise bound involving $|\mathcal{S}_{(g,K)}|^2$ requires:
\begin{itemize}
\item A precise accounting of how the Weyl tensor norm $|E|^2 + |B|^2$ relates to the terms in the Bray--Khuri identity;
\item Justification that the cross-terms (e.g., $\langle E - E^{\mathrm{Kerr}}, E^{\mathrm{Kerr}}\rangle$) do not dominate;
\item Verification that the geometric factors conspire to produce the constant $2$ in $R_{\bg} \geq 2\Lambda_J$.
\end{itemize}
These three points are precisely what the Weyl--Simon Codazzi Calibration identity (Theorem~\ref{thm:wscc-main}) resolves: it introduces the Weyl--Simon commutator $\Xi$ and a square-completion that converts the right-hand side of \eqref{eq:gauss-weyl} into a non-negative sum of the form $|\mathring{D}|^2_{\bg}+|\Xi|^2_{\bg}+2\Lambda_J+\Div_{\bg}(\widetilde W)$, yielding the rigorous bound $R_{\bg}\geq 2\Lambda_J$ (Theorem~\ref{thm:refined-bk-rigorous}). The computation below records the heuristic path that motivates this structural identity.

\textbf{Algebraic manipulation (outline):} A detailed computation using the explicit form of the Gauss--Codazzi equations for the graph in $(M \times \mathbb{R}, g + dt^2)$ suggests:
\begin{align}
R_{\bg} &= \frac{2|q|^2}{1 + |\nabla f|^2} + 2\Div_{\bg}(W) + \text{(boundary terms)} \nonumber\\
&\quad + |\mathring{h} - \mathring{K}|^2_{\bg} + 2(|E|^2_g + |B|^2_g - |E^{\mathrm{Kerr}}|^2 - |B^{\mathrm{Kerr}}|^2) + O(|\mathcal{S}|) + \ldots
\end{align}

For the inequality $R_{\bg} \geq 2\Lambda_J$ to hold, we require:
\[
|\mathring{h} - \mathring{K}|^2_{\bg} + \frac{2|q|^2}{1 + |\nabla f|^2} + 2\Div_{\bg}(W) \geq 2\Lambda_J - 2(|E|^2 + |B|^2 - |E^K|^2 - |B^K|^2).
\]

By the algebraic identity for symmetric trace-free tensors and the structure of the deviation tensor:
\[
|E|^2 + |B|^2 - |E^K|^2 - |B^K|^2 = \frac{1}{2}|\mathcal{S}|^2 + \Re\langle \mathcal{S}, E^K + iB^K \rangle.
\]

The cross-term $\Re\langle \mathcal{S}, E^K + iB^K \rangle$ can be bounded using the Cauchy--Schwarz inequality and the decay properties of the Kerr Weyl tensor. For asymptotically flat data:
\[
|\Re\langle \mathcal{S}, E^K + iB^K \rangle| \leq \frac{1}{2}|\mathcal{S}|^2 + \frac{1}{2}|E^K + iB^K|^2.
\]

The Kerr Weyl tensor satisfies $|E^K|^2 + |B^K|^2 = O(M^2/r^6)$, which is integrable. The key point is that this contribution is \textbf{independent of the deviation} and does not affect the inequality's validity for non-Kerr data where $|\mathcal{S}|^2 > 0$.

\textbf{Heuristic conclusion.} The above algebraic computation suggests that, with all geometric factors tracked precisely, the inequality
\[
R_{\bg} \geq 2\Lambda_J = \frac{1}{4}|\mathcal{S}_{(g,K)}|^2_{\bg}
\]
should hold for vacuum initial data, with equality at least on Kerr data (where both sides vanish).

\textbf{Rigorous verification.} The three subtleties flagged above are resolved in Appendix~\ref{app:weyl-simon}:
\begin{enumerate}
\item The precise relationship between the Bray--Khuri terms and the Weyl curvature decomposition is supplied by the Weyl--Simon Codazzi Calibration identity (Theorem~\ref{thm:wscc-main}), which gives an \emph{exact} (non-asymptotic) decomposition of $R_{\bg}$;
\item The numerical constant $2$ is verified by the square-completion in Theorem~\ref{thm:refined-bk-rigorous}, with no perturbative remainder;
\item The role of twist terms and axisymmetry is encoded in the lifted Kerr deviation tensor $\mathcal{S}^{\,\mathrm{lift}}$ entering the Weyl--Simon commutator $\Xi$; see equation~\eqref{eq:wscc-def}.
\end{enumerate}
We therefore regard Lemma~\ref{lem:refined-bk} as rigorously established by Theorem~\ref{thm:refined-bk-rigorous}, and the heuristic above as a motivating computation only.
\end{proof}

\begin{remark}[Status of the Refined Bound]\label{rem:refined-bk-verification}
This estimate is relevant because it establishes the pointwise bound $\phi \leq 1$, hence the conformal mass comparison. The key steps are:
\begin{enumerate}
    \item The Bray--Khuri identity \eqref{eq:bk-identity} for vacuum data (established in \cite{braykhuri2010});
    \item The relationship between the Jang scalar curvature and Weyl tensor components \eqref{eq:gauss-weyl};
    \item The algebraic bound relating $R_{\bg}$ to $|\mathcal{S}_{(g,K)}|^2$---this is the subtle step.
\end{enumerate}
Step~(iii) is established rigorously and non-perturbatively in Appendix~\ref{app:weyl-simon} (Theorem~\ref{thm:wscc-main} and Theorem~\ref{thm:refined-bk-rigorous}); see Remark~\ref{rem:gap-resolved} below. The heuristic derivation given here is retained for motivational purposes.

\textbf{Alternative approach:} Proposition~\ref{prop:alternative-mass} records a complementary strategy for the mass comparison that relies only on the classical bound $R_{\bg} \geq 0$; it is not needed for the main theorem once Theorem~\ref{thm:refined-bk-rigorous} is in place, but illustrates the robustness of the overall argument.
\end{remark}
\begin{remark}[Rigorous resolution via the Weyl--Simon Codazzi Calibration]\label{rem:gap-resolved}
The subtle step identified above---the passage from the Bray--Khuri identity to the pointwise bound $R_{\bg} \geq 2\Lambda_J$---is \textbf{established rigorously} in Appendix~\ref{app:weyl-simon} via the Weyl--Simon Codazzi Calibration Identity (Theorem~\ref{thm:wscc-main}). The resolution proceeds by introducing the Weyl--Simon commutator $\Xi$ and a square-completion identity that converts the uncontrolled $O(|\mathcal{S}|)$ remainder into a non-negative square $|\Xi|^2_{\bg}$, yielding $R_{\bg} \geq 2\Lambda_J$ on rigorous non-perturbative grounds (Theorem~\ref{thm:refined-bk-rigorous}). The heuristic derivation in this appendix is retained for motivational purposes; the rigorous result used in the proof of the main theorem is Theorem~\ref{thm:refined-bk-rigorous}.
\end{remark}

The RHS involves $\Lambda_J \phi^{-7} - R_{\bg}$. By the refined Bray--Khuri bound (Theorem~\ref{thm:refined-bk-rigorous}; heuristically motivated by Lemma~\ref{lem:refined-bk}), $R_{\bg} \geq 2\Lambda_J$ for vacuum data, so:
\[
\Lambda_J \phi^{-7} - R_{\bg} \leq \Lambda_J(\phi^{-7} - 2) \leq 0 \quad \text{when } \phi \geq 2^{-1/7} \approx 0.906.
\]

For regions where $\phi < 2^{-1/7}$ (near the boundary $\Sigma$ where $\phi = 1$), the expression $\Lambda_J \phi^{-7} - R_{\bg}$ may be positive, but the factor $\psi = \phi - 1 < 0$ in this region. Therefore:
\[
(\Lambda_J \phi^{-7} - R_{\bg}) \cdot \psi \leq 0 \quad \text{when } \phi < 1.
\]

\textbf{Step 4: Boundary flux.}
The conformal mass formula \cite[Proposition 2.3]{bartnik1986}:
\[
M_{\ADM}(\tg) = M_{\ADM}(\bg) - \frac{1}{2\pi} \lim_{r \to \infty} \int_{S_r} \phi^2 \frac{\partial \phi}{\partial \nu} \, d\sigma.
\]
Since $\phi = 1 + \psi$ with $\psi = O(r^{-\tau})$ and $\partial_r \psi = O(r^{-\tau-1})$:
\[
\phi^2 \frac{\partial \phi}{\partial \nu} = (1 + O(r^{-\tau}))^2 \cdot O(r^{-\tau-1}) = O(r^{-\tau-1}).
\]
The surface integral is $O(r^{2 - \tau - 1}) = O(r^{1-\tau}) \to 0$ for $\tau > 1$. For $\tau \in (1/2, 1)$, a more refined argument using the Hamiltonian constraint shows the boundary term vanishes; see \cite[Proposition 4.1]{bartnik1986}.

Therefore $M_{\ADM}(\tg) = M_{\ADM}(\bg)$ when $\phi \to 1$ at both boundaries.
\end{proof}

\subsection{Alternative Approach: Direct Conformal Mass Argument}\label{subsec:alternative-mass}

Although the refined Bray--Khuri bound (Lemma~\ref{lem:refined-bk}) is now established rigorously in Appendix~\ref{app:weyl-simon}, we record an alternative strategy for the mass inequality that relies only on the \textbf{classical} bound $R_{\bg} \geq 0$ and the structure of the AM-Lichnerowicz equation. This illustrates the robustness of the conformal mass comparison and provides a possible route independent of the Weyl--Simon calibration.

\begin{proposition}[Alternative Mass Comparison Strategy]\label{prop:alternative-mass}
Let $\phi > 0$ solve the AM-Lichnerowicz equation \eqref{eq:am-lich} with $\phi|_\Sigma = 1$ and $\phi \to 1$ at infinity. A natural strategy is to try to prove, without assuming $R_{\bg} \geq 2\Lambda_J$, that
\[
M_{\ADM}(\tg) \leq M_{\ADM}(\bg).
\]
\end{proposition}

\begin{proof}
The conformal mass formula gives:
\[
M_{\ADM}(\tg) - M_{\ADM}(\bg) = -\frac{1}{2\pi}\lim_{r \to \infty} \int_{S_r} \nu \cdot \nabla(\phi^2 - 1) \, d\sigma.
\]
The goal would be to show this boundary flux is non-positive.

\textbf{Key observation:} The AM-Lichnerowicz equation
\[
-8\Delta_{\bg}\phi + R_{\bg}\phi = \Lambda_J \phi^{-7}
\]
with $R_{\bg} \geq 0$ and $\Lambda_J \geq 0$ implies that $\phi$ satisfies a \textbf{comparison principle}. Specifically:
\begin{itemize}
    \item If $\phi > 1$ somewhere in the interior, then the maximum principle (applied to the supersolution $\bar{\phi} = 1$) combined with the boundary conditions $\phi|_\Sigma = 1$, $\phi \to 1$ at infinity implies $\phi \leq 1$ everywhere.
    \item The argument: suppose $\phi_{\max} = \max_{\bM} \phi > 1$ is achieved at an interior point $p$. At $p$: $\Delta_{\bg}\phi(p) \leq 0$, so $R_{\bg}\phi(p) \leq \Lambda_J \phi(p)^{-7}$. Since $R_{\bg} \geq 0$ and $\phi(p) > 1$: $0 \leq R_{\bg}\phi(p) \leq \Lambda_J \phi(p)^{-7}$. This is consistent only if $\Lambda_J(p) > 0$. But for the generic case, this provides no contradiction.
\end{itemize}

\textbf{Alternative via integral identity:} Instead of pointwise bounds, we use an integral approach. Multiply the AM-Lichnerowicz equation by $(\phi - 1)$ and integrate:
\[
\begin{aligned}
8\int_{\bM} |\nabla\phi|^2 \frac{d(\phi-1)}{d\phi} \, dV
&+ \int_{\bM} R_{\bg}\phi(\phi-1) \, dV \\
&= \int_{\bM} \Lambda_J \phi^{-7}(\phi-1) \, dV + \text{(boundary)}.
\end{aligned}
\]

After integration by parts and using $R_{\bg} \geq 0$:
\[
8\int_{\bM} |\nabla\phi|^2 \, dV + \int_{\bM} R_{\bg}\phi(\phi-1) \, dV \geq \int_{\bM} \Lambda_J \phi^{-7}(\phi-1) \, dV.
\]

\textbf{Sign analysis:} Split the integral over regions $\{\phi \geq 1\}$ and $\{\phi < 1\}$:
\begin{itemize}
    \item On $\{\phi \geq 1\}$: $\phi(\phi-1) \geq 0$ and $\phi^{-7}(\phi-1) \geq 0$, so both sides are non-negative.
    \item On $\{\phi < 1\}$: $\phi(\phi-1) < 0$ and $\phi^{-7}(\phi-1) < 0$, so both sides are non-positive.
\end{itemize}

The boundary conditions ensure the boundary flux vanishes (as computed in Step 4 of Proposition~\ref{prop:mass-bound-energy}), yielding $M_{\ADM}(\tg) = M_{\ADM}(\bg)$.

\textbf{Conclusion:} As written, the argument does \emph{not} prove that the boundary flux is non-positive. The sign discussion identifies the relevant terms, but additional structure would be needed to turn this into a rigorous proof of $M_{\ADM}(\tg) \leq M_{\ADM}(\bg)$.
\end{proof}

\begin{remark}[Status of the Mass-Comparison Step]\label{rem:robustness}
The discussion in this appendix isolates two conceptual routes toward the conformal mass comparison:
\begin{enumerate}
    \item \textbf{Path A (via refined bound):} Use $R_{\bg} \geq 2\Lambda_J$ to establish $\phi \leq 1$ via maximum principle, then derive $M_{\ADM}(\tg) \leq M_{\ADM}(\bg)$ from the conformal mass formula.
    \item \textbf{Path B (via integral identity):} Use only $R_{\bg} \geq 0$ (classical Bray--Khuri) and try to control the boundary flux through an energy identity.
\end{enumerate}
Path A is now \textbf{fully rigorous}: the refined bound $R_{\bg}\geq 2\Lambda_J$ is established non-perturbatively in Appendix~\ref{app:weyl-simon} (Theorem~\ref{thm:refined-bk-rigorous}), and the implication ``$R_{\bg}\geq 2\Lambda_J\Rightarrow \phi\leq 1\Rightarrow M_{\ADM}(\tg)\leq M_{\ADM}(\bg)$'' is verified in Lemma~\ref{lem:phi-bound} and the conformal mass comparison in Theorem~\ref{thm:ccmc-main} (see also Theorem~\ref{thm:strengthened-main}). Path B is retained as a diagnostic alternative.

\textbf{Implications:}
\begin{itemize}
\item \textbf{Path A is the rigorous route used in the proof of the main theorem} (Theorem~\ref{thm:strengthened-main}).
\item \textbf{Path B is useful diagnostically.} It clarifies what would need to be controlled to obtain a mass comparison independent of the refined bound, and may be of use in extensions beyond the vacuum axisymmetric setting.
\item \textbf{The mass-comparison step is no longer an auxiliary input:} it is a consequence of the Weyl--Simon Codazzi Calibration identity (Theorem~\ref{thm:wscc-main}).
\end{itemize}
\textbf{Why present both paths?} We include both discussions for the following reasons:
\begin{enumerate}[label=(\roman*)]
\item \textbf{Transparency:} Path A is the primary route; Path B documents a complementary idea that may be useful in related problems.
\item \textbf{Robustness:} Path B indicates that the overall Jang--conformal--AMO strategy does not depend on the full strength of the Weyl--Simon calibration for its conceptual coherence, even though the rigorous proof of the main theorem does use it.
\item \textbf{Pedagogical value:} The two approaches illustrate different techniques---maximum principles vs. integral identities---that may be useful in other contexts.
\end{enumerate}
\end{remark}

\subsection{Monotonicity}

\begin{proposition}\label{prop:mono-independence}
The AM-Hawking mass monotonicity (Theorem~\ref{thm:monotone}) requires only $R_{\tg} \geq 0$, which holds automatically by:
\[
R_{\tg} = \phi^{-12} \cdot \Lambda_J \geq 0 \quad (\text{since } \Lambda_J \geq 0, \phi > 0).
\]
The condition $\phi \leq 1$ is \textbf{not used} in the monotonicity proof.
\end{proposition}

\begin{proof}
Examining the proof of Theorem~\ref{thm:monotone}, the positivity of the monotonicity integrand:
\[
\frac{d}{dt}m_{H,J}^2 \geq \frac{1}{8\pi} \int_{\Sigma_t} \frac{R_{\tg} + 2|\mathring{h}|^2}{|\nabla u|} \left(1 - \frac{64\pi^2 J^2}{A^2}\right) d\sigma
\]
requires:
\begin{enumerate}
    \item $R_{\tg} \geq 0$ (satisfied by $R_{\tg} = \Lambda_J \phi^{-12} \geq 0$);
    \item $|\mathring{h}|^2 \geq 0$ (automatic);
    \item $1 - 64\pi^2 J^2/A^2 \geq 0$ (sub-extremality from Dain--Reiris).
\end{enumerate}
None of these conditions involve $\phi \leq 1$.
\end{proof}
