\title[Critical Dirichlet Spectral Zeta Asymptotics]{Critical Dirichlet Spectral Zeta Asymptotics on Graphs of Quadratic Volume Growth}
\author[D. Xu]{Da Xu}
\address{China Mobile Research Institute, Beijing, P.R.~China}
\email{xudayj@chinamobile.com}
\subjclass[2020]{Primary 58J50, 31C20, 60J10; Secondary 05C50, 39A12, 05C81}
\keywords{spectral zeta, graph Laplacian, heat kernel, quadratic volume growth, random walk homogenisation}
\date{}

\begin{abstract}\sloppy
We study the Dirichlet spectral zeta value $Z_H(1)=\tr(\LL_H^{-1})$ for large finite domains in bounded-degree graphs of quadratic volume growth. In the critical two-dimensional regime one expects growth of order $|H|\log|H|$, with leading coefficient determined by the on-diagonal heat kernel. We prove a time-domain criterion that yields this asymptotic from explicit geometric, on-diagonal, and long-time Dirichlet assumptions. For metric-ball exhaustions $G_n=B_{R_n}(x_0)$ with $N_n=|G_n|$, under uniform quadratic growth, a Poincar\'e inequality, an annulus-volume bound, a long-time Dirichlet heat-kernel estimate, and a weaker time-summed on-diagonal hypothesis with coefficient $\cG$, we prove $Z_n(1)/(N_n\log N_n)\to \cG$; this formulation allows periodic residue-class oscillations. Under a stronger quantitative on-diagonal asymptotic, we obtain sharp $O(1)$ control of truncated return sums, the bound $Z_n(1)\le \cG N_n\log N_n+O(N_n)$, and mesoscopic Green-window, heat-trace, reciprocal spectral-mass, and counting consequences. The proof separates the contribution of geometry, the on-diagonal heat kernel, and the long-time Dirichlet tail, and the same argument extends to abstract exhaustions once the boundary-layer and tail estimates are imposed directly at the domain level. In the appendix we verify the long-time and mesoscopic inputs on square domains for a broad nonnegative diagonal-symbol class, and for an explicit lazy axial-diagonal family we obtain a corresponding square-domain theorem together with a two-term $R^2$ expansion.
\end{abstract}

\section{Introduction}

We study the Dirichlet spectral zeta value on finite domains in infinite graphs. For a finite connected graph $H$ (no killing), the random-walk Laplacian has eigenvalues $0=\lambda_0 < \lambda_1 \leq \lambda_2 \leq \cdots$, so $\sum_k \lambda_k^{-1}$ must be interpreted via the pseudoinverse (cf.\ \Cref{rem:kirchhoff-precise}). Under the \emph{Dirichlet} setup used throughout this paper, obtained by killing the walk upon exiting a finite domain, all eigenvalues are strictly positive and the spectral zeta function is unambiguously
\[
  \zeta_H(s)=\sum_{k\ge1}\lambda_k(\LL_H)^{-s},\qquad Z_H(1):=\zeta_H(1)=\tr(\LL_H^{-1}).
\]
In continuum spectral geometry, critical two-dimensional phenomena are reflected in logarithmic divergences of Green traces and related zeta quantities. The discrete quadratic-growth case is the graph-theoretic analogue of that borderline regime, and the main question is whether one can read off the leading $N\log N$ asymptotic directly from the return probability of the underlying walk.

\begin{remark}[Connection to Kirchhoff Index]\label{rem:kirchhoff}
The value $Z_H(1) = \sum_k \lambda_k^{-1}$ is closely related to the Kirchhoff index $K(H)$, which is the sum of effective resistances $\Reff(x,y)$ between all pairs of vertices \cite{KleinRandic93}. For the standard (non-Dirichlet) Laplacian on a connected graph, there are exact identities relating the trace of the pseudoinverse to $K(H)$ (see \cite[Chapter 9]{LyonsPeres16}).
\end{remark}

\noindent\textit{Dirichlet convention.} Under the Dirichlet setup used in this paper (killing at the boundary of a finite domain),
we recall explicitly that
$Z_H(1)=\sum_k \lambda_k(\LL_H)^{-1}=\sum_{v\in V(H)} G_H(v,v)$; see also \Cref{rem:kirchhoff-precise}.

\begin{remark}[Kirchhoff index: explicit identities under our conventions]\label{rem:kirchhoff-precise}
Let $H$ be a finite connected graph (no killing). Write $D$ for the degree matrix, $A$ for the adjacency, and $L=D-A$ (combinatorial Laplacian). Then, with $0=\mu_1(L)<\mu_2\le\cdots\le\mu_n$ denoting the eigenvalues of $L$,
\[
  K(H)=\sum_{\{x,y\}}\Reff(x,y)=n\sum_{k=2}^n \mu_k(L)^{-1};
\]
see \cite[Thm.~9.23]{LyonsPeres16}. For the Dirichlet operator $\LL_H$ used in this paper (killing at $\partial H$), we have $Z_H(1)=\sum_k\lambda_k(\LL_H)^{-1}=\sum_{v\in H}G_H(v,v)$; this is the natural Dirichlet analogue of the global Kirchhoff identity. Thanks to bounded degree, traces computed in $\ell^2(m)$ or $\ell^2(|\cdot|)$ agree by similarity, cf.\ \Cref{rem:measures_operators}.
\end{remark}

We investigate the asymptotic behaviour of $Z_n(1)$ for large finite subgraphs $G_n$ exhausting an infinite graph $G$. Our focus is the critical case of \emph{quadratic volume growth} ($|B_R| \asymp R^2$), corresponding to an effective dimension $d=2$. This is the recurrent borderline regime, and it is precisely where the Green trace is expected to pick up the logarithmic factor summarized in \Cref{tab:growth}. Our aim is to show that, under suitable geometric regularity and a sharp long-time Dirichlet estimate, the leading asymptotic of $Z_n(1)$ is determined by the on-diagonal heat kernel. The results apply directly to models where these inputs can be verified, such as $\mathbb Z^2$ and a broad class of periodic finite-range walks.

\subsection{Results and Assumptions}
The central statement of the paper is the metric-ball criterion: in the quadratic-growth regime, once one has large-scale regularity, negligible boundary layer, and a sharp long-time Dirichlet tail, the leading Green-trace asymptotic is governed by the on-diagonal heat kernel. The remaining theorems extract consequences and extensions of that principle.
\begin{enumerate}
  \item Main criterion on metric balls: under \textup{(VG(2))}, \textup{(PI)}, \textup{(AV)}, a long-time Dirichlet tail on balls, and the weaker time-summed on-diagonal input \textup{(TS)}, one has
  \[
    Z_n(1)\sim \cG N_n\log N_n.
  \]
  \item Quantitative refinement: under the stronger pointwise asymptotic \textup{(QH)}, the same argument yields in addition
  \[
    Z_n(1)\le \cG N_n\log N_n+O(N_n).
  \]
  \item Mesoscopic consequences: under \textup{(TS)} one obtains logarithmically macroscopic Green-window asymptotics, while under \textup{(QH)} one further gets uniform heat-trace asymptotics, a quantitative windowed Green-trace identity, an asymptotically exact reciprocal spectral-mass law on slowly trimmed polynomial annuli, two-sided reciprocal-mass and counting laws on fixed annuli, and a low-lying spectral counting consequence in the diffusive subwindow.
  \item Abstract exhaustion extensions: once the boundary layer and long-time Dirichlet tail are imposed directly at the domain level, the same method yields the ratio limit and its quantitative refinement without any metric-ball hypothesis.
\end{enumerate}

The appendix treats square domains. For a broad diagonal-symbol class it proves the long-time Dirichlet and mesoscopic heat-trace inputs, and for a one-parameter axial-diagonal family it combines these inputs into an explicit model theorem. It also establishes a two-term expansion
\[
  \tr(\mathcal L_R^{-1})
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}R^2\log R^2
  + C_F R^2
  + o(R^2)
\]
for that class. The role of the various assumptions is straightforward: \textup{(VG(2))} and \textup{(PI)} provide PHI, Gaussian bounds, exit estimates, and Faber--Krahn; for metric balls, \textup{(AV)} makes the boundary layer negligible at scale $N_n\log N_n$; and the leading constant comes from the on-diagonal return probability. The ratio limit needs only the weaker time-summed hypothesis \Cref{ass:TS}, whereas the sharp $O(N_n)$ refinement and the mesoscopic statements use the summable pointwise remainder \Cref{ass:QH}. We collect the standing notation in the following paragraph.

\paragraph{Notation and Standing Assumptions.}
\noindent\textbf{Graph/metric.} $G=(V,E)$ is infinite, connected, with bounded degree. Closed balls are denoted by $B_R(x)=\{y:d(x,y)\le R\}$ and $V(x,R)=|B_R(x)|$.

\noindent\textbf{Dirichlet objects on a finite domain $H$.} For $H\subset V$ finite, $p_t^{H}(x,y)$ is the Dirichlet heat kernel (LSRW killed on leaving $H$), $G_H(x,y)=\sum_{t\ge0} p_t^{H}(x,y)$, $Z_H(1)=\sum_{x\in H} G_H(x,x)=\tr(\LL_H^{-1})$, $(\lambda_k(H))_{k\ge1}$ are the Dirichlet eigenvalues of $\LL_H$, and $\phi_1^{H}$ is an $L^2$-normalised ground state with respect to counting measure.

\noindent\textbf{Exhaustion/scale.} For $G_n=B_{R_n}(x_0)$ write $N_n=|G_n|$. Fix $\eta\in(0,1/2)$ and set the interior time scale $T_n:=\lfloor R_n^{2(1-2\eta)}\rfloor$.

\noindent\textbf{Structural hypotheses.} (VG(2)) $V(x,R)\asymp R^2$; (PI) scale-invariant Poincar\'e inequality; (AV) uniform annulus-volume bound $|B_R(x)\setminus B_{R-w}(x)|\lesssim Rw$ for $1\le w\le R$; (QH) uniform on-diagonal asymptotic $p_t(x,x)=\cG t^{-1}+O(t^{-1-\delta})$ for integers $t\to\infty$, with some $\cG>0,\delta>0$; and (LT) the long-time Dirichlet estimate on metric balls.

\noindent\textbf{Consequences used repeatedly.} From VD+PI we obtain PHI, Gaussian bounds, exit/maximal estimates, and Faber--Krahn; see \Cref{rem:chain}. We additionally assume the long-time Dirichlet estimate on balls (LT), used only in the upper bound.

\noindent\textbf{Conventions.} Symbols $C,c,c_i$ depend only on $(\mathrm{VG(2)},\mathrm{PI},\mathrm{AV},\Delta,\mathrm{QH},\mathrm{LT})$ and the chosen $\eta$, insofar as the corresponding hypotheses are invoked. We write $A\lesssim B$ for $A\le C B$, and $A\asymp B$ if $A\lesssim B\lesssim A$. The symbol ``$\log$'' denotes the natural logarithm.

\noindent\textbf{Acronyms.} VD $=$ volume doubling; PI $=$ Poincar\'e inequality; PHI $=$ parabolic Harnack inequality; FK $=$ Faber--Krahn; LT $=$ long-time Dirichlet heat-kernel bound on balls; LSRW $=$ lazy simple random walk.

\begin{definition}[Heat-kernel constant]\label{def:G}
Under the assumption of quantitative homogenisation, the LSRW on $G$ satisfies
\[
  p_t(x,x)=\frac{\cG}{t}+O(t^{-1-\delta})
\]
as $t\to\infty$ through integers, for constants $\cG>0$ and $\delta>0$ independent of $x$. We call $\cG$ the \emph{heat-kernel constant} of $G$.
\end{definition}


\begin{remark}[Explicit form of $\cG$]\label{rem:G_formula}
In many standard \emph{aperiodic} cases, such as lazy random walks on $\mathbb{Z}^d$ or other walks for which the on-diagonal local limit theorem is genuinely pointwise in time, that asymptotic yields an explicit formula (see, e.g., \cite[Ch. 2]{LawlerLimic10}). In $d=2$:
\[
\cG = \frac{1}{2\pi\sqrt{\det\Sigma}}.
\]
For periodic but time-periodic walks, the same expression gives the coefficient governing the time-summed trace asymptotic after averaging over the period, rather than a literal limit of $t\,p_t(x,x)$. In disordered media where a homogenized covariance is known, $\Sigma$ plays the role of the effective diffusivity.
\end{remark}

This assumption is available in many standard aperiodic settings and, after adding laziness, in many periodic finite-range walks. In the general irregular setting it is best regarded as an explicit quantitative on-diagonal homogenisation hypothesis; see \Cref{rem:QH_validity} for discussion.

From the viewpoint of hypotheses, the weakest main theorem is the ratio-limit criterion \Cref{thm:TS_balls}. We state first the stronger quantitative metric-ball refinement because it is the form proved directly by the detailed lower and upper bounds in the body of the paper and it contains the sharpest remainder information available from the present method.

\begin{theorem}[Quantitative refinement on metric balls]\label{thm:main}
Let $G$ be an infinite, connected, bounded degree graph satisfying (VG(2)) and (PI), and let $\{G_n\}$ be an exhaustion by metric balls $G_n=B_{R_n}(x_0)$ with $N_n=|G_n|\to\infty$. Assume in addition the annulus-volume bound (AV), the quantitative on-diagonal asymptotic (QH) with heat-kernel constant $\cG>0$ and exponent $\delta>0$, and the long-time Dirichlet estimate (LT): there exists $C_{LT}<\infty$ such that for every metric ball $H=B_R(x_0)$, every integer $t\ge R^2$, and every $v\in H$,
\[
  p_t^H(v,v)\le \frac{C_{LT}}{|V(H)|} e^{-\lambda_1(H)t}.
\]
Then
\[
  \frac{Z_n(1)}{N_n\log N_n}\;\longrightarrow\;\cG.
\]
Moreover,
\[
  Z_n(1)\le \cG N_n\log N_n+O(N_n).
\]
\end{theorem}

\paragraph{Proof sketch.} The argument is an interior--boundary decomposition. Fix $\eta\in(0,1/2)$ and write $G_n=I_n\cup E_n$, where $I_n=\{x: d(x,\partial G_n)>R_n^{1-\eta}\}$ is the deep interior and $E_n$ is the boundary layer. For $x\in I_n$ the probability of exiting by time $T_n:=\lfloor R_n^{2(1-2\eta)}\rfloor$ is super-polynomially small, so $p_t^{G_n}(x,x)=p_t(x,x)+o(t^{-1})$ uniformly for integers $1\le t\le T_n$. Summing \textup{(QH)} then yields $2\cG(1-2\eta)\log R_n+O(1)$ for the interior Green function. The boundary layer is negligible by \textup{(AV)}. For the upper bound, the short-time contribution is again governed by the on-diagonal input, while the long-time contribution is controlled by \textup{(LT)} together with Faber--Krahn, producing only $O(1)$ per vertex. Aggregating over $G_n$ yields the quantitative refinement. The leading-order criterion under the weaker time-summed input is then extracted later in \Cref{thm:TS_balls}.

\begin{remark}[Mesoscopic return traces]\label{cor:heat_trace}
The same interior--boundary decomposition also yields a mesoscopic heat-trace asymptotic in the regime $1\ll t\ll R_n^2$, and after summation in time this produces a windowed Green-trace asymptotic; see \Cref{thm:mesoscopic_heat_trace} and \Cref{cor:mesoscopic_green_window} below for abstract-domain formulations under \textup{(QH)}.
\end{remark}

\begin{proposition}[Upper bounds under averaged on-diagonal control]\label{prop:weaker_remainder}
Assume (VG(2)), (PI), and the long-time Dirichlet estimate (LT). Suppose instead of (QH) we only have two-sided pointwise bounds $c_1 t^{-1}\le p_t(x,x)\le c_2 t^{-1}$ for integers $t\ge t_0$ and an averaged deviation: there exist $t_0,C<\infty$, $\alpha>0$ with
\[
  \Big| \frac{1}{N_n} \sum_{x\in G_n} \Big( t p_t(x,x) - \cG \Big) \Big| \le C (\log t)^{-\alpha}\qquad(t\in\mathbb N,\ t\ge t_0,\;\forall n).
\]
Then:
\begin{enumerate}
  \item If $\alpha>1$, $Z_n(1)\le\cG N_n \log N_n + O(N_n)$.
  \item If $\alpha=1$, $Z_n(1)\le\cG N_n \log N_n + O(N_n \log\log N_n)$.
\end{enumerate}
\emph{No universal bound is claimed for $\alpha<1$.} The summable $t^{-1-\delta}$ rate (QH) corresponds to $\alpha=\infty$ and removes the logarithmic loss.
\end{proposition}
\begin{proof}
Set $T:=\lfloor R_n^2\rfloor$ and define
\[
  a_t^{(n)}:=\frac1{N_n}\sum_{x\in G_n}\bigl(t p_t(x,x)-\cG\bigr),
  \qquad |a_t^{(n)}|\le C(\log t)^{-\alpha}\quad (t\in\mathbb N,\ t\ge t_0).
\]
Using $p_t^{G_n}(x,x)\le p_t(x,x)$ for integers $1\le t\le T$, we obtain
\begin{align*}
Z_n(1)
&\le \sum_{x\in G_n}\sum_{t=1}^{T} p_t(x,x)
   + \sum_{x\in G_n}\sum_{t>T} p_t^{G_n}(x,x) \\
&\le O(N_n) + N_n\sum_{t=t_0}^{T}\left(\frac{\cG}{t}+\frac{a_t^{(n)}}{t}\right)
   + \sum_{x\in G_n}\sum_{t>T} p_t^{G_n}(x,x).
\end{align*}
The harmonic sum gives $\sum_{t=t_0}^{T} t^{-1}=\log T+O(1)$. Moreover,
\[
\sum_{t=t_0}^{T}\frac{|a_t^{(n)}|}{t}
\le C\sum_{t=t_0}^{T}\frac{1}{t(\log t)^{\alpha}}
=
\begin{cases}
O(1), & \alpha>1,\\[2mm]
O(\log\log T), & \alpha=1.
\end{cases}
\]
Hence
\[
\sum_{x\in G_n}\sum_{t=1}^{T} p_t(x,x)
\le
\begin{cases}
\cG N_n\log T + O(N_n), & \alpha>1,\\[2mm]
\cG N_n\log T + O\bigl(N_n\log\log T\bigr), & \alpha=1.
\end{cases}
\]
For the tail, the same argument as in the proof of \Cref{lem:upper}, based on \Cref{prop:LT}, yields
\[
\sum_{x\in G_n}\sum_{t>T} p_t^{G_n}(x,x)=O(N_n).
\]
Finally, $\log T = 2\log R_n + O(1) = \log N_n + O(1)$ because $N_n\asymp R_n^2$. Substituting this into the preceding estimates proves both claims.
\end{proof}

\begin{remark}[Log--log loss without the long-time bound / quantitative rate]\label{rem:loglog}
If one replaces the spatially averaged long-time bound (LT) by the crude spectral estimate $p_t^{G_n}(x,x)\le e^{-\lambda_1 t}$, or if one replaces the summable $t^{-1-\delta}$ error in (QH) by only $O(t^{-1})$, then an extra $\log\log N_n$ term may appear in the aggregated contribution of large times or fluctuation sums, inflating the natural upper bound to $O(N_n \log\log N_n)$. This highlights the strength of the two quantitative inputs when one seeks sharp remainder control.
\end{remark}

\Cref{thm:TS_balls} gives the leading-order criterion, while \Cref{thm:main} is its quantitative refinement under the summable remainder hypothesis \Cref{ass:QH}. For Euclidean domains and tori, related zeta and Green-trace asymptotics have been studied in \cite{Colin85,FrankSabin11}. In the graph setting, closely related quantities such as Green traces and the Kirchhoff index are classical on regular graphs and electrical networks; see \cite{MizunoTachikawa03,KleinRandic93,LyonsPeres16}. The present contribution is to extract from that background a domain-level transfer principle for the critical graph setting, and then verify its hypotheses on an explicit nontrivial square-domain family.

\subsection{Examples}
The cleanest verified examples lie in periodic two-dimensional settings.

\begin{example}[The $\mathbb{Z}^2$ case]\label{rem:pi2}
Consider the standard lattice $\mathbb{Z}^2$. The LSRW (defined in \Cref{sec:prelim}) has a step distribution covariance matrix $\Sigma = \frac{1}{4} I_2$. Using \Cref{rem:G_formula}:
\[
\cG = \frac{1}{2\pi \sqrt{\det(\Sigma)}} = \frac{1}{2\pi (1/4)} = \frac{2}{\pi}.
\]
\Cref{thm:TS_balls} implies $\lim_{n \to \infty} Z_n(1) / (N_n \log N_n) = 2/\pi$.
For axis-parallel squares, an independent spectral computation yields the full asymptotic with the same constant; see \Cref{thm:Z2_square}.
%
\smallskip
\noindent\emph{Laziness convention.}
The value $\cG=2/\pi$ corresponds to the \emph{lazy} simple random walk with holding probability $1/2$ used throughout this paper (see \Cref{sec:prelim} and \Cref{rem:non-lazy}). For the \emph{non-lazy} simple random walk on $\mathbb{Z}^2$, the return probabilities are periodic rather than of the form $\cG/t$: one has $p_{2m+1}(0,0)=0$ and $p_{2m}(0,0)\sim 1/(\pi m)$. Summing over even times therefore still yields
\[
\sum_{t\le R^2} p_t(0,0)=\frac{1}{\pi}\log(R^2)+O(1)=\frac{2}{\pi}\log R+O(1),
\]
so the corresponding trace coefficient is $\lim_{n\to\infty} Z_n(1)/(N_n\log N_n)=1/\pi$.
\end{example}


\begin{example}[Periodic and model-dependent irregular structures]\label{ex:irregular}
Beyond $\mathbb{Z}^2$, the most transparent examples satisfying the standing assumptions are periodic settings in which the on-diagonal coefficient is uniform in space after our normalization. They include:
\begin{enumerate}
    \item \textbf{Vertex-transitive periodic graphs:} graphs with a cocompact $\mathbb{Z}^2$ action, for which the on-diagonal coefficient is automatically independent of the vertex.
    \item \textbf{Lazy periodic finite-range walks:} for instance the lazy versions of the king, triangular, and knight walks.
\end{enumerate}
If the non-lazy walk has a genuine time-periodic return asymptotic along residue classes, then the periodic variant \Cref{cor:periodic_balls} applies after identifying the summed coefficient $\cG_{\mathrm{per}}$ from \Cref{prop:periodic_TS}.
Random media such as uniformly elliptic random conductance models or supercritical percolation clusters motivate this line of questions, but verifying either \Cref{ass:QH} or the weaker time-summed input \Cref{ass:TS} there is model-dependent and lies outside the scope of the present paper. In periodic settings, the annulus-volume bound (AV) is standard; in less regular media it must be verified or imposed separately.
\end{example}


\subsection{Method and Structure}

The proof is entirely time-domain and is organized around an interior--boundary decomposition. Rather than applying Tauberian theorems to the spectral measure, we work directly with heat-kernel sums. This keeps the source of each term transparent: geometry controls exit probabilities and spectral gaps, the on-diagonal input controls the short-time mass, and the long-time Dirichlet estimate controls the tail.

For the lower bound, vertices deep in the interior do not feel the boundary up to the diffusive timescale, so the killed walk inherits the full-space return sum. For the upper bound, we split into short and long times. The short-time part is estimated by the full-space on-diagonal asymptotic, while the long-time contribution is controlled by \textup{(LT)} together with Faber--Krahn. This is precisely what avoids the $\log\log$ loss produced by cruder large-time estimates; see \Cref{rem:IU_necessity}. For the sharp $O(1)$ remainder in \Cref{thm:main} we use the quantitative homogenisation rate \Cref{ass:QH}; for the ratio limit alone, the weaker time-summed input \Cref{ass:TS} is sufficient.

\paragraph{Structure of the paper.}
\Cref{sec:prelim} records the assumptions and the analytic tools used in the proof, including the weaker time-summed input and a periodic residue-class criterion. \Cref{sec:decomposition} introduces the interior-boundary decomposition. \Cref{sec:lower_bound} and \Cref{sec:upper_bound} prove the lower and upper bounds under the quantitative pointwise hypothesis. \Cref{sec:discussion} then states the main ratio-limit criterion \Cref{thm:TS_balls}, records the quantitative refinement \Cref{thm:main}, proves a TS-based Green-window theorem together with sharper QH-based mesoscopic heat-trace and windowed Green-trace results, reciprocal spectral-mass and annulus-counting corollaries, and a spectral counting corollary, and formulates abstract exhaustion variants, including a quantitative one. The appendices briefly compare other growth regimes and then study square domains: first a general diagonal-symbol theorem, then a second-order square expansion, and finally an explicit axial-diagonal family containing $\mathbb Z^2$ and the king walk.


\section{Preliminaries and Analytic Tools}\label{sec:prelim}


\paragraph{Notation (reference).} All symbols were fixed in the central Notation block; we only introduce new ones explicitly when they appear.

Let $G = (V,E)$ be an infinite, connected graph with bounded maximum degree $\Delta < \infty$. We define metric balls as closed: $B_R(x) = \{y \in V : d_G(x,y) \leq R\}$.

\paragraph{Convention on constants, normalization, uniformity.}
Throughout, implicit constants depend only on the structural data \textbf{(VD, PI, $\Delta$)} and, when used, on \textbf{(QH)}$=(\cG,\delta,t_0,C_{QH})$ and \textbf{(LT)}. All $L^2$ norms are with respect to counting measure; the ground state $\phi_1$ is $L^2$-normalised; no volume renormalisation is performed. Unless stated otherwise, $O(1)$ terms and statements claimed "uniformly in $x$" are uniform across space.

\subsection{Geometric and Analytic Assumptions}

\begin{definition}[Quadratic Volume Growth (VG(2))]
$G$ has (uniform) quadratic volume growth if there exist $c_1, c_2 > 0$ such that for all $x \in V$ and $R \geq 1$,
\begin{equation}\label{eq:quad}
c_1 R^{2} \leq V(x,R) \leq c_2 R^{2}.
\end{equation}
\end{definition}
This implies the Volume Doubling (VD) property: $V(x,2R) \leq C_D V(x,R)$.

\begin{assumption}[Annulus-volume bound (AV)]\label{ass:AV}
There exists $C_{AV}<\infty$ such that for all $x\in V$, all $R\ge 1$ and all integers $w$ with $1\le w\le R$,
\[
  |B_R(x)\setminus B_{R-w}(x)|\le C_{AV}\,R\,w.
\]
\end{assumption}
\noindent This hypothesis is satisfied in the standard periodic examples considered later (e.g. $\mathbb Z^2$ and other $\mathbb Z^2$-periodic walks), and it is precisely what ensures that the boundary layer of thickness $o(R)$ occupies $o(|B_R|)$ vertices.

\begin{definition}[Poincaré Inequality (PI)]\label{def:PI}
$G$ satisfies a (scaled) Poincaré inequality if there exists $C_P > 0$ such that for any ball $B_R=B_R(x_0)$ and any function $f: V \to \R$,
\[
\sum_{x \in B_R} (f(x) - \bar{f}_{B_R})^2 \leq C_P R^2 \, \cE_{B_{2R}}(f,f),
\]
where $\bar{f}_{B_R}$ is the average of $f$ over $B_R$ (w.r.t. counting measure), and the local Dirichlet form $\cE_U(f,f)$ is defined as
\[
\cE_U(f,f) = \sum_{\substack{\{x,y\} \in E \\ x,y \in U}} (f(x)-f(y))^2.
\]
\end{definition}

\begin{remark}\label{rem:PI_equivalence}
Under the VD condition, this formulation of PI is equivalent to the local version (see \cite{HajlaszKoskela00}). The combination of VD and PI (VD+PI) is central to analysis on graphs (see \cite{GrigoryanTelcs12}).
\end{remark}

\paragraph{Standing assumptions.} Throughout the quantitative part of the paper we assume $G$ is infinite, connected, has bounded degree $\Delta < \infty$, satisfies VG(2) (and thus VD), PI, and the annulus-volume bound (AV) of \Cref{ass:AV}; the additional hypothesis (QH) is stated in \Cref{ass:QH}.

\subsection{Random Walk and the Analytic Framework}

We consider the \emph{lazy} simple random walk (LSRW) $(X_t)_{t \geq 0}$. This is a discrete-time Markov chain where at each step, the walk stays put with probability $1/2$ or moves to a uniformly chosen neighbor with probability $1/2$. The transition matrix is $P = \frac{1}{2}(I + P_{SRW})$, where $P_{SRW}(x,y) = 1/\deg(x)$ if $y \sim x$. The heat kernel is $p_t(x,y) = \Prob_x[X_t = y]$. The generator (Laplacian) is $\LL = I - P$.

Laziness ensures the walk is aperiodic, simplifying spectral analysis. It also relates the generators: $\LL_{\text{LSRW}} = \frac{1}{2} \LL_{\text{SRW}}$ (see \Cref{rem:non-lazy}).


\begin{remark}[Measures and Operators]\label{rem:measures_operators}
The LSRW is reversible with respect to the degree measure $m(x)=\deg(x)$. The Laplacian $\LL$ is self-adjoint on the Hilbert space $\ell^2(m)$. Due to the bounded-degree assumption ($\Delta < \infty$), the counting measure $|\cdot|$ and the degree measure $m$ are comparable: $m(A) \asymp |A|$. This allows seamless transition between analytic results formulated with respect to $m$ and geometric properties formulated with respect to $|\cdot|$; see \cite{Coulhon03} and the discussion leading to \Cref{prop:LT}.


On a finite subgraph $H$, the Dirichlet Laplacian $\LL_H$ acting on $\ell^2(m|_H)$ is similar to the operator acting on $\ell^2(|\cdot||_H)$ (via the transformation induced by the square root of the measure densities, $M^{1/2}$). Since the trace is similarity-invariant in finite dimensions, $Z_H(1) = \tr(\LL_H^{-1})$ is independent of the chosen (comparable) inner product.


We emphasize that we work with the probabilistic Laplacian $\LL=I-P$ (or its similar normalized form), not the combinatorial Laplacian $D-A$.


Unless otherwise stated, all heat-kernel quantities ($p_t$, $p_t^H$, etc.) are those of the LSRW probability kernel. We often suppress the graph index in the Green function (e.g., writing $G_H(v,v)$ instead of $G_{H}(v,v)$) when clear from context.
\end{remark}

The combination of VD and PI is fundamental:

\begin{theorem}[\cite{Delmotte99}]\label{thm:Delmotte}
The combination of VD and PI is equivalent to the Parabolic Harnack Inequality (PHI).
\end{theorem}

PHI provides strong regularity for solutions to the heat equation, which translates to precise estimates on the random walk.

\begin{remark}[Analytic implication chain]\label{rem:chain}
Under VD+PI one has PHI by Delmotte~\cite{Delmotte99}. PHI yields \emph{two-sided Gaussian bounds} for the heat kernel (see \cite{Delmotte99}; also \cite[§5.3]{Grigoryan09}), which in turn imply
\emph{maximal/exit} estimates (e.g.\ \cite[Thm.~5.5.3]{Grigoryan09}). The Faber–Krahn inequality (FK) follows under VD+PI (see \cite[Ch.~8]{Grigoryan09} and \cite{BarlowBass04}). These are the only consequences used directly in the proof. In regular settings one may also combine PHI with stronger boundary regularity to verify long-time Dirichlet bounds by other methods, but the theorem itself assumes the required long-time estimate as an input.
\end{remark}

\subsection{Heat Kernel Toolbox}
We summarize the crucial analytic tools.

\subsubsection{Maximal Inequality and Gaussian Bounds}


\begin{proposition}[Consequences of PHI, \cite{Delmotte99, Grigoryan09}]\label{prop:maximal}
Under VD+PI:
\begin{enumerate}

    \item (Gaussian Bounds) There exist $C_G, c_G > 0$ such that for every integer $t\ge 1$,
    \[
    p_t(x,y) \leq \frac{C_G}{\sqrt{V(x,\sqrt{t}) V(y,\sqrt{t})}} \exp\left(-c_G \frac{d(x,y)^2}{t}\right).
    \]
    Under the VG(2) assumption, this specializes further to, for $t\ge 1$,
    \[
    p_t(x,y) \lesssim t^{-1} \exp\left(-c_G \frac{d(x,y)^2}{t}\right).
    \]
    \item (Maximal Inequality) There exist $C_M,c_M > 0$ such that for any $v \in V$, any integer $t \geq 0$, and any $d \geq 1$,
    {\small
    \begin{equation}\label{eq:max_ineq}
    \Prob_v\left( \max_{0 \leq s \leq t} d_G(v,X_s) \geq d \right) \leq C_M \exp\left(-c_M \frac{d^2}{t+1}\right).
    \end{equation}}
\end{enumerate}
\end{proposition}
\begin{proof}[Note on derivation]
The Gaussian upper bounds are a direct consequence of PHI (see \cite{Delmotte99} for the general form). The maximal inequality (exit time estimate) follows from the Gaussian upper bounds via standard arguments involving chaining and union bounds (see, e.g., \cite[Theorem 5.5.3]{Grigoryan09} or arguments in \cite{Delmotte99}).
\end{proof}


\subsubsection{Quantitative On-Diagonal Heat-Kernel Asymptotics}
A crucial ingredient is the sharp, uniform homogenisation of the heat kernel. We state this as a formal assumption.

\begin{assumption}[Quantitative on-diagonal asymptotic (QH)]\label{ass:QH}
There exists a constant $\cG > 0$ (the heat-kernel constant), a rate exponent $\delta>0$, a time $t_0\in\N$, and a constant $C_{QH}<\infty$ such that, uniformly in $x\in V$ and for all integers $t\ge t_0$,
\begin{equation}\label{eq:return-prob}
  p_t(x,x) = \frac{\cG}{t} + r_t(x),\qquad |r_t(x)| \le C_{QH}\, t^{-1-\delta}.
\end{equation}
Equivalently, $|t\,p_t(x,x)-\cG|\le C_{QH} t^{-\delta}$. All constants are independent of $x$.
\end{assumption}


\begin{remark}[Context and literature for \Cref{ass:QH}]\label{rem:QH_validity}
VD+PI already give PHI and Gaussian heat-kernel control. To identify the constant by summation with an $O(1)$ remainder, however, we explicitly assume the stronger uniform estimate $t p_t(x,x)=\cG+O(t^{-\delta})$, equivalently $p_t(x,x)-\cG/t=O(t^{-1-\delta})$, which ensures that the error series $\sum_t (p_t(x,x)-\cG/t)$ converges uniformly in $x$. This hypothesis can be verified in many standard aperiodic settings and in periodic finite-range walks after adding laziness. In stochastic homogenisation problems the available local limit theorems are typically model-dependent, and translating them into the global, uniform-in-space remainder \eqref{eq:return-prob} may require additional work; we therefore do not assert such an implication here in full generality. For background on random conductance and percolation models, see \cite{Biskup11,AndresDeuschelSlowik19,Barlow04,ArmstrongDario18,GloriaOtto17,CroydonHambly21}. Without summable decay one typically sees a $\log\log$ loss in the corresponding upper bounds; see \Cref{prop:weaker_remainder} and \Cref{rem:loglog}.
\end{remark}


\begin{remark}[Weaker QH assumptions]\label{rem:weaker_QH}
If one assumes a weaker form of QH, e.g., $p_t(x,x) = \cG/t + o(1/t)$ uniformly in $x$ as $t\to\infty$ through integers, without a polynomial rate, then summation over integers $1\le t\le R$ yields $\cG \log R + o(\log R)$. This is sufficient to establish the leading order asymptotic $Z_n(1) \sim \cG N_n \log N_n$, but does not by itself provide sharp quantitative remainder bounds.

If the error is exactly $O(1/t)$, i.e., $|t p_t(x,x) - \cG| = O(1)$, the summation error $\sum r_t(x)$ could be $O(\log R)$, leading to significantly larger aggregate error terms. The polynomial rate $\delta>0$ in \Cref{ass:QH} is precisely what allows the error sum to be $O(1)$.
\end{remark}


We derive the uniform sum of the return probabilities. By \Cref{ass:QH}, for every integer $R\ge 2$ we split the sum:
\begin{align*}
\sum_{t=1}^{R} p_t(x,x) &= \sum_{t=1}^{t_0-1} p_t(x,x) + \sum_{t=t_0}^{R} \left(\frac{\cG}{t} + r_t(x)\right) \\
&= O(1) + \cG \sum_{t=t_0}^{R} \frac{1}{t} + \sum_{t=t_0}^{R} r_t(x).
\end{align*}
The harmonic sum is $\sum_{t=t_0}^{R} \frac{1}{t} = \log R + O(1)$.
Crucially, the error sum is bounded uniformly in $x$ because the rate is polynomially fast ($\delta>0$) and thus summable ($1+\delta > 1$):
\[
\left|\sum_{t=t_0}^{R} r_t(x)\right| \leq \sum_{t=t_0}^{\infty} |r_t(x)| \leq C_{QH} \sum_{t=t_0}^{\infty} t^{-1-\delta} < C'.
\]
The constant $C'$ is independent of $x$ due to the uniformity assumed in \Cref{ass:QH}.
Thus, for integers $R\ge 2$, we obtain the uniform asymptotic:
\begin{equation}\label{eq:return-sum}
\sum_{t=1}^{R} p_t(x,x) = \cG \log R + O(1), \qquad R \geq 2,
\end{equation}
where the $O(1)$ is \emph{uniform in $x$}. In particular, for integers $R\ge 2$, choosing $R^2$ as the diffusive timescale yields the "master" identity
\begin{equation}\label{eq:master-sum}
  \sum_{t=1}^{R^{2}} p_t(x,x)
  \;=\; 2\,\cG \log R + O(1),
\end{equation}
uniformly in $x$.

\begin{assumption}[Uniform time-summed on-diagonal asymptotic (TS)]\label{ass:TS}
There exists a constant $\cG>0$ such that
\[
  \sup_{x\in V}\left|\sum_{t=1}^{T} p_t(x,x)-\cG \log T\right|=o(\log T)
\]
as $T\to\infty$ through integers.
\end{assumption}

\begin{remark}[From (QH) to (TS)]\label{rem:QH_implies_TS}
The stronger hypothesis \Cref{ass:QH} implies \Cref{ass:TS} with the same constant $\cG$. Indeed, \eqref{eq:return-sum} gives the sharper estimate
\[
  \sup_{x\in V}\left|\sum_{t=1}^{T} p_t(x,x)-\cG \log T\right|=O(1).
\]
The time-summed condition \Cref{ass:TS} is therefore the natural leading-order input when one only seeks the ratio limit for $Z_n(1)/(N_n\log N_n)$.
\end{remark}

\begin{proposition}[Uniform local limit theorem implies \Cref{ass:TS}]\label{prop:ullt_TS}
Assume there exists $\cG>0$ such that
\[
  \sup_{x\in V}\bigl|t\,p_t(x,x)-\cG\bigr|\longrightarrow 0
\]
as $t\to\infty$ through integers. Then \Cref{ass:TS} holds with the same constant $\cG$.
\end{proposition}

\begin{proof}
Fix $\varepsilon>0$. Choose $t_\varepsilon$ so that
\[
  \sup_{x\in V}\bigl|t\,p_t(x,x)-\cG\bigr|\le \varepsilon
\]
for all integers $t\ge t_\varepsilon$. Then, uniformly in $x$ and for every $T\ge t_\varepsilon$,
\begin{align*}
\left|\sum_{t=1}^{T} p_t(x,x)-\cG \log T\right|
&\le \sum_{t=1}^{t_\varepsilon-1} p_t(x,x)
   + \left|\cG \sum_{t=t_\varepsilon}^{T}\frac1t-\cG\log T\right|
   + \sum_{t=t_\varepsilon}^{T}\frac{\varepsilon}{t} \\
&\le C_\varepsilon + \varepsilon \log T
\end{align*}
with $C_\varepsilon<\infty$ independent of $x$ and $T$. Dividing by $\log T$ and letting $T\to\infty$, then $\varepsilon\downarrow0$, proves \Cref{ass:TS}.
\end{proof}

\begin{proposition}[Periodic residue-class asymptotics imply \Cref{ass:TS}]\label{prop:periodic_TS}
Assume there exist an integer $q\ge 1$, constants $\gamma_0,\dots,\gamma_{q-1}\ge 0$, an exponent $\delta>0$, an integer $m_0\ge 1$, and a constant $C<\infty$ such that, uniformly in $x\in V$, for every residue class $r\in\{0,\dots,q-1\}$ and every integer $m\ge m_0$,
\[
  p_{qm+r}(x,x)=\frac{\gamma_r}{m}+\varepsilon_{m,r}(x),
  \qquad |\varepsilon_{m,r}(x)|\le C m^{-1-\delta}.
\]
Then \Cref{ass:TS} holds with
\[
  \cG_{\mathrm{per}}:=\sum_{r=0}^{q-1}\gamma_r.
\]
\end{proposition}

\begin{proof}
Fix $T\ge q m_0$. Write $T=qM+s$ with $M\ge m_0$ and $0\le s\le q-1$. For each residue class $r$, the number of terms $t\le T$ of the form $t=qm+r$ is $M+O(1)$, uniformly in $r$. Hence, uniformly in $x$,
\begin{align*}
\sum_{t=1}^{T} p_t(x,x)
&= O(1)+\sum_{r=0}^{q-1}\sum_{m=m_0}^{M}
\left(\frac{\gamma_r}{m}+\varepsilon_{m,r}(x)\right) \\
&= \left(\sum_{r=0}^{q-1}\gamma_r\right)\log M + O(1)
\end{align*}
because $\sum_{m\ge m_0} m^{-1-\delta}<\infty$. Since $\log M=\log T-\log q+O(T^{-1})$, we conclude that
\[
  \sum_{t=1}^{T} p_t(x,x)=\cG_{\mathrm{per}}\log T+O(1)
\]
uniformly in $x$, which is exactly \Cref{ass:TS}.
\end{proof}



\subsubsection{Dirichlet Problem, Faber--Krahn, and the Long-Time Input}
Let $H \subset G$ be a finite connected subgraph. The Dirichlet generator $\LL_H$ corresponds to the LSRW killed on leaving $V(H)$, with exit time $\tau_H = \inf\{t \geq 0 : X_t \notin V(H)\}$. Thus
\[
  p_t^H(x,y) = \Prob_x[X_t = y,\, t < \tau_H],\qquad
  G_H(x,y) = \sum_{t=0}^{\infty} p_t^H(x,y) = (\LL_H^{-1})(x,y),
\]
and $Z_H(1) = \sum_{v \in V(H)} G_H(v,v)$.

To control the long-time behaviour we use the standing Dirichlet estimate (LT). Since $\LL_H=I-P_H$, decay is governed by $(1-\lambda_1(H))^t$, and we repeatedly use $(1-\lambda_1(H))^t\le e^{-\lambda_1(H)t}$. We also record the standard Faber--Krahn bound.

\begin{proposition}[Faber--Krahn and the standing long-time input]\label{prop:LT}
Let $H=B_R(x_0)$ be a metric ball. Assume VG(2)+PI and, for item~\textup{(2)}, the long-time Dirichlet estimate \textup{(LT)} from \Cref{thm:main}. Then:
\begin{enumerate}
    \item \textbf{(Faber--Krahn)} There exists $c_{FK}>0$ such that $\lambda_1(H)\ge c_{FK}/R^{2}$.
    \item \textbf{(Long-time Dirichlet bound)} There exists $C_{LT}>0$ such that for every integer $t\ge R^{2}$,
    \begin{equation}\label{eq:LT}
        \sup_{v \in V(H)} p_t^H(v,v) \leq \frac{C_{LT}}{|V(H)|} e^{-\lambda_1(H) t}.
    \end{equation}
\end{enumerate}
The constant $c_{FK}$ depends only on the uniform geometric structural parameters \textup{(VG(2), PI,} $\Delta$), while item~\textup{(2)} is simply the standing long-time assumption.
\end{proposition}

\begin{proof}
Item~(1) is the standard Faber--Krahn estimate under VD+PI on metric balls; see \cite{Grigoryan09, BarlowBass04}. Item~(2) is precisely the assumed long-time Dirichlet estimate (LT).
\end{proof}

\section{Interior--Boundary Decomposition and Volume Estimates}\label{sec:decomposition}

We analyze the exhaustion by metric balls $G_n = B_{R_n}(x_0)$, with $N_n = |V(G_n)|\asymp R_n^2$ by VG(2). Fix $\eta \in (0, \tfrac{1}{2})$ and set $W_n = R_n^{1-\eta}$. Define the \emph{interior} $I_n$ and \emph{boundary layer} $E_n$ by
\begin{align*}
I_n &:= \{x \in V(G_n) : d_G(x, V \setminus V(G_n)) > W_n\}, \\
E_n &:= V(G_n) \setminus I_n.
\end{align*}

The relevant time window is $T_n:=\lfloor R_n^{2(1-2\eta)}\rfloor$. Since the typical displacement on this scale is $R_n^{1-2\eta}$ while $W_n=R_n^{1-\eta}$, one has $R_n^{1-2\eta}/W_n=R_n^{-\eta}\to0$. Thus vertices in $I_n$ behave like bulk points up to time $T_n$, while the boundary layer remains negligible.

\begin{lemma}[Boundary Layer Volume]\label{lem:boundary_volume}
Assume (AV). For the choice $W_n = R_n^{1-\eta}$, we have
\[
    |E_n| = O\bigl(N_n^{1-\eta/2}\bigr).
\]
\end{lemma}
\begin{proof}
Set $R=R_n$ and $W=W_n=R^{1-\eta}$. If $v\in E_n$, there exists $y\notin B_R(x_0)$ with $d(v,y)\le W$, hence
$d(x_0,v)\ge R-W$ by the triangle inequality; thus $E_n\subseteq B_R(x_0)\setminus B_{R-W}(x_0)$.
We apply (AV) with $w=\lceil W\rceil$ to obtain
\[
  |E_n|\le |B_R(x_0)\setminus B_{R-W}(x_0)|\le C_{AV} R\,\lceil W\rceil\lesssim R W.
\]
With $W=R^{1-\eta}$ we obtain $|E_n|\lesssim R^{2-\eta}$. Since VG(2) gives $N_n\asymp R^2$, this yields $|E_n|=O(N_n^{1-\eta/2})$.
\end{proof}

\section{Lower Bound Analysis}\label{sec:lower_bound}

\begin{lemma}\label{lem:lower}
For the fixed $\eta \in (0,1/2)$, there exists a constant $C_1 > 0$ such that for all $v \in I_n$,
\[
G_{G_n}(v,v) \geq 2\cG(1-2\eta)\log R_n - C_1.
\]
\end{lemma}

\begin{proof}
Let $\tau_{\partial} = \min\{t \geq 0 : X_t \notin V(G_n)\}$ be the exit time. We set the time horizon $T = \lfloor R_n^{2(1-2\eta)} \rfloor$. Since $\eta < 1/2$, $T \to \infty$.
\[
G_{G_n}(v,v) \geq \sum_{t=1}^{T} p_t^{G_n}(v,v).
\]
\[
  p_t^{G_n}(v,v)\ge p_t(v,v)-\Prob_v(\tau_{\partial}\le t),
\]
and hence
\begin{equation}\label{eq:lower_decomp}
\sum_{t=1}^{T} p_t^{G_n}(v,v) \geq \sum_{t=1}^{T} p_t(v,v) - \sum_{t=1}^{T} \Prob_v(\tau_{\partial} \leq t).
\end{equation}

\begin{align*}
\sum_{t=1}^{T} p_t(v,v) &= \cG \log T + O(1)
  = \cG \log(R_n^{2(1-2\eta)}) + O(1) \\
&= 2\cG (1-2\eta)\log R_n + O(1).
\end{align*}

Let $D = d_G(v, V \setminus V(G_n))$. Since $v \in I_n$, $D > R_n^{1-\eta}$. By \Cref{prop:maximal}, for every integer $1\le t \leq T$,
\[
\Prob_v(\tau_{\partial} \leq t) \leq \Prob_v\bigl( \max_{0 \leq s \leq t} d_G(v,X_s) \geq D \bigr) \leq C_M \exp\Bigl(-c_M \frac{D^2}{t+1}\Bigr).
\]
Since $T+1 \leq 2 R_n^{2(1-2\eta)}$,
\[
c_M \frac{D^2}{T+1} \geq c_M \frac{R_n^{2(1-\eta)}}{2 R_n^{2(1-2\eta)}} = \frac{c_M}{2} R_n^{2\eta}.
\]
Let $c' = c_M/2$. The total error term is bounded by:
\[
\sum_{t=1}^T \Prob_v(\tau_{\partial} \leq t) \leq (T+1) C_M \exp(-c' R_n^{2\eta}).
\]
Since $T = O(R_n^2)$ and $\eta>0$, this is $o(1)$. Combining with \eqref{eq:lower_decomp} proves the lemma.
\end{proof}

\begin{corollary}\label{cor:lower}
For any $\eta \in (0,1/2)$, there exists $C_2 > 0$ such that
\[
Z_n(1) \geq \cG(1-2\eta) N_n \log N_n - C_2 N_n.
\]
\end{corollary}

\begin{proof}
We sum the bound of \Cref{lem:lower} over the interior $I_n$.
\[
Z_n(1) \geq \sum_{v \in I_n} G_{G_n}(v,v) \geq |I_n|\Bigl[2\cG(1-2\eta)\log R_n - C_1\Bigr].
\]
Since $N_n \asymp R_n^2$, one has $\log N_n = 2\log R_n + O(1)$, and by \Cref{lem:boundary_volume},
\[
  |I_n| = N_n - O(N_n^{1-\eta/2}).
\]
\begin{align*}
Z_n(1) &\geq \bigl[N_n - O(N_n^{1-\eta/2})\bigr]
          \Bigl[\cG(1-2\eta)\bigl(\log N_n + O(1)\bigr) - C_1\Bigr] \\
&= \cG(1-2\eta) N_n \log N_n - O\bigl(N_n^{1-\eta/2}\log N_n\bigr) - O(N_n) \\
&= \cG(1-2\eta) N_n \log N_n - O(N_n).\qedhere
\end{align*}
\end{proof}

\section{Upper Bound Analysis}\label{sec:upper_bound}

The upper bound requires uniform control over the Green function, including vertices near the boundary. This is where the long-time Dirichlet heat-kernel hypothesis (LT) enters.

\begin{lemma}\label{lem:upper}
There exists a constant $C_3 > 0$ such that for any $v \in V(G_n)$,
\[
G_{G_n}(v,v) \leq 2\cG \log R_n + C_3.
\]
\end{lemma}

\begin{proof}
Let $v \in V(G_n)$. Extracting the initial $t=0$ term $p_0^{G_n}(v,v)=1$, we split the remaining Green function sum at the characteristic diffusive time scale $T = \lfloor R_n^2 \rfloor$:
\[
G_{G_n}(v,v) = 1 + \sum_{t=1}^{T} p_t^{G_n}(v,v) + \sum_{t > T} p_t^{G_n}(v,v).
\]

\begin{align*}
\sum_{t=1}^{T} p_t^{G_n}(v,v) &\leq \sum_{t=1}^{T} p_t(v,v) = \cG \log T + O(1) \\
&= \cG \log(R_n^2) + O(1) = 2\cG \log R_n + O(1).
\end{align*}

Let $\lambda_1 = \lambda_1(G_n)$. By \Cref{prop:LT}, for every integer $t>T\ge R_n^2-1$,
\begin{equation}
p_t^{G_n}(v,v) \leq \frac{C_{LT}}{N_n} e^{-\lambda_1 t}.
\end{equation}

\[
S := \sum_{t > T} p_t^{G_n}(v,v)
\]
therefore satisfies, with $r:=e^{-\lambda_1}$,
\[
S \leq \frac{C_{LT}}{N_n} \sum_{t=T+1}^\infty r^t = \frac{C_{LT}}{N_n} \frac{r^{T+1}}{1-r}.
\]

By Faber--Krahn, $\lambda_1 \geq c_{FK}/R_n^2$. Since $T+1 > R_n^2$,
\[
  r^{T+1} \leq \exp(-\lambda_1(T+1)) \leq e^{-c_{FK}}.
\]
Also $0<\lambda_1\le 1$, so $1-e^{-u}\ge e^{-1}u$ for $0<u\le 1$ yields
\[
1-r = 1-e^{-\lambda_1} \ge e^{-1}\lambda_1.
\]

\[
S \leq \frac{C_{LT}}{N_n} \frac{e^{-c_{FK}}}{e^{-1}\lambda_1} = \frac{e C_{LT} e^{-c_{FK}}}{N_n \lambda_1}.
\]
Since $N_n \asymp R_n^2$ and $\lambda_1 \gtrsim R_n^{-2}$, one has $S=O(1)$ uniformly in $v$. Therefore
\[
G_{G_n}(v,v) \leq 2\cG \log R_n + O(1),
\]
uniformly in $v$.
\end{proof}


\begin{remark}[On the sharpness of the upper bound and the long-time estimate]\label{rem:IU_necessity}
The long-time Dirichlet estimate (LT) yields the sharp $O(1)$ remainder in the pointwise bound because it supplies the spatial factor $1/N_n$ together with exponential decay. If one used only the crude pointwise spectral bound $p_t^{G_n}(v,v)\le (1-\lambda_1)^t\le e^{-\lambda_1 t}$, the tail sum would be $\sum_{t>T} e^{-\lambda_1 t} \approx 1/\lambda_1 \asymp R_n^2$. To make this tail $O(1)$, one would need to choose a much longer time scale $T \gtrsim R_n^2 \log R_n$. This would increase the short-time sum by $\cG \log(R_n^2 \log R_n) - \cG \log(R_n^2) = \cG \log\log R_n$, thereby introducing a $\log\log$ term into the corresponding upper bounds.
\end{remark}

\begin{corollary}\label{cor:upper}
There exists $C_4 > 0$ such that
\[
Z_n(1) \leq \cG N_n \log N_n + C_4 N_n.
\]
\end{corollary}

\begin{proof}
Since $N_n \asymp R_n^2$, one has $2\log R_n = \log N_n + O(1)$. Substituting into \Cref{lem:upper} gives
\begin{align*}
G_{G_n}(v,v) &\leq 2\cG \log R_n + C_3 = \cG (\log N_n + O(1)) + C_3 = \cG \log N_n + O(1).
\end{align*}
Summing this uniform bound over all $v \in V(G_n)$ gives the result.
\end{proof}

\section{Criterion, Consequences, and Abstract Exhaustions}\label{sec:discussion}

This section records the main consequences of the time-domain argument. We begin with the leading-order statement on metric balls and its quantitative refinement, then turn to mesoscopic spectral consequences under quantitative on-diagonal input, and finally state the corresponding results for abstract exhaustions. The appendix treats the square-domain case, where the long-time and mesoscopic inputs can be proved directly for an explicit family.

\begin{proof}[Proof of \Cref{thm:main}]
The upper bound $Z_n(1)\le \cG N_n\log N_n + C_4 N_n$ is exactly \Cref{cor:upper}. It remains to identify the ratio limit by combining the lower and upper asymptotic bounds.


Let $\eta \in (0,1/2)$ be arbitrary. By \Cref{cor:lower}, the lower asymptotic bound is:
\begin{align*}
\liminf_{n \to \infty} \frac{Z_n(1)}{N_n \log N_n} &\geq \lim_{n \to \infty} \left( \cG(1-2\eta) - \frac{C_2}{\log N_n} \right) = \cG(1-2\eta).
\end{align*}

By \Cref{cor:upper}, the upper asymptotic bound is:
\begin{align*}
\limsup_{n \to \infty} \frac{Z_n(1)}{N_n \log N_n} &\leq \lim_{n \to \infty} \left( \cG + \frac{C_4}{\log N_n} \right) = \cG.
\end{align*}

Combining the two bounds we obtain
\[
\cG(1-2\eta) \leq \liminf_{n \to \infty} \frac{Z_n(1)}{N_n \log N_n} \leq \limsup_{n \to \infty} \frac{Z_n(1)}{N_n \log N_n} \leq \cG.
\]
Since $\eta > 0$ can be chosen arbitrarily small, we conclude that
\[
\lim_{n \to \infty} \frac{Z_n(1)}{N_n \log N_n} = \cG.
\]
\end{proof}

\subsection{Leading-Order Criteria on Balls}

\begin{theorem}[Leading-order criterion on metric balls]\label{thm:TS_balls}
Let $G$ be an infinite, connected, bounded degree graph satisfying \textup{(VG(2))} and \textup{(PI)}, and let $G_n=B_{R_n}(x_0)$ be an exhaustion by metric balls with $N_n=|G_n|\to\infty$. Assume in addition \textup{(AV)}, the time-summed on-diagonal asymptotic \textup{(TS)} with constant $\cG>0$, and the long-time Dirichlet bound \eqref{eq:LT} on metric balls. Then
\[
  \frac{Z_n(1)}{N_n\log N_n}\longrightarrow \cG.
\]
\end{theorem}

\begin{proof}
Fix $\eta\in(0,1/2)$ and set $T_n=\lfloor R_n^{2(1-2\eta)}\rfloor$. For $v\in I_n$, the same exit-probability estimate as in the proof of \Cref{lem:lower} gives
\[
  G_{G_n}(v,v)\ge \sum_{t=1}^{T_n}p_t(v,v)-o(1).
\]
By \Cref{ass:TS},
\[
  \sum_{t=1}^{T_n}p_t(v,v)=\cG\log T_n+o(\log T_n)
  =2\cG(1-2\eta)\log R_n+o(\log R_n),
\]
uniformly in $v$. Summing over $I_n$ and using \Cref{lem:boundary_volume} yields
\[
  \liminf_{n\to\infty}\frac{Z_n(1)}{N_n\log N_n}\ge \cG(1-2\eta).
\]
For the upper bound, \Cref{ass:TS} gives uniformly in $v\in G_n$,
\[
  \sum_{t=1}^{\lfloor R_n^2\rfloor} p_t(v,v)=\cG\log(R_n^2)+o(\log R_n)=2\cG\log R_n+o(\log R_n),
\]
while the long-time tail is still $O(1)$ by the proof of \Cref{lem:upper}. Hence
\[
  G_{G_n}(v,v)\le 2\cG\log R_n+o(\log R_n),
\]
uniformly in $v$, and therefore
\[
  \limsup_{n\to\infty}\frac{Z_n(1)}{N_n\log N_n}\le \cG.
\]
Letting $\eta\downarrow 0$ completes the proof.
\end{proof}

\begin{corollary}[Periodic oscillations are admissible through time summation]\label{cor:periodic_balls}
Under the geometric assumptions of \Cref{thm:TS_balls}, suppose instead of \textup{(QH)} that the residue-class asymptotic in \Cref{prop:periodic_TS} holds and that \textup{(LT)} is available on the same metric balls. Then
\[
  \frac{Z_n(1)}{N_n\log N_n}\longrightarrow \cG_{\mathrm{per}}
  :=\sum_{r=0}^{q-1}\gamma_r.
\]
\end{corollary}

\begin{proof}
By \Cref{prop:periodic_TS}, the residue-class hypothesis implies \Cref{ass:TS} with constant $\cG_{\mathrm{per}}$. The conclusion therefore follows immediately from \Cref{thm:TS_balls}.
\end{proof}

\begin{corollary}[Uniform pointwise $t^{-1}$ asymptotics suffice for the ratio limit]\label{cor:ullt_ratio}
Under the geometric assumptions of \Cref{thm:TS_balls}, assume
\[
  \sup_{x\in V}\bigl|t\,p_t(x,x)-\cG\bigr|\longrightarrow 0
\]
as $t\to\infty$ through integers, and assume \textup{(LT)} on the same metric balls. Then
\[
  \frac{Z_n(1)}{N_n\log N_n}\longrightarrow \cG.
\]
\end{corollary}

\begin{proof}
By \Cref{prop:ullt_TS}, the uniform pointwise asymptotic implies \Cref{ass:TS}. The claim is therefore a direct application of \Cref{thm:TS_balls}.
\end{proof}

\subsection{Mesoscopic Spectral Consequences}

\begin{theorem}[Macroscopic Green-window asymptotic under \textup{(TS)}]\label{thm:TS_window}
Let $G$ be an infinite, connected, bounded degree graph satisfying \textup{(VG(2))}, \textup{(PI)}, and \Cref{ass:TS} with constant $\cG>0$. Let $(H_n)$ be finite connected domains in $G$ with $N_n:=|H_n|\to\infty$. Assume there exists a scale $\rho_n\to\infty$ such that $N_n\asymp \rho_n^2$ and, for every fixed $\eta\in(0,1/2)$, the boundary layer
\[
  E_n^{(\eta)}:=\{x\in H_n:\ d(x,V\setminus H_n)\le \rho_n^{1-\eta}\}
\]
satisfies $|E_n^{(\eta)}|=o(N_n)$. Fix $\eta\in(0,1/2)$ and let $(a_n)$ and $(b_n)$ be integer sequences such that
\[
  2\le a_n\le b_n\le \rho_n^{2(1-2\eta)}
\]
for all sufficiently large $n$, and assume moreover that
\[
  \liminf_{n\to\infty}\frac{\log(b_n/a_n)}{\log \rho_n}>0.
\]
Then
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  \sim
  \cG N_n\log\frac{b_n}{a_n}.
\]
Equivalently,
\[
  \sum_k
  \frac{(1-\lambda_k(H_n))^{a_n}-(1-\lambda_k(H_n))^{b_n+1}}
       {\lambda_k(H_n)}
  \sim
  \cG N_n\log\frac{b_n}{a_n},
\]
where $(\lambda_k(H_n))$ are the Dirichlet eigenvalues of $\LL_{H_n}$.
\end{theorem}

\begin{proof}
Set $E_n:=E_n^{(\eta)}$ and $I_n:=H_n\setminus E_n$. For $x\in I_n$, the distance to the complement exceeds $\rho_n^{1-\eta}$, while $b_n\le \rho_n^{2(1-2\eta)}$. The maximal inequality from \Cref{prop:maximal} therefore yields
\[
  \sup_{1\le t\le b_n}\Prob_x(\tau_{H_n}\le t)\le C e^{-c\rho_n^{2\eta}}
\]
for some $c>0$, uniformly in $x\in I_n$. Hence
\[
  \sum_{t=a_n}^{b_n} p_t^{H_n}(x,x)
  \ge
  \sum_{t=a_n}^{b_n} p_t(x,x)-b_n C e^{-c\rho_n^{2\eta}}
  =
  \sum_{t=a_n}^{b_n} p_t(x,x)+o(1)
\]
uniformly in $x\in I_n$, since $b_n$ grows at most polynomially in $\rho_n$.

Write
\[
  F_x(T):=\sum_{t=1}^{T} p_t(x,x).
\]
By \Cref{ass:TS},
\[
  F_x(T)=\cG\log T+o(\log T)
\]
uniformly in $x$. Since $a_n,b_n\le \rho_n^{2(1-2\eta)}$, both $\log a_n$ and $\log b_n$ are $O(\log \rho_n)$. Therefore
\[
  \sum_{t=a_n}^{b_n} p_t(x,x)
  =
  F_x(b_n)-F_x(a_n-1)
  =
  \cG\log\frac{b_n}{a_n-1}+o(\log \rho_n)
\]
uniformly in $x$. Because $a_n\ge 2$, one has $\log(a_n/(a_n-1))\le \log 2=O(1)$, while $\log(b_n/a_n)\ge \kappa\log \rho_n\to\infty$ for some $\kappa>0$ and all large $n$,
\[
  \log\frac{b_n}{a_n-1}
  =
  \log\frac{b_n}{a_n}+O(1)
\]
and hence
\[
  \sum_{t=a_n}^{b_n} p_t(x,x)
  =
  \cG\log\frac{b_n}{a_n}+o\!\left(\log\frac{b_n}{a_n}\right)
\]
uniformly in $x$.

Summing over $x\in I_n$ gives
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  \ge
  |I_n|
  \left(
    \cG\log\frac{b_n}{a_n}
    + o\!\left(\log\frac{b_n}{a_n}\right)
  \right)
  + o(N_n).
\]
Since $|I_n|=N_n-o(N_n)$, this yields
\[
  \liminf_{n\to\infty}
  \frac{\sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)}
       {N_n\log(b_n/a_n)}
  \ge \cG.
\]

For the upper bound, $p_t^{H_n}(x,x)\le p_t(x,x)$ gives
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  \le
  N_n
  \left(
    \cG\log\frac{b_n}{a_n}
    + o\!\left(\log\frac{b_n}{a_n}\right)
  \right),
\]
whence
\[
  \limsup_{n\to\infty}
  \frac{\sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)}
       {N_n\log(b_n/a_n)}
  \le \cG.
\]
This proves the theorem.
\end{proof}

\begin{corollary}[Polynomial Green windows]\label{cor:TS_window_polynomial}
Under the assumptions of \Cref{thm:TS_window}, fix exponents
\[
  0<\alpha<\beta<1-2\eta.
\]
If
\[
  a_n=\lceil \rho_n^{2\alpha}\rceil,
  \qquad
  b_n=\lfloor \rho_n^{2\beta}\rfloor,
\]
then
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  \sim
  2\cG(\beta-\alpha)N_n\log \rho_n.
\]
Equivalently,
\[
  \sum_k
  \frac{(1-\lambda_k(H_n))^{a_n}-(1-\lambda_k(H_n))^{b_n+1}}
       {\lambda_k(H_n)}
  \sim
  2\cG(\beta-\alpha)N_n\log \rho_n.
\]
\end{corollary}

\begin{proof}
One has
\[
  \log\frac{b_n}{a_n}
  =
  2(\beta-\alpha)\log \rho_n+o(\log \rho_n).
\]
The claim therefore follows immediately from \Cref{thm:TS_window}.
\end{proof}

\begin{corollary}[Logarithmic-time density law]\label{cor:TS_log_density}
Under the assumptions of \Cref{thm:TS_window}, define finite measures on $(0,1-2\eta)$ by
\[
  \nu_n^{(\eta)}
  :=
  \frac{1}{N_n\log\rho_n}
  \sum_{t=2}^{\lfloor \rho_n^{2(1-2\eta)}\rfloor}
  \left(\sum_{x\in H_n} p_t^{H_n}(x,x)\right)
  \delta_{\frac{\log t}{2\log\rho_n}}.
\]
Then for every compact interval $[\alpha,\beta]\subset(0,1-2\eta)$,
\[
  \nu_n^{(\eta)}([\alpha,\beta])\longrightarrow 2\cG(\beta-\alpha).
\]
In particular, $\nu_n^{(\eta)}$ converges vaguely on $(0,1-2\eta)$ to the absolutely continuous measure with density $2\cG$.
\end{corollary}

\begin{proof}
Fix $0<\alpha<\beta<1-2\eta$. Then
\[
  \nu_n^{(\eta)}([\alpha,\beta])
  =
  \frac{1}{N_n\log\rho_n}
  \sum_{t=\lceil \rho_n^{2\alpha}\rceil}^{\lfloor \rho_n^{2\beta}\rfloor}
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  + o(1),
\]
because the boundary terms at the endpoints involve only $O(1)$ times and each heat trace is at most $N_n$. The claim therefore follows from \Cref{cor:TS_window_polynomial}. The vague convergence statement is immediate from convergence on compact intervals.
\end{proof}

\begin{theorem}[Uniform mesoscopic heat-trace asymptotic]\label{thm:mesoscopic_heat_trace}
Let $G$ be an infinite, connected, bounded degree graph satisfying \textup{(VG(2))}, \textup{(PI)}, and \textup{(QH)} with constant $\cG>0$ and exponent $\delta>0$. Let $(H_n)$ be finite connected domains in $G$ with $N_n:=|H_n|\to\infty$. Assume there exists a scale $\rho_n\to\infty$ such that $N_n\asymp \rho_n^2$ and, for every fixed $\eta\in(0,1/2)$, the boundary layer
\[
  E_n^{(\eta)}:=\{x\in H_n:\ d(x,V\setminus H_n)\le \rho_n^{1-\eta}\}
\]
satisfies $|E_n^{(\eta)}|=o(N_n)$. Fix $\eta\in(0,1/2)$ and let $(a_n)$ and $(b_n)$ be integer sequences such that
\[
  a_n\to\infty,
  \qquad
  a_n\le b_n\le \rho_n^{2(1-2\eta)}
\]
for all sufficiently large $n$, and suppose moreover that $b_n/a_n=O(1)$. Then
\[
  \sup_{a_n\le t\le b_n}
  \frac{t}{N_n}
  \left|
    \sum_{x\in H_n} p_t^{H_n}(x,x)-\frac{\cG N_n}{t}
  \right|
  \longrightarrow 0.
\]
\end{theorem}

\begin{proof}
Set $I_n^{(\eta)}:=H_n\setminus E_n^{(\eta)}$. For $x\in I_n^{(\eta)}$, the distance to the complement is greater than $\rho_n^{1-\eta}$. Since $b_n\le \rho_n^{2(1-2\eta)}$ for large $n$, the maximal inequality from \Cref{prop:maximal} gives
\[
  \sup_{a_n\le t\le b_n}\Prob_x(\tau_{H_n}\le t)\le C_M \exp(-c\rho_n^{2\eta})
\]
uniformly in $x\in I_n^{(\eta)}$, for some $c>0$. Therefore
\[
  p_t^{H_n}(x,x)
  \ge p_t(x,x)-C_M e^{-c\rho_n^{2\eta}}
\]
for every $t\in[a_n,b_n]$, while the trivial domination gives $p_t^{H_n}(x,x)\le p_t(x,x)$ for all $x\in H_n$.

By \textup{(QH)},
\[
  p_t(x,x)=\frac{\cG}{t}+O(t^{-1-\delta})
\]
uniformly in $x$ and in integers $t\ge a_n$ for all sufficiently large $n$. Since $t\ge a_n$ on the chosen window, this gives
\[
  p_t(x,x)=\frac{\cG}{t}+O(a_n^{-1-\delta})
\]
uniformly in $x$ and in $t\in[a_n,b_n]$.

Summing over the interior vertices yields
\[
  \sum_{x\in I_n^{(\eta)}} p_t^{H_n}(x,x)
  \ge
  |I_n^{(\eta)}|
  \left(
    \frac{\cG}{t}
    + O(a_n^{-1-\delta})
    - O(e^{-c\rho_n^{2\eta}})
  \right).
\]
Since $|I_n^{(\eta)}|=N_n-o(N_n)$, uniformly for $t\in[a_n,b_n]$ we obtain
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  \ge
  \frac{\cG N_n}{t}
  - \frac{\cG |E_n^{(\eta)}|}{t}
  + O(N_n a_n^{-1-\delta})
  - O(N_n e^{-c\rho_n^{2\eta}}).
\]
Multiplying by $t/N_n$, the boundary term becomes $o(1)$ because $|E_n^{(\eta)}|=o(N_n)$. The summable remainder contributes
\[
  \frac{t}{N_n} O(N_n a_n^{-1-\delta}) = O\!\left(\frac{t}{a_n^{1+\delta}}\right)=o(1)
\]
uniformly on $[a_n,b_n]$ since $t\le b_n=O(a_n)$. The exponential term is also $o(1)$ after multiplication by $t/N_n$, because $t\le b_n\le \rho_n^{2(1-2\eta)}$ grows at most polynomially while $e^{-c\rho_n^{2\eta}}$ decays super-polynomially.

For the upper bound, \textup{(QH)} and $p_t^{H_n}(x,x)\le p_t(x,x)$ give
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  \le
  N_n\left(\frac{\cG}{t}+O(a_n^{-1-\delta})\right)
\]
uniformly for $t\in[a_n,b_n]$. After multiplication by $t/N_n$, the error is again $o(1)$ because $t=O(a_n)$. Combining the lower and upper bounds proves the theorem.
\end{proof}

\begin{proposition}[Integrated mesoscopic heat trace on arbitrary subdiffusive windows]\label{prop:integrated_mesoscopic_heat_trace}
Let $G$ and $(H_n)$ satisfy the assumptions of \Cref{thm:mesoscopic_heat_trace}, except that the bounded-aspect-ratio condition $b_n/a_n=O(1)$ is not imposed. Then there exists $c>0$ such that
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  =
  \cG N_n\sum_{t=a_n}^{b_n}\frac1t
  +
  O\!\left(
    |E_n^{(\eta)}|\sum_{t=a_n}^{b_n}\frac1t
    + N_n a_n^{-\delta}
    + N_n b_n e^{-c\rho_n^{2\eta}}
  \right).
\]
\end{proposition}

\begin{proof}
Set $E_n:=E_n^{(\eta)}$ and $I_n:=H_n\setminus E_n$. As in the proof of \Cref{thm:mesoscopic_heat_trace}, there exists $c>0$ such that for all sufficiently large $n$, all $x\in I_n$, and all integers $t\in[a_n,b_n]$,
\[
  p_t^{H_n}(x,x)\ge p_t(x,x)-C e^{-c\rho_n^{2\eta}},
\]
while always $p_t^{H_n}(x,x)\le p_t(x,x)$. By \textup{(QH)},
\[
  p_t(x,x)=\frac{\cG}{t}+O(t^{-1-\delta})
\]
uniformly in $x$ and $t\ge a_n$. Therefore, uniformly on $[a_n,b_n]$,
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  \le
  \frac{\cG N_n}{t}+O(N_n t^{-1-\delta}),
\]
and
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  \ge
  \frac{\cG N_n}{t}
  - \frac{\cG |E_n|}{t}
  + O(N_n t^{-1-\delta})
  - O(N_n e^{-c\rho_n^{2\eta}}).
\]
Summing over $t=a_n,\dots,b_n$ gives
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  \le
  \cG N_n\sum_{t=a_n}^{b_n}\frac1t
  + O\!\left(N_n\sum_{t=a_n}^{b_n} t^{-1-\delta}\right),
\]
and
\begin{align*}
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  &\ge
  \cG N_n\sum_{t=a_n}^{b_n}\frac1t
  - \cG |E_n|\sum_{t=a_n}^{b_n}\frac1t \\
  &\quad
  + O\!\left(N_n\sum_{t=a_n}^{b_n} t^{-1-\delta}\right)
  - O\!\left(N_n b_n e^{-c\rho_n^{2\eta}}\right).
\end{align*}
Since $\sum_{t=a_n}^{b_n} t^{-1-\delta}=O(a_n^{-\delta})$, the claim follows.
\end{proof}

\begin{corollary}[Mesoscopic Green-window asymptotic]\label{cor:mesoscopic_green_window}
Under the assumptions of \Cref{prop:integrated_mesoscopic_heat_trace}, assume moreover that
\[
  |E_n^{(\eta)}|\sum_{t=a_n}^{b_n}\frac1t=o(N_n).
\]
Then
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  =
  \cG N_n\sum_{t=a_n}^{b_n}\frac1t+o(N_n)
\]
and hence
\[
  \sum_{t=a_n}^{b_n}\sum_{x\in H_n} p_t^{H_n}(x,x)
  =
  \cG N_n \log\frac{b_n}{a_n}+o(N_n).
\]
Equivalently,
\[
  \sum_k
  \frac{(1-\lambda_k(H_n))^{a_n}-(1-\lambda_k(H_n))^{b_n+1}}
       {\lambda_k(H_n)}
  =
  \cG N_n \log\frac{b_n}{a_n}+o(N_n),
\]
where $(\lambda_k(H_n))$ are the Dirichlet eigenvalues of $\LL_{H_n}$.
\end{corollary}

\begin{proof}
The first identity is immediate from \Cref{prop:integrated_mesoscopic_heat_trace}, because $a_n^{-\delta}\to0$ and $b_n e^{-c\rho_n^{2\eta}}=o(1)$ as $b_n\le \rho_n^{2(1-2\eta)}$. Also
\[
  \sum_{t=a_n}^{b_n}\frac1t
  =
  \log\frac{b_n}{a_n}+O(a_n^{-1})
  =
  \log\frac{b_n}{a_n}+o(1)
\]
because $a_n\to\infty$. This proves the first identity.

For the spectral form, use
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)=\sum_k (1-\lambda_k(H_n))^t
\]
and sum the resulting geometric series:
\[
  \sum_{t=a_n}^{b_n}(1-\lambda_k(H_n))^t
  =
  \frac{(1-\lambda_k(H_n))^{a_n}-(1-\lambda_k(H_n))^{b_n+1}}
       {\lambda_k(H_n)}.
\]
Summing over $k$ gives the claimed equivalent formulation.
\end{proof}

\begin{corollary}[Mesoscopic spectral localization]\label{cor:mesoscopic_counting}
Under the assumptions of \Cref{thm:mesoscopic_heat_trace}, define
\[
  \mathcal N_{H_n}(\Lambda):=\#\{k:\lambda_k(H_n)\le \Lambda\}.
\]
Then:
\begin{enumerate}
  \item For every fixed $c>0$,
\[
  \sup_{a_n\le t\le b_n}\frac{t}{N_n}\,\mathcal N_{H_n}\!\left(\frac{c}{t}\right)
  \le e^c \cG + o(1).
\]
  \item For every fixed $K>1$,
\[
  \inf_{a_n\le t\le b_n}\frac{t}{N_n}\,\mathcal N_{H_n}\!\left(\frac{K\log t}{t}\right)
  \ge \cG + o(1).
\]
\end{enumerate}
\end{corollary}

\begin{proof}
For item~(1), let $t\in[a_n,b_n]$ and consider eigenvalues $\lambda_k(H_n)\le c/t$. Then
\[
  (1-\lambda_k(H_n))^t \ge \left(1-\frac{c}{t}\right)^t.
\]
Using the spectral representation
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)=\sum_k (1-\lambda_k(H_n))^t,
\]
we obtain
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  \ge
  \mathcal N_{H_n}\!\left(\frac{c}{t}\right)\left(1-\frac{c}{t}\right)^t.
\]
For $t\ge a_n\to\infty$,
\[
  \left(1-\frac{c}{t}\right)^t = e^{-c+o(1)}
\]
uniformly on $[a_n,b_n]$. Combining this with \Cref{thm:mesoscopic_heat_trace} yields
\[
  \mathcal N_{H_n}\!\left(\frac{c}{t}\right)
  \le
  \frac{(\cG+o(1))N_n/t}{(1-c/t)^t}
  =
  \left(e^c\cG+o(1)\right)\frac{N_n}{t}
\]
uniformly for $a_n\le t\le b_n$, which is the claim.

For item~(2), let $t\in[a_n,b_n]$ and write
\[
  M_t:=\mathcal N_{H_n}\!\left(\frac{K\log t}{t}\right).
\]
If $\lambda_k(H_n)>K\log t/t$, then
\[
  (1-\lambda_k(H_n))^t \le e^{-\lambda_k(H_n)t}\le t^{-K}.
\]
Hence
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  = \sum_k (1-\lambda_k(H_n))^t
  \le M_t + N_n t^{-K}.
\]
By \Cref{thm:mesoscopic_heat_trace},
\[
  \sum_{x\in H_n} p_t^{H_n}(x,x)
  =
  \left(\cG+o(1)\right)\frac{N_n}{t}
\]
uniformly on $[a_n,b_n]$. Therefore
\[
  M_t
  \ge
  \left(\cG+o(1)\right)\frac{N_n}{t}-N_n t^{-K}.
\]
Since $K>1$, one has $N_n t^{-K}=o(N_n/t)$ uniformly for $t\in[a_n,b_n]$. Multiplying by $t/N_n$ and taking the infimum over the window gives item~(2).
\end{proof}

\begin{corollary}[Reciprocal spectral mass on polynomial annuli]\label{cor:reciprocal_annulus}
Let $G$ and $(H_n)$ satisfy the geometric hypotheses and the quantitative on-diagonal assumption \textup{(QH)} from \Cref{thm:mesoscopic_heat_trace}. Fix $\eta\in(0,1/2)$ such that
\[
  E_n^{(\eta)}:=\{x\in H_n:\ d(x,V\setminus H_n)\le \rho_n^{1-\eta}\}
  \qquad\text{satisfies}\qquad
  |E_n^{(\eta)}|=o(N_n).
\]
Fix exponents
\[
  0<\alpha<\beta<1-2\eta
\]
and a constant $L>2$. Set
\[
  a_n=\lceil \rho_n^{2\alpha}\rceil,
  \qquad
  b_n=\lfloor \rho_n^{2\beta}\rfloor,
\]
and define
\[
  A_n(L):=\left\{k:\ \frac{L}{b_n}\le \lambda_k(H_n)\le \frac{1}{L a_n}\right\},
  \qquad
  c_L:=e^{-2/L}-e^{-L}>0.
\]
Then
\[
  \left(2\cG(\beta-\alpha)+o(1)\right)N_n\log\rho_n
  \le
  \sum_{k\in A_n(L)} \frac{1}{\lambda_k(H_n)}
  \le
  \left(\frac{2\cG(\beta-\alpha)}{c_L}+o(1)\right)N_n\log\rho_n.
\]
In particular,
\[
  \sum_{k\in A_n(L)} \frac{1}{\lambda_k(H_n)}
  \asymp
  N_n\log\rho_n.
\]
\end{corollary}

\begin{proof}
Let
\[
  K_n(\lambda)
  :=
  \sum_{t=a_n}^{b_n}(1-\lambda)^t
  =
  \frac{(1-\lambda)^{a_n}-(1-\lambda)^{b_n+1}}{\lambda}.
\]
Since \textup{(QH)} implies \textup{(TS)}, \Cref{cor:TS_window_polynomial} gives
\[
  S_n
  :=
  \sum_k K_n(\lambda_k(H_n))
  =
  \left(2\cG(\beta-\alpha)+o(1)\right)N_n\log\rho_n.
\]

For $k\in A_n(L)$ and all sufficiently large $n$, one has $\lambda_k(H_n)\le 1/(L a_n)\le 1/2$, hence
\[
  (1-\lambda_k(H_n))^{a_n}\ge e^{-2a_n\lambda_k(H_n)}\ge e^{-2/L}
\]
and
\[
  (1-\lambda_k(H_n))^{b_n+1}\le e^{-(b_n+1)\lambda_k(H_n)}\le e^{-L}.
\]
Therefore
\[
  \frac{c_L}{\lambda_k(H_n)}
  \le
  K_n(\lambda_k(H_n))
  \le
  \frac{1}{\lambda_k(H_n)}
\]
on $A_n(L)$. Since $S_n$ dominates the contribution of $A_n(L)$, this yields
\[
  c_L \sum_{k\in A_n(L)} \frac{1}{\lambda_k(H_n)}
  \le
  S_n,
\]
which proves the upper bound.

For the lower bound, decompose $S_n=S_n^{\mathrm{low}}+S_n^{\mathrm{mid}}+S_n^{\mathrm{high}}$, where the middle piece is the contribution of $A_n(L)$. On $A_n(L)$ we have $K_n(\lambda)\le 1/\lambda$, so it remains to show
\[
  S_n^{\mathrm{low}}+S_n^{\mathrm{high}}=O_L(N_n).
\]

For the low part, apply \Cref{thm:mesoscopic_heat_trace} to the singleton window $[b_n,b_n]$ to obtain
\[
  \sum_k (1-\lambda_k(H_n))^{b_n}
  =
  \left(\cG+o(1)\right)\frac{N_n}{b_n}.
\]
If $\lambda_k(H_n)<L/b_n$, then for large $n$ we have $\lambda_k(H_n)\le 1/2$ and therefore
\[
  (1-\lambda_k(H_n))^{b_n}\ge e^{-2b_n\lambda_k(H_n)}\ge e^{-2L}.
\]
Hence
\[
  \#\left\{k:\lambda_k(H_n)<\frac{L}{b_n}\right\}
  \le
  e^{2L}\sum_k (1-\lambda_k(H_n))^{b_n}
  =
  \left(e^{2L}\cG+o(1)\right)\frac{N_n}{b_n}.
\]
Since $K_n(\lambda)\le b_n-a_n+1\le b_n$, it follows that
\[
  S_n^{\mathrm{low}}
  \le
  b_n\,\#\left\{k:\lambda_k(H_n)<\frac{L}{b_n}\right\}
  =
  O_L(N_n).
\]

For the high part, if $\lambda_k(H_n)>1/(L a_n)$, then
\[
  K_n(\lambda_k(H_n))
  \le
  \frac{(1-\lambda_k(H_n))^{a_n}}{\lambda_k(H_n)}
  \le
  L a_n (1-\lambda_k(H_n))^{a_n}.
\]
Applying \Cref{thm:mesoscopic_heat_trace} to the singleton window $[a_n,a_n]$ gives
\[
  \sum_k (1-\lambda_k(H_n))^{a_n}
  =
  \left(\cG+o(1)\right)\frac{N_n}{a_n},
\]
and therefore
\[
  S_n^{\mathrm{high}}
  \le
  L a_n \sum_k (1-\lambda_k(H_n))^{a_n}
  =
  (L\cG+o(1))N_n
  =
  O_L(N_n).
\]

Combining the decomposition with $S_n^{\mathrm{mid}}\le \sum_{k\in A_n(L)}\lambda_k(H_n)^{-1}$ yields
\[
  S_n
  \le
  \sum_{k\in A_n(L)} \frac{1}{\lambda_k(H_n)}
  + O_L(N_n).
\]
Since $\log\rho_n\to\infty$, the $O_L(N_n)$ term is negligible compared with $N_n\log\rho_n$, and the lower bound follows from the asymptotic for $S_n$.
\end{proof}

\begin{theorem}[Asymptotically exact reciprocal mass on slowly trimmed annuli]\label{thm:trimmed_reciprocal_annulus}
Under the assumptions and notation of \Cref{cor:reciprocal_annulus}, let $(L_n)$ be any positive sequence such that
\[
  L_n\longrightarrow\infty,
  \qquad
  L_n=o(\log\rho_n),
  \qquad
  e^{2L_n}=o(\log\rho_n).
\]
Define the slowly trimmed annulus
\[
  A_n^\star
  :=
  \left\{
    k:\ \frac{L_n}{b_n}\le \lambda_k(H_n)\le \frac{1}{L_n a_n}
  \right\}.
\]
Then
\[
  \sum_{k\in A_n^\star}\frac{1}{\lambda_k(H_n)}
  =
  \left(2\cG(\beta-\alpha)+o(1)\right)N_n\log\rho_n.
\]
\end{theorem}

\begin{proof}
Let
\[
  K_n(\lambda)
  :=
  \sum_{t=a_n}^{b_n}(1-\lambda)^t
  =
  \frac{(1-\lambda)^{a_n}-(1-\lambda)^{b_n+1}}{\lambda},
\]
and write
\[
  S_n:=\sum_k K_n(\lambda_k(H_n)).
\]
Since \textup{(QH)} implies \textup{(TS)}, \Cref{cor:TS_window_polynomial} gives
\[
  S_n
  =
  \left(2\cG(\beta-\alpha)+o(1)\right)N_n\log\rho_n.
\]

Decompose $S_n=S_n^{\mathrm{low}}+S_n^{\star}+S_n^{\mathrm{high}}$ according to the partition
\[
  \lambda_k(H_n)<\frac{L_n}{b_n},
  \qquad
  \frac{L_n}{b_n}\le \lambda_k(H_n)\le \frac{1}{L_n a_n},
  \qquad
  \lambda_k(H_n)>\frac{1}{L_n a_n}.
\]
We first show that the outer pieces are negligible on the $N_n\log\rho_n$ scale.

For the low part, if $\lambda_k(H_n)<L_n/b_n$, then for all sufficiently large $n$ one has $\lambda_k(H_n)\le 1/2$ and hence
\[
  (1-\lambda_k(H_n))^{b_n}\ge e^{-2b_n\lambda_k(H_n)}\ge e^{-2L_n}.
\]
Applying \Cref{thm:mesoscopic_heat_trace} to the singleton window $[b_n,b_n]$ gives
\[
  \sum_k (1-\lambda_k(H_n))^{b_n}
  =
  \left(\cG+o(1)\right)\frac{N_n}{b_n}.
\]
Therefore
\[
  \#\left\{k:\lambda_k(H_n)<\frac{L_n}{b_n}\right\}
  \le
  e^{2L_n}\sum_k (1-\lambda_k(H_n))^{b_n}
  =
  \left(e^{2L_n}\cG+o(e^{2L_n})\right)\frac{N_n}{b_n}.
\]
Since always $K_n(\lambda)\le b_n-a_n+1\le b_n$, it follows that
\[
  S_n^{\mathrm{low}}
  \le
  b_n\,\#\left\{k:\lambda_k(H_n)<\frac{L_n}{b_n}\right\}
  =
  O(e^{2L_n}N_n)
  =
  o(N_n\log\rho_n).
\]

For the high part, if $\lambda_k(H_n)>1/(L_n a_n)$, then
\[
  K_n(\lambda_k(H_n))
  \le
  \frac{(1-\lambda_k(H_n))^{a_n}}{\lambda_k(H_n)}
  \le
  L_n a_n (1-\lambda_k(H_n))^{a_n}.
\]
Applying \Cref{thm:mesoscopic_heat_trace} to the singleton window $[a_n,a_n]$ yields
\[
  \sum_k (1-\lambda_k(H_n))^{a_n}
  =
  \left(\cG+o(1)\right)\frac{N_n}{a_n},
\]
and therefore
\[
  S_n^{\mathrm{high}}
  \le
  L_n a_n\sum_k (1-\lambda_k(H_n))^{a_n}
  =
  (L_n\cG+o(L_n))N_n
  =
  o(N_n\log\rho_n).
\]

Hence
\[
  S_n^\star
  =
  S_n-S_n^{\mathrm{low}}-S_n^{\mathrm{high}}
  =
  \left(2\cG(\beta-\alpha)+o(1)\right)N_n\log\rho_n.
\]

For $\lambda\in[L_n/b_n,\,1/(L_n a_n)]$ and all sufficiently large $n$, one has $\lambda\le 1/2$ and therefore
\[
  (1-\lambda)^{a_n}\ge e^{-2a_n\lambda}\ge e^{-2/L_n},
  \qquad
  (1-\lambda)^{b_n+1}\le e^{-(b_n+1)\lambda}\le e^{-L_n}.
\]
Thus, on the trimmed annulus,
\[
  \frac{c_n}{\lambda}\le K_n(\lambda)\le \frac{1}{\lambda},
  \qquad
  c_n:=e^{-2/L_n}-e^{-L_n}=1-o(1).
\]
Summing over $A_n^\star$ gives
\[
  c_n\sum_{k\in A_n^\star}\frac{1}{\lambda_k(H_n)}
  \le
  S_n^\star
  \le
  \sum_{k\in A_n^\star}\frac{1}{\lambda_k(H_n)}.
\]
Since $c_n\to1$ and $S_n^\star=\left(2\cG(\beta-\alpha)+o(1)\right)N_n\log\rho_n$, the claimed asymptotic follows.
\end{proof}

\begin{remark}[Choice of trimming scale]
The hypotheses of \Cref{thm:trimmed_reciprocal_annulus} allow any slowly diverging trimming parameter, for instance
\[
  L_n=\frac14\log\log\rho_n.
\]
Thus the exact reciprocal-mass law holds on a polynomial annulus after deleting only a vanishing logarithmic fraction of each diffusive edge.
\end{remark}

\begin{corollary}[Counting on polynomial spectral annuli]\label{cor:annulus_count}
Under the assumptions and notation of \Cref{cor:reciprocal_annulus},
\[
  \left(2L\cG(\beta-\alpha)+o(1)\right)\frac{N_n\log\rho_n}{b_n}
  \le
  \#A_n(L)
  \le
  \left(\frac{2\cG(\beta-\alpha)}{L c_L}+o(1)\right)\frac{N_n\log\rho_n}{a_n}.
\]
In particular,
\[
  \#A_n(L)\gtrsim N_n\rho_n^{-2\beta}\log\rho_n,
  \qquad
  \#A_n(L)\lesssim N_n\rho_n^{-2\alpha}\log\rho_n.
\]
\end{corollary}

\begin{proof}
For $k\in A_n(L)$ one has
\[
  L a_n \le \frac{1}{\lambda_k(H_n)}\le \frac{b_n}{L}.
\]
Therefore
\[
  L a_n\,\#A_n(L)
  \le
  \sum_{k\in A_n(L)}\frac{1}{\lambda_k(H_n)}
  \le
  \frac{b_n}{L}\,\#A_n(L).
\]
Combining these inequalities with \Cref{cor:reciprocal_annulus} gives the claim.
\end{proof}

\begin{remark}[Scope of the counting statement]\label{rem:mesoscopic_not_weyl}
\Cref{thm:trimmed_reciprocal_annulus}, \Cref{cor:reciprocal_annulus}, \Cref{cor:annulus_count}, and \Cref{cor:mesoscopic_counting} are intentionally weaker than a full Weyl law for the low-lying Dirichlet spectrum, but they recover progressively sharper spectral information from the time-domain analysis alone. The trimmed reciprocal-mass theorem gives an asymptotically exact $N_n\log\rho_n$ law on polynomial bands after removing only slowly shrinking edge regions; the fixed-annulus reciprocal-mass and counting corollaries force those bands to carry the correct diffusive amount of low-lying spectral mass; and the pointwise counting bound identifies the natural $N_n/t$ scale below the threshold $c/t$. The lower counting bound still requires the inflated threshold $(K\log t)/t$, because the heat-trace information alone does not exclude spectral mass from leaking to scales slightly above $1/t$. Removing that logarithmic gap would require additional spectral input beyond the present time-domain argument.
\end{remark}

\subsection{General Exhaustions}

\begin{theorem}[Abstract exhaustion criterion beyond metric balls]\label{thm:general_exhaustion}
Let $G$ be an infinite, connected, bounded degree graph satisfying \textup{(VG(2))}, \textup{(PI)}, and \Cref{ass:TS} with constant $\cG>0$. Let $(H_n)$ be finite connected domains in $G$ with $N_n:=|H_n|\to\infty$. Assume there exists a scale $\rho_n\to\infty$ such that $N_n\asymp \rho_n^2$ and, for every fixed $\eta\in(0,1/2)$, the boundary layer
\[
  E_n^{(\eta)}:=\{x\in H_n:\ d(x,V\setminus H_n)\le \rho_n^{1-\eta}\}
\]
satisfies $|E_n^{(\eta)}|=o(N_n)$. Assume moreover that there exist constants $C_{LT},c_{LT}>0$ such that for every $n$, every integer $t\ge \rho_n^2$, and every $v\in H_n$,
\[
  p_t^{H_n}(v,v)\le \frac{C_{LT}}{N_n} e^{-c_{LT}t/\rho_n^2}.
\]
Then
\[
  \frac{Z_{H_n}(1)}{N_n\log N_n}\longrightarrow \cG.
\]
\end{theorem}

\begin{proof}
Fix $\eta\in(0,1/2)$, set $I_n^{(\eta)}:=H_n\setminus E_n^{(\eta)}$, and let $T_n:=\lfloor \rho_n^{2(1-2\eta)}\rfloor$. For $v\in I_n^{(\eta)}$, the distance to the complement is greater than $\rho_n^{1-\eta}$, so the maximal inequality from \Cref{prop:maximal} gives
\[
  \sup_{1\le t\le T_n}\Prob_v(\tau_{H_n}\le t)=o(1).
\]
Therefore
\[
  G_{H_n}(v,v)\ge \sum_{t=1}^{T_n} p_t(v,v)-o(1)
  =\cG\log T_n+o(\log T_n)
  =2\cG(1-2\eta)\log \rho_n+o(\log \rho_n),
\]
uniformly in $v\in I_n^{(\eta)}$. Summing over $I_n^{(\eta)}$ and using $|E_n^{(\eta)}|=o(N_n)$ gives
\[
  \liminf_{n\to\infty}\frac{Z_{H_n}(1)}{N_n\log N_n}\ge \cG(1-2\eta).
\]

For the upper bound, split at $\lfloor \rho_n^2\rfloor$. The short-time contribution satisfies
\[
  \sum_{t=1}^{\lfloor \rho_n^2\rfloor} p_t^{H_n}(v,v)
  \le \sum_{t=1}^{\lfloor \rho_n^2\rfloor} p_t(v,v)
  = 2\cG\log \rho_n + o(\log \rho_n)
\]
uniformly in $v$, by \Cref{ass:TS}. The assumed long-time Dirichlet bound gives
\[
  \sum_{t>\rho_n^2} p_t^{H_n}(v,v)
  \le \frac{C_{LT}}{N_n}\sum_{t>\rho_n^2} e^{-c_{LT} t/\rho_n^2}
  = O\!\left(\frac{\rho_n^2}{N_n}\right)
  = O(1)
\]
because $N_n\asymp \rho_n^2$. Hence
\[
  G_{H_n}(v,v)\le 2\cG\log \rho_n+o(\log \rho_n)
\]
uniformly in $v$, and so
\[
  \limsup_{n\to\infty}\frac{Z_{H_n}(1)}{N_n\log N_n}\le \cG.
\]
Letting $\eta\downarrow0$ finishes the proof.
\end{proof}

\begin{theorem}[Abstract quantitative refinement for general exhaustions]\label{thm:general_exhaustion_qh}
Let $G$ be an infinite, connected, bounded degree graph satisfying \textup{(VG(2))}, \textup{(PI)}, and \textup{(QH)} with constant $\cG>0$ and exponent $\delta>0$. Let $(H_n)$ be finite connected domains in $G$ with $N_n:=|H_n|\to\infty$. Assume there exists a scale $\rho_n\to\infty$ such that $N_n\asymp \rho_n^2$ and, for every fixed $\eta\in(0,1/2)$, the boundary layer
\[
  E_n^{(\eta)}:=\{x\in H_n:\ d(x,V\setminus H_n)\le \rho_n^{1-\eta}\}
\]
satisfies $|E_n^{(\eta)}|=o(N_n)$. Assume moreover that there exist constants $C_{LT},c_{LT}>0$ such that for every $n$, every integer $t\ge \rho_n^2$, and every $v\in H_n$,
\[
  p_t^{H_n}(v,v)\le \frac{C_{LT}}{N_n} e^{-c_{LT}t/\rho_n^2}.
\]
Then
\[
  \frac{Z_{H_n}(1)}{N_n\log N_n}\longrightarrow \cG.
\]
Moreover,
\[
  Z_{H_n}(1)\le \cG N_n\log N_n + O(N_n).
\]
\end{theorem}

\begin{proof}
Since \textup{(QH)} implies the time-summed input \textup{(TS)}, the ratio limit follows from \Cref{thm:general_exhaustion}.

For the upper bound, fix $n$ and $v\in H_n$, and split at $T_n:=\lfloor \rho_n^2\rfloor$. Using $p_t^{H_n}(v,v)\le p_t(v,v)$ for integers $1\le t\le T_n$, we obtain
\[
  G_{H_n}(v,v)\le 1+\sum_{t=1}^{T_n} p_t(v,v)+\sum_{t>T_n} p_t^{H_n}(v,v).
\]
By \textup{(QH)},
\[
\sum_{t=1}^{T_n} p_t(v,v)
= \sum_{t=1}^{T_n}\left(\frac{\cG}{t}+O(t^{-1-\delta})\right)
= \cG\log T_n + O(1)
= 2\cG\log \rho_n + O(1),
\]
uniformly in $v$, because $\sum_{t\ge1} t^{-1-\delta}<\infty$. The assumed long-time Dirichlet estimate gives
\[
  \sum_{t>T_n} p_t^{H_n}(v,v)
  \le \frac{C_{LT}}{N_n}\sum_{t>T_n} e^{-c_{LT}t/\rho_n^2}
  = O\!\left(\frac{\rho_n^2}{N_n}\right)
  = O(1),
\]
since $N_n\asymp \rho_n^2$. Therefore
\[
  G_{H_n}(v,v)\le 2\cG\log \rho_n + O(1)
\]
uniformly in $v\in H_n$. Summing over $H_n$ and using $\log N_n = 2\log \rho_n + O(1)$ yields
\[
  Z_{H_n}(1)\le \cG N_n\log N_n + O(N_n).
\]
\end{proof}

\begin{remark}[Tauberian perspective]
An alternative (spectral) approach is to study the Stieltjes transform of the empirical spectral measure of $\LL_{G_n}$ and apply Tauberian theorems to the heat-trace or resolvent. We use the time-domain method because it isolates three inputs---QH, FK/PI, and LT---and makes their roles explicit.
\end{remark}

\subsection{Assumptions and Scope}


\begin{remark}[The role of metric balls and general exhaustions]\label{rem:general_exhaustions}
For both \Cref{thm:TS_balls} and its quantitative refinement \Cref{thm:main}, the metric-ball hypothesis is used only to manufacture two domain-level inputs required by the proof: a negligible boundary layer and a sharp long-time Dirichlet tail. \Cref{thm:general_exhaustion} and \Cref{thm:general_exhaustion_qh} isolate exactly these two requirements in abstract form and show that no other step of the argument depends on the domains being balls.

For metric balls, the annulus-volume estimate (AV) gives the boundary-layer control. For other exhaustions, the main issue is therefore not the time-domain decomposition itself but the verification of the domain-level hypotheses in \Cref{thm:general_exhaustion} and \Cref{thm:general_exhaustion_qh}. In particular, highly irregular ``rooms-and-corridors'' boundaries may destroy the needed long-time Dirichlet control and can reintroduce the $\log\log$ losses discussed in \Cref{rem:IU_necessity}. By contrast, Følner exhaustions with quantitatively thin boundary layers and regular Euclidean-type domains in periodic graphs are natural settings for the abstract criterion.
\end{remark}

\begin{remark}[Modularity of the assumptions]\label{rem:modular_assumptions}
The standing hypotheses play different roles. The pair \textup{(VG(2))}+\textup{(PI)} gives PHI, Gaussian bounds, exit estimates, and Faber--Krahn. For metric balls, \textup{(AV)} controls the boundary layer. The assumptions \textup{(TS)} and \textup{(QH)} concern the on-diagonal heat kernel, with \textup{(QH)} providing the sharper remainder needed for quantitative and mesoscopic statements. Finally, \textup{(LT)} is a large-time Dirichlet input used in the upper bounds. \Cref{thm:general_exhaustion} and \Cref{thm:general_exhaustion_qh} isolate the corresponding domain-level requirements, and the square-domain appendix verifies them in an explicit family.
\end{remark}

\begin{remark}[Poincar\'e inequality and QH]
The Poincaré inequality is essential in our argument: it gives PHI, Gaussian bounds, and Faber--Krahn. The stronger quantitative homogenisation assumption \Cref{ass:QH} is \emph{not} needed for the leading ratio limit once the weaker time-summed input \Cref{ass:TS} is available, but it is crucial for the sharp $O(1)$ control of the truncated return sums and hence for the quantitative refinement \Cref{thm:main}; see \Cref{rem:weaker_QH}. Without PI, even graphs with VG(2) may have bottlenecks that destroy the required large-scale regularity.
\end{remark}


\begin{remark}[Relaxing Bounded Degree and Weighted Graphs]\label{rem:weighted}
The bounded-degree assumption is used mainly to compare counting measure and degree measure (\Cref{rem:measures_operators}). A weighted version should be possible, but it would require reformulating the hypotheses in terms of the speed measure $m(x)=\sum_y w_{xy}$ and the weighted Dirichlet form, and then verifying the weighted analogues of PHI, FK, and the long-time Dirichlet estimate; see \cite{Delmotte99}. We do not pursue that extension here.
\end{remark}


\begin{remark}[Continuous vs. Discrete Time]\label{rem:continuous_time}
The analysis is carried out for the discrete-time LSRW generated by $\LL$. Since $Z_n(1)$ depends only on the eigenvalues of $\LL_n$, the quantity itself is unchanged by passing to continuous time. Under the corresponding continuous-time on-diagonal input $h_t(x,x)=\cG/t+O(t^{-1-\delta})$ and the same long-time Dirichlet control, the same argument gives the analogous asymptotic.
\end{remark}


\begin{remark}[Non-lazy random walks and the effect of laziness]\label{rem:non-lazy}
The paper is formulated for the lazy walk because the assumption (QH) is then genuinely pointwise in time. If one starts from an aperiodic walk and introduces laziness parameter $\alpha$, then
\[
\LL_{\alpha}=(1-\alpha)\LL_{0},
\]
and the effective diffusivity scales as $\Sigma_{\alpha}=(1-\alpha)\Sigma_{0}$. In dimension two this scales the corresponding leading trace coefficient by the factor $(1-\alpha)^{-1}$.

For genuinely periodic non-lazy walks, however, one must keep track of oscillations in the return probability. The simplest example is SRW on $\mathbb{Z}^2$: one has $p_{2m+1}(0,0)=0$ and $p_{2m}(0,0)\sim 1/(\pi m)$, so the pointwise asymptotic required in \Cref{ass:QH} fails, even though the summed trace coefficient is still $1/\pi$. The lazy walk analyzed in \Cref{rem:pi2} removes this period-two oscillation and satisfies $p_t(0,0)\sim (2/\pi)t^{-1}$.
\end{remark}

\section{Conclusion and Outlook}

We have proved a time-domain criterion for the critical two-dimensional asymptotic
\[
  Z_H(1)\asymp |H|\log|H|
\]
for Dirichlet graph Laplacians, together with a quantitative $O(|H|)$ refinement under stronger on-diagonal control. The proof separates the geometric input, the on-diagonal heat-kernel input, and the long-time Dirichlet input. This also makes it possible to pass from metric balls to abstract exhaustions and to derive the mesoscopic consequences proved in the body of the paper. On square domains, the appendix verifies the needed long-time and mesoscopic estimates in an explicit family.

Several natural questions remain. A first problem is to obtain a genuine two-term expansion
\[
  Z_n(1)=\cG N_n\log N_n + C_* N_n + o(N_n),
\]
which would require finer control of both boundary layers and long-time Dirichlet mass. A second problem is to derive the time-summed hypothesis \textup{(TS)} from model-specific homogenisation results in random media under assumptions weaker than the quantitative pointwise input \textup{(QH)}. A third direction is to identify broad classes of non-ball exhaustions, such as Følner-type domains with quantitative boundary regularity, for which the abstract domain-level hypotheses can be verified.

At the mesoscopic level, the new trimmed-annulus theorem shows that the reciprocal spectral mass already concentrates asymptotically on diffusive polynomial bands after only a slowly vanishing edge trimming. It is therefore natural to ask whether one can remove that trimming entirely and upgrade \Cref{thm:trimmed_reciprocal_annulus} and \Cref{cor:mesoscopic_counting} to a genuine Weyl-type statement at scale $1/t$ rather than at logarithmically inflated thresholds. On homogeneous lattices the $O(N_n)$ correction is closely related to lattice Green-function constants; understanding the analogous second-order term in irregular media remains an interesting open problem.

\section{Appendix: Comparison with Other Growth Regimes}\label{app:growth}
The quadratic volume growth assumption (effective dimension $d=2$) is critical for the $N_n \log N_n$ behaviour, as summarized in \Cref{tab:growth}. We briefly sketch the arguments for other dimensions.

\begin{itemize}
\item \textbf{Linear growth ($d=1$):} E.g., $\mathbb{Z}$. $N_n \asymp R_n$. $p_t(x,x) \sim C t^{-1/2}$. The characteristic time scale is $T \asymp R_n^2 \asymp N_n^2$. The Green function for interior vertices is $G_{G_n}(x,x) \approx \sum_{t=1}^{T} t^{-1/2} \asymp T^{1/2} \asymp N_n$.

More precisely, for a path (interval) of length $R$, the Green function satisfies $G(x,x) \asymp \min\{x, R-x\}$ (the expected time spent at $x$ before exiting either end). Summing this over $x=1,\dots,R$ yields $\sum_x G(x,x) \asymp R^2 \asymp N_n^2$.


\item \textbf{Polynomial growth ($d>2$):} E.g., $\mathbb{Z}^d$ with $d\geq 3$. Then $p_t(x,x) \sim C t^{-d/2}$, so the walk is transient because $d/2 > 1$. Hence the full Green function $G(x,x) = \sum_{t=0}^{\infty} p_t(x,x)$ converges. The Dirichlet Green function $G_{G_n}(x,x)$ is uniformly bounded by $G(x,x)$, and therefore
\[
Z_n(1) = \sum_{v \in G_n} G_{G_n}(v,v) \asymp N_n.
\]
\end{itemize}


\section{Appendix: Square-Domain Verification Program}
\label{app:alt_pi}

\paragraph{Square domains.}
This appendix treats the square-domain case in detail. First, we identify the leading square-domain spectral-zeta asymptotic for a broad diagonal-symbol class. Second, for the corresponding nonnegative symbol class, we prove the long-time diagonal Dirichlet bound. Third, we derive the mesoscopic square heat-trace law on subdiffusive windows. Fourth, we specialize the abstract square analysis to the lazy axial-diagonal family and obtain a corresponding explicit theorem.

We work on the square
\[
V_R=\{1,\dots,R\}^2 \subset \mathbb{Z}^2,
\]
with $N_R=R^2$.

\begin{theorem}[General square-domain asymptotic for diagonal symbol models]\label{thm:poly_square}
For each $R\ge1$, let $P_R$ be a self-adjoint operator on $\ell^2(V_R)$ that is diagonalized by the sine-product basis $(\psi_k\otimes\psi_\ell)_{1\le k,\ell\le R}$, with eigenvalues
\[
  F(2\cos\alpha_k,2\cos\alpha_\ell),
  \qquad
  \alpha_k:=\frac{\pi k}{R+1},
\]
for some fixed function $F:[-2,2]^2\to\mathbb R$ independent of $R$. Let $\mathcal L_R:=I-P_R$. Assume
\[
  F(2,2)=1,
  \qquad
  1-F(x,y)>0
  \quad\text{for all }(x,y)\in[-2,2]^2\setminus\{(2,2)\},
\]
and assume that, as $(a,b)\to(0,0)$,
\[
  1-F(2\cos a,2\cos b)
  =
  \frac12\bigl(\sigma_1 a^2+\sigma_2 b^2\bigr)+O(a^4+b^4)
\]
for some $\sigma_1,\sigma_2>0$. Then
\[
  \tr(\mathcal L_R^{-1})
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\,R^2\log R^2
  + O(R^2).
\]
\end{theorem}

\begin{proof}
For $k=1,\dots,R$ set $\alpha_k=\pi k/(R+1)$ and
\[
  \psi_k(i):=\sqrt{\frac{2}{R+1}}\sin(\alpha_k i),
  \qquad 1\le i\le R.
\]
By assumption, the tensor-product basis $(\psi_k\otimes\psi_\ell)_{1\le k,\ell\le R}$ diagonalizes $\mathcal L_R$ with eigenvalues
\[
  1-F(2\cos\alpha_k,2\cos\alpha_\ell),
  \qquad 1\le k,\ell\le R.
\]
Therefore
\[
  \tr(\mathcal L_R^{-1})
  =
  \sum_{k,\ell=1}^R
  \frac{1}{1-F(2\cos\alpha_k,2\cos\alpha_\ell)}.
\]

Define
\[
  H(a,b)
  :=
  \frac{1}{1-F(2\cos a,2\cos b)}
  -
  \frac{2}{\sigma_1 a^2+\sigma_2 b^2},
  \qquad (a,b)\in(0,\pi]^2.
\]
By the assumed expansion at the origin, $H$ extends continuously to $(0,0)$. Since $1-F(x,y)>0$ away from $(2,2)$, the function $H$ is bounded on $[0,\pi]^2$. Consequently,
\[
  \tr(\mathcal L_R^{-1})
  =
  2\sum_{k,\ell=1}^R\frac{1}{\sigma_1\alpha_k^2+\sigma_2\alpha_\ell^2}
  + O(R^2).
\]
Using $\alpha_k=\pi k/(R+1)$, this becomes
\[
  \tr(\mathcal L_R^{-1})
  =
  \frac{2(R+1)^2}{\pi^2}
  \sum_{k,\ell=1}^R \frac{1}{\sigma_1 k^2+\sigma_2\ell^2}
  + O(R^2).
\]

For the lattice sum one has
\[
  \sum_{k,\ell=1}^R \frac{1}{\sigma_1 k^2+\sigma_2\ell^2}
  =
  \frac{\pi}{2\sqrt{\sigma_1\sigma_2}}\log R+O(1).
\]
Indeed, $(x,y)\mapsto (\sigma_1 x^2+\sigma_2 y^2)^{-1}$ is decreasing in each variable on $[1,\infty)^2$, so
\[
  \sum_{k,\ell=2}^R \frac{1}{\sigma_1 k^2+\sigma_2\ell^2}
  =
  \int_1^R\!\!\int_1^R \frac{dx\,dy}{\sigma_1 x^2+\sigma_2 y^2}
  + O(1).
\]
After the change of variables $u=\sqrt{\sigma_1}\,x$, $v=\sqrt{\sigma_2}\,y$, the integral becomes
\[
  \frac{1}{\sqrt{\sigma_1\sigma_2}}
  \int_{\sqrt{\sigma_1}}^{R\sqrt{\sigma_1}}
  \int_{\sqrt{\sigma_2}}^{R\sqrt{\sigma_2}}
  \frac{du\,dv}{u^2+v^2}.
\]
This rectangle integral differs by $O(1)$ from the quarter-annulus integral over radii between fixed positive constants times $1$ and $R$, hence equals
\[
  \frac{1}{\sqrt{\sigma_1\sigma_2}}
  \left(\frac{\pi}{2}\log R + O(1)\right).
\]
Substituting gives
\begin{align*}
  \tr(\mathcal L_R^{-1})
  &=
  \frac{2(R+1)^2}{\pi^2}
  \left(
    \frac{\pi}{2\sqrt{\sigma_1\sigma_2}}\log R + O(1)
  \right)
  + O(R^2) \\
  &=
  \frac{1}{\pi\sqrt{\sigma_1\sigma_2}}R^2\log R \\
  &\qquad + O(R^2).
\end{align*}
Since $\log R^2=2\log R$, this is exactly
\[
  \tr(\mathcal L_R^{-1})
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\,R^2\log R^2 + O(R^2).
\]
\end{proof}

\begin{remark}[Verified square-domain model class]\label{rem:poly_square_class}
\Cref{thm:poly_square} applies to any square-domain model whose killed transition operator is diagonal in the sine-product basis with a fixed symbol $F$. Polynomial-adjacency models are a concrete subclass, but the theorem is not limited to polynomial symbols. This includes the nearest-neighbour square-lattice walk, the king walk, and similar explicit periodic models built from horizontal, vertical, and diagonal nearest-neighbour couplings.
\end{remark}

\begin{proposition}[Long-time diagonal bound for nonnegative square-symbol models]\label{prop:poly_square_LT}
Assume the hypotheses of \Cref{thm:poly_square}, and in addition assume
\[
  0\le F(x,y)\le 1
  \qquad\text{for all }(x,y)\in[-2,2]^2.
\]
For $t\in\mathbb N$ define the kernel
\[
  K_t^{(R)}(u,v):=\langle \delta_u, P_R^t\delta_v\rangle,
  \qquad u,v\in V_R.
\]
Then there exists $C_{LT}(F)<\infty$ such that for every $R\ge1$, every $u\in V_R$, and every integer $t\ge R^2$,
\[
  K_t^{(R)}(u,u)
  \le
  \frac{C_{LT}(F)}{|V_R|}e^{-\lambda_1(\mathcal L_R)t}.
\]
In particular, the long-time Dirichlet estimate \textup{(LT)} holds on axis-parallel squares for every lazy finite-range model covered by \Cref{thm:poly_square}.
\end{proposition}

\begin{proof}
Write
\[
  \Lambda(a,b):=1-F(2\cos a,2\cos b),
  \qquad (a,b)\in[0,\pi]^2.
\]
By the hypotheses of \Cref{thm:poly_square}, $\Lambda(a,b)>0$ away from $(0,0)$ and
\[
  \Lambda(a,b)=\frac12(\sigma_1 a^2+\sigma_2 b^2)+O(a^4+b^4)
\]
as $(a,b)\to(0,0)$. Hence the ratio
\[
  \frac{\Lambda(a,b)}{a^2+b^2}
\]
extends continuously to $[0,\pi]^2\setminus\{(0,0)\}$ and has a positive limit inferior at the origin. By compactness, there exists $c_F>0$ such that
\begin{equation}\label{eq:poly_square_coercive}
  \Lambda(a,b)\ge c_F(a^2+b^2)
  \qquad\text{for all }(a,b)\in[0,\pi]^2.
\end{equation}
On the other hand, the expansion at the origin yields $C_F<\infty$ such that
\begin{equation}\label{eq:poly_square_lambda1_upper}
  \lambda_1(\mathcal L_R)\le \Lambda(\alpha_1,\alpha_1)\le \frac{C_F}{R^2}
\end{equation}
for all $R\ge1$.

Fix $u=(i,j)\in V_R$. Since $P_R$ is diagonalized by the sine-product basis, the eigenvalues of $P_R$ are
\[
  \mu_{k,\ell}:=F(2\cos\alpha_k,2\cos\alpha_\ell)\in[0,1],
  \qquad
  1\le k,\ell\le R,
\]
and
\[
  K_t^{(R)}(u,u)
  =
  \sum_{k,\ell=1}^R \mu_{k,\ell}^t \psi_k(i)^2\psi_\ell(j)^2.
\]
Using
\[
  \psi_k(i)^2\le \frac{2}{R+1},
  \qquad
  \psi_\ell(j)^2\le \frac{2}{R+1},
\]
we obtain
\[
  K_{R^2}^{(R)}(u,u)
  \le
  \frac{4}{(R+1)^2}
  \sum_{k,\ell=1}^R \mu_{k,\ell}^{R^2}.
\]
Since $0\le \mu_{k,\ell}\le1$, one has $\mu_{k,\ell}^{R^2}\le e^{-(1-\mu_{k,\ell})R^2}=e^{-\Lambda(\alpha_k,\alpha_\ell)R^2}$. By \eqref{eq:poly_square_coercive},
\[
  \mu_{k,\ell}^{R^2}
  \le
  \exp\!\left(
    -c_F R^2(\alpha_k^2+\alpha_\ell^2)
  \right)
  \le
  \exp\!\left(
    -c_F\pi^2\frac{R^2}{(R+1)^2}(k^2+\ell^2)
  \right).
\]
Since $R^2/(R+1)^2\ge 1/4$ for every $R\ge1$,
\[
  \sum_{k,\ell=1}^R \mu_{k,\ell}^{R^2}
  \le
  \sum_{k,\ell\ge1} e^{-(c_F\pi^2/4)(k^2+\ell^2)}
  =:M_F<\infty.
\]
Therefore
\begin{equation}\label{eq:poly_square_time_R2}
  K_{R^2}^{(R)}(u,u)\le \frac{4M_F}{(R+1)^2}\le \frac{C'_F}{R^2}.
\end{equation}

Now let $t\ge R^2$ be an integer. Writing the spectral decomposition again and using that every eigenvalue of $P_R$ is at most $1-\lambda_1(\mathcal L_R)$, we get
\begin{align*}
  K_t^{(R)}(u,u)
  &= \sum_{k,\ell=1}^R \mu_{k,\ell}^{t-R^2}\mu_{k,\ell}^{R^2}\psi_k(i)^2\psi_\ell(j)^2 \\
  &\le (1-\lambda_1(\mathcal L_R))^{t-R^2}
      \sum_{k,\ell=1}^R \mu_{k,\ell}^{R^2}\psi_k(i)^2\psi_\ell(j)^2 \\
  &\le e^{-\lambda_1(\mathcal L_R)(t-R^2)}K_{R^2}^{(R)}(u,u).
\end{align*}
Combining this with \eqref{eq:poly_square_time_R2} and \eqref{eq:poly_square_lambda1_upper} yields
\[
  K_t^{(R)}(u,u)
  \le
  \frac{C'_F}{R^2}
  e^{\lambda_1(\mathcal L_R)R^2}
  e^{-\lambda_1(\mathcal L_R)t}
  \le
  \frac{C'_F e^{C_F}}{R^2}e^{-\lambda_1(\mathcal L_R)t}.
\]
Since $|V_R|=R^2$, this is exactly the claimed bound.
\end{proof}

\begin{theorem}[Mesoscopic heat trace on square symbol models]\label{thm:poly_square_heat_trace}
Assume the hypotheses of \Cref{thm:poly_square} and \Cref{prop:poly_square_LT}. Let $(a_R)$ and $(b_R)$ be integer sequences such that
\[
  a_R\longrightarrow\infty,
  \qquad
  a_R\le b_R,
  \qquad
  b_R=o(R^2).
\]
Then
\[
  \sup_{a_R\le t\le b_R}
  \frac{t}{|V_R|}
  \left|
    \sum_{u\in V_R} K_t^{(R)}(u,u)
    -
    \frac{|V_R|}{2\pi\sqrt{\sigma_1\sigma_2}\,t}
  \right|
  \longrightarrow 0.
\]
Equivalently,
\[
  \sum_{u\in V_R} K_t^{(R)}(u,u)
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\frac{R^2}{t}
  + o\!\left(\frac{R^2}{t}\right)
\]
uniformly for integers $t\in[a_R,b_R]$.
\end{theorem}

\begin{proof}
Write
\[
  \lambda_{k,\ell}^{(R)}:=1-F(2\cos\alpha_k,2\cos\alpha_\ell),
  \qquad
  q_{k,\ell}^{(R)}:=\frac12(\sigma_1\alpha_k^2+\sigma_2\alpha_\ell^2),
\]
and
\[
  H_R(t):=\sum_{u\in V_R}K_t^{(R)}(u,u)=\sum_{k,\ell=1}^R (1-\lambda_{k,\ell}^{(R)})^t.
\]
By the expansion in \Cref{thm:poly_square}, there exists $C_F<\infty$ such that
\begin{equation}\label{eq:poly_square_quartic_error}
  \left|
    \lambda_{k,\ell}^{(R)}-q_{k,\ell}^{(R)}
  \right|
  \le
  C_F(\alpha_k^4+\alpha_\ell^4)
  \qquad(1\le k,\ell\le R).
\end{equation}
By \eqref{eq:poly_square_coercive}, there exists $c_0>0$ such that
\begin{equation}\label{eq:poly_square_q_compare}
  \lambda_{k,\ell}^{(R)}\ge c_0 q_{k,\ell}^{(R)}
  \qquad(1\le k,\ell\le R).
\end{equation}

Fix $M\ge1$ and split the index set into
\[
  \mathcal L_t:=\{(k,\ell): q_{k,\ell}^{(R)}\le M/t\},
  \qquad
  \mathcal H_t:=\{(k,\ell): q_{k,\ell}^{(R)}> M/t\}.
\]

On $\mathcal L_t$, \eqref{eq:poly_square_quartic_error} and $q_{k,\ell}^{(R)}\le M/t$ give
\[
  t\bigl|\lambda_{k,\ell}^{(R)}-q_{k,\ell}^{(R)}\bigr|
  \le
  C_F t(\alpha_k^4+\alpha_\ell^4)
  \le
  \frac{4C_F M^2}{t\min(\sigma_1,\sigma_2)^2}
  =: \varepsilon_R(M),
\]
where $\varepsilon_R(M)\to0$ uniformly for $t\in[a_R,b_R]$ because $a_R\to\infty$. Also,
\[
  t\bigl(\lambda_{k,\ell}^{(R)}\bigr)^2
  \le \frac{C M^2}{t}=o(1)
\]
uniformly on $\mathcal L_t$. Therefore
\[
  (1-\lambda_{k,\ell}^{(R)})^t
  =
  e^{-t q_{k,\ell}^{(R)}}\bigl(1+o_R(1)\bigr)
\]
uniformly on $\mathcal L_t$ and for $t\in[a_R,b_R]$. Since $\#\mathcal L_t=O(R^2/t)$ uniformly in that range, we obtain
\begin{equation}\label{eq:poly_square_low_freq}
  \sum_{(k,\ell)\in\mathcal L_t}(1-\lambda_{k,\ell}^{(R)})^t
  =
  \sum_{(k,\ell)\in\mathcal L_t}e^{-t q_{k,\ell}^{(R)}}
  + o\!\left(\frac{R^2}{t}\right)
\end{equation}
uniformly for $t\in[a_R,b_R]$.

For the high-frequency contribution, \eqref{eq:poly_square_q_compare} gives
\[
  \sum_{(k,\ell)\in\mathcal H_t}(1-\lambda_{k,\ell}^{(R)})^t
  \le
  \sum_{q_{k,\ell}^{(R)}>M/t} e^{-c_0 t q_{k,\ell}^{(R)}}.
\]
Set
\[
  a_i:=\frac{\sigma_i\pi^2}{2},
  \qquad
  s_R(t):=\frac{t}{(R+1)^2}.
\]
Then $q_{k,\ell}^{(R)}=s_R(t)(a_1k^2+a_2\ell^2)/t$, so the right-hand side is bounded by a Gaussian lattice tail:
\[
  \sum_{a_1k^2+a_2\ell^2> M/s_R(t)}
  e^{-c_0 s_R(t)(a_1k^2+a_2\ell^2)}.
\]
By comparison with the corresponding integral after the rescaling
\[
  x=\sqrt{s_R(t)}\,k,
  \qquad
  y=\sqrt{s_R(t)}\,\ell,
\]
there exists a function $\omega(M)\downarrow0$ as $M\to\infty$ such that
\begin{equation}\label{eq:poly_square_high_freq}
  \sum_{(k,\ell)\in\mathcal H_t}(1-\lambda_{k,\ell}^{(R)})^t
  \le
  \omega(M)\frac{R^2}{t}
\end{equation}
uniformly in $R$ and $t\in[a_R,b_R]$.

It remains to evaluate the quadratic model sum
\[
  G_R(t):=\sum_{k,\ell=1}^R e^{-t q_{k,\ell}^{(R)}}
  =
  \left(
    \sum_{k=1}^R e^{-a_1 s_R(t)k^2}
  \right)
  \left(
    \sum_{\ell=1}^R e^{-a_2 s_R(t)\ell^2}
  \right).
\]
Because $b_R=o(R^2)$, one has $\sup_{a_R\le t\le b_R}s_R(t)\to0$. For each fixed $a>0$ and $0<s\le1$, elementary integral comparison gives
\[
  \sum_{m=1}^\infty e^{-a s m^2}
  =
  \frac12\sqrt{\frac{\pi}{a s}}+O(1),
\]
uniformly as $s\downarrow0$. The tail beyond $m=R$ is $O(e^{-a s_R(t)R^2})=O(e^{-c a_R})=o(1)$ uniformly for $t\in[a_R,b_R]$. Hence
\[
  \sum_{k=1}^R e^{-a_i s_R(t)k^2}
  =
  \frac12\sqrt{\frac{\pi}{a_i s_R(t)}}+O(1)
\]
uniformly in $t\in[a_R,b_R]$, for $i=1,2$. Multiplying the two one-dimensional asymptotics yields
\[
  G_R(t)
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\frac{(R+1)^2}{t}
  + O\!\left(\frac{R}{\sqrt t}\right)+O(1)
\]
uniformly for $t\in[a_R,b_R]$. Since $b_R=o(R^2)$ and $a_R\to\infty$,
\[
  \frac{R/\sqrt t}{R^2/t}=\frac{\sqrt t}{R}\longrightarrow0,
  \qquad
  \frac{1}{R^2/t}\longrightarrow0
\]
uniformly for $t\in[a_R,b_R]$. Therefore
\begin{equation}\label{eq:poly_square_gaussian_main}
  G_R(t)
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\frac{R^2}{t}
  + o\!\left(\frac{R^2}{t}\right)
\end{equation}
uniformly on the whole window.

Combining \eqref{eq:poly_square_low_freq}, \eqref{eq:poly_square_high_freq}, and \eqref{eq:poly_square_gaussian_main}, then sending $M\to\infty$, proves the theorem.
\end{proof}

\begin{lemma}[Weighted quadrant lattice sum]\label{lem:weighted_lattice_sum}
For every $\sigma_1,\sigma_2>0$, there exists a constant $C_{\sigma_1,\sigma_2}\in\mathbb R$ such that
\[
  \sum_{k,\ell=1}^R \frac{1}{\sigma_1 k^2+\sigma_2 \ell^2}
  =
  \frac{\pi}{2\sqrt{\sigma_1\sigma_2}}\log R
  + C_{\sigma_1,\sigma_2}
  + o(1).
\]
\end{lemma}

\begin{proof}
Set $\lambda:=\sqrt{\sigma_1/\sigma_2}$. For $k,R\ge1$, define
\[
  E_{k,R}
  :=
  \sum_{\ell=1}^R \frac{1}{\sigma_1 k^2+\sigma_2 \ell^2}
  -
  \int_0^R \frac{dy}{\sigma_1 k^2+\sigma_2 y^2}.
\]
Since $y\mapsto (\sigma_1 k^2+\sigma_2 y^2)^{-1}$ is positive and decreasing on $[0,\infty)$,
\[
  0\le E_{k,R}\le \frac{1}{\sigma_1 k^2}.
\]
Moreover, for each fixed $k$ one has
\[
  E_{k,R}\longrightarrow
  e_k
  :=
  \sum_{\ell=1}^\infty \frac{1}{\sigma_1 k^2+\sigma_2 \ell^2}
  -
  \int_0^\infty \frac{dy}{\sigma_1 k^2+\sigma_2 y^2}.
\]
Since $\sum_{k\ge1} k^{-2}<\infty$, dominated convergence gives
\[
  \sum_{k=1}^R E_{k,R}
  =
  E_{\sigma_1,\sigma_2}+o(1)
\]
for some finite constant $E_{\sigma_1,\sigma_2}$.

Also
\[
  \int_0^R \frac{dy}{\sigma_1 k^2+\sigma_2 y^2}
  =
  \frac{1}{\sqrt{\sigma_1\sigma_2}\,k}\arctan\!\left(\frac{R}{\lambda k}\right).
\]
Hence
\[
  \sum_{k,\ell=1}^R \frac{1}{\sigma_1 k^2+\sigma_2 \ell^2}
  =
  E_{\sigma_1,\sigma_2}
  +
  \frac{1}{\sqrt{\sigma_1\sigma_2}}
  \sum_{k=1}^R \frac1k \arctan\!\left(\frac{R}{\lambda k}\right)
  + o(1).
\]
Using $\arctan(u)=\frac{\pi}{2}-\arctan(u^{-1})$ for $u>0$,
\[
  \sum_{k=1}^R \frac1k \arctan\!\left(\frac{R}{\lambda k}\right)
  =
  \frac{\pi}{2}\sum_{k=1}^R \frac1k
  -
  \sum_{k=1}^R \frac1k \arctan\!\left(\frac{\lambda k}{R}\right).
\]
The harmonic sum satisfies
\[
  \sum_{k=1}^R \frac1k = \log R+\gamma+o(1).
\]
For
\[
  \phi_\lambda(x):=
  \begin{cases}
    \dfrac{\arctan(\lambda x)}{x}, & x>0,\\[2mm]
    \lambda, & x=0,
  \end{cases}
\]
the function $\phi_\lambda$ is continuous on $[0,1]$, so
\[
  \sum_{k=1}^R \frac1k \arctan\!\left(\frac{\lambda k}{R}\right)
  =
  \frac1R\sum_{k=1}^R \phi_\lambda(k/R)
  \longrightarrow
  \int_0^1 \frac{\arctan(\lambda x)}{x}\,dx.
\]
Combining these identities proves the claim, with
\[
  C_{\sigma_1,\sigma_2}
  :=
  E_{\sigma_1,\sigma_2}
  +
  \frac{1}{\sqrt{\sigma_1\sigma_2}}
  \left(
    \frac{\pi\gamma}{2}
    -
    \int_0^1 \frac{\arctan(\lambda x)}{x}\,dx
  \right).
\]
\end{proof}

\begin{theorem}[Second-order square asymptotic for diagonal symbol models]\label{thm:poly_square_second}
Under the assumptions of \Cref{thm:poly_square}, there exists a constant $C_F\in\mathbb R$ such that
\[
  \tr(\mathcal L_R^{-1})
  =
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\,R^2\log R^2
  +
  C_F R^2
  + o(R^2).
\]
\end{theorem}

\begin{proof}
With the function $H$ from the proof of \Cref{thm:poly_square}, one has
\[
  \tr(\mathcal L_R^{-1})
  =
  \sum_{k,\ell=1}^R H(\alpha_k,\alpha_\ell)
  +
  2\sum_{k,\ell=1}^R\frac{1}{\sigma_1\alpha_k^2+\sigma_2\alpha_\ell^2}.
\]
Since $H$ is continuous on $[0,\pi]^2$, the Riemann-sum convergence gives
\[
  \sum_{k,\ell=1}^R H(\alpha_k,\alpha_\ell)
  =
  \frac{(R+1)^2}{\pi^2}\int_0^\pi\!\!\int_0^\pi H(a,b)\,da\,db
  + o(R^2).
\]
Also,
\[
  2\sum_{k,\ell=1}^R\frac{1}{\sigma_1\alpha_k^2+\sigma_2\alpha_\ell^2}
  =
  \frac{2(R+1)^2}{\pi^2}
  \sum_{k,\ell=1}^R \frac{1}{\sigma_1 k^2+\sigma_2\ell^2}.
\]
Applying \Cref{lem:weighted_lattice_sum} yields
\begin{align*}
  2\sum_{k,\ell=1}^R\frac{1}{\sigma_1\alpha_k^2+\sigma_2\alpha_\ell^2}
  &=
  \frac{2(R+1)^2}{\pi^2}
  \left(
    \frac{\pi}{2\sqrt{\sigma_1\sigma_2}}\log R
    +
    C_{\sigma_1,\sigma_2}
    + o(1)
  \right) \\
  &=
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}\,R^2\log R^2
  +
  \frac{2C_{\sigma_1,\sigma_2}}{\pi^2}R^2
  + o(R^2),
\end{align*}
because $(R+1)^2\log R = R^2\log R + o(R^2)$. Combining the two displayed asymptotics proves the claim.
\end{proof}

\begin{theorem}[Explicit square-domain asymptotic on $\mathbb Z^2$]\label{thm:Z2_square}
Let $\mathcal{L}_{R}^{\mathrm{SRW}}$ be the Dirichlet generator of the non-lazy simple random walk on $V_R$, and let $\mathcal{L}_{R}^{\mathrm{lazy}}=\frac12 \mathcal{L}_{R}^{\mathrm{SRW}}$ be the corresponding lazy Dirichlet generator. Then
\[
\tr\!\bigl((\mathcal{L}_{R}^{\mathrm{SRW}})^{-1}\bigr)
=\frac{1}{\pi}R^2\log R^2+O(R^2),
\]
and therefore
\[
\tr\!\bigl((\mathcal{L}_{R}^{\mathrm{lazy}})^{-1}\bigr)
=\frac{2}{\pi}R^2\log R^2+O(R^2).
\]
Equivalently,
\[
\frac{\tr\!\bigl((\mathcal{L}_{R}^{\mathrm{lazy}})^{-1}\bigr)}{R^2\log R^2}\longrightarrow \frac{2}{\pi}.
\]
\end{theorem}

\begin{proof}
For the non-lazy walk one has
\[
  P_R^{\mathrm{SRW}}
  =
  \frac14\bigl(A_R\otimes I+I\otimes A_R\bigr)
  =
  Q_{\mathrm{SRW}}(A_R\otimes I,I\otimes A_R)
\]
with
\[
  Q_{\mathrm{SRW}}(X,Y):=\frac{X+Y}{4}.
\]
Moreover,
\[
  1-Q_{\mathrm{SRW}}(2\cos a,2\cos b)
  =
  1-\frac{\cos a+\cos b}{2}
  =
  \frac14(a^2+b^2)+O(a^4+b^4),
\]
so \Cref{thm:poly_square} applies with $\sigma_1=\sigma_2=\tfrac12$. This yields
\[
  \tr\!\bigl((\mathcal{L}_{R}^{\mathrm{SRW}})^{-1}\bigr)
  =
  \frac{1}{\pi}R^2\log R^2+O(R^2).
\]
Finally, $\mathcal{L}_{R}^{\mathrm{lazy}}=\frac12\mathcal{L}_{R}^{\mathrm{SRW}}$, so
\[
\tr\!\bigl((\mathcal{L}_{R}^{\mathrm{lazy}})^{-1}\bigr)
=2\,\tr\!\bigl((\mathcal{L}_{R}^{\mathrm{SRW}})^{-1}\bigr)
=\frac{2}{\pi}R^2\log R^2+O(R^2).
\]
\end{proof}

\subsection{A one-parameter axial-diagonal family}\label{app:king}

We now pass from the abstract square-symbol class to a concrete one-parameter family. The next corollary identifies the spectral-zeta constant, while \Cref{thm:axial_diagonal_verified} collects the long-time, mesoscopic, and zeta conclusions for the lazy model.

For $0\le \theta<1$, let $P^{(\theta),0}$ be the non-lazy walk on $\mathbb Z^2$ that moves to each axial neighbour $(\pm1,0),(0,\pm1)$ with probability $(1-\theta)/4$ and to each diagonal neighbour $(\pm1,\pm1)$ with probability $\theta/4$. Let $\mathcal L_R^{(\theta),0}$ be its Dirichlet generator on $V_R$, and let $\mathcal L_R^{(\theta),1/2}=\frac12\mathcal L_R^{(\theta),0}$ be the corresponding lazy generator.

\begin{corollary}[Square-domain asymptotic for the axial-diagonal family]\label{cor:axial_diagonal_square}
For every $0\le \theta<1$,
\[
\tr\!\bigl((\mathcal L_R^{(\theta),0})^{-1}\bigr)
=\frac{1}{\pi(1+\theta)}R^2\log R^2+O(R^2),
\]
and therefore
\[
\tr\!\bigl((\mathcal L_R^{(\theta),1/2})^{-1}\bigr)
=\frac{2}{\pi(1+\theta)}R^2\log R^2+O(R^2).
\]
\end{corollary}

\begin{proof}
Let $A_R$ denote the adjacency matrix of the path $\{1,\dots,R\}$. On $V_R$ the non-lazy transition operator is
\[
  P_R^{(\theta),0}
  =
  \frac{1-\theta}{4}\bigl(A_R\otimes I+I\otimes A_R\bigr)
  +
  \frac{\theta}{4}A_R\otimes A_R
  =
  Q_\theta(A_R\otimes I,I\otimes A_R),
\]
where
\[
  Q_\theta(X,Y):=\frac{1-\theta}{4}(X+Y)+\frac{\theta}{4}XY.
\]
Since $Q_\theta(2,2)=1$, it remains to check the hypotheses of \Cref{thm:poly_square}. Writing $x=2-a$, $y=2-b$ with $a,b\in[0,4]$, one obtains
\[
  1-Q_\theta(x,y)
  =
  \frac{1+\theta}{4}(a+b)-\frac{\theta}{4}ab.
\]
Because $ab\le 2(a+b)$ on $[0,4]^2$ and $\theta<1$, this gives
\[
  1-Q_\theta(x,y)\ge \frac{1-\theta}{4}(a+b),
\]
so $1-Q_\theta(x,y)>0$ unless $(a,b)=(0,0)$, i.e.\ unless $(x,y)=(2,2)$.

Moreover,
\begin{align*}
  1-Q_\theta(2\cos a,2\cos b)
  &=
  1-\frac{1-\theta}{2}(\cos a+\cos b)-\theta\cos a\cos b \\
  &=
  \frac{1+\theta}{4}(a^2+b^2)+O(a^4+b^4),
\end{align*}
so \Cref{thm:poly_square} applies with $\sigma_1=\sigma_2=(1+\theta)/2$. Therefore
\[
  \tr\!\bigl((\mathcal L_R^{(\theta),0})^{-1}\bigr)
  =
  \frac{1}{\pi(1+\theta)}R^2\log R^2+O(R^2).
\]
The lazy statement follows from $\mathcal L_R^{(\theta),1/2}=\frac12\mathcal L_R^{(\theta),0}$.
\end{proof}

\begin{theorem}[Verified square-domain criterion for the lazy axial-diagonal family]\label{thm:axial_diagonal_verified}
Fix $0\le \theta<1$, and let $p_t^{(\theta),1/2,V_R}(u,v)$ denote the Dirichlet heat kernel of the lazy axial-diagonal walk on the square $V_R=\{1,\dots,R\}^2$. Then:
\begin{enumerate}
  \item There exists $C_\theta<\infty$ such that for every $R\ge1$, every $u\in V_R$, and every integer $t\ge R^2$,
  \[
    p_t^{(\theta),1/2,V_R}(u,u)
    \le
    \frac{C_\theta}{R^2}e^{-\lambda_1(\mathcal L_R^{(\theta),1/2})t}.
  \]
  \item If $(a_R)$ and $(b_R)$ are integer sequences with
  \[
    a_R\to\infty,
    \qquad
    a_R\le b_R,
    \qquad
    b_R=o(R^2),
  \]
  then
  \[
    \sup_{a_R\le t\le b_R}
    \frac{t}{R^2}
    \left|
      \sum_{u\in V_R} p_t^{(\theta),1/2,V_R}(u,u)
      -
      \frac{2}{\pi(1+\theta)}\frac{R^2}{t}
    \right|
    \longrightarrow 0.
  \]
  \item The Dirichlet spectral zeta value satisfies
  \[
    \tr\!\bigl((\mathcal L_R^{(\theta),1/2})^{-1}\bigr)
    =
    \frac{2}{\pi(1+\theta)}R^2\log R^2+O(R^2).
  \]
\end{enumerate}
In particular, on square domains the lazy axial-diagonal family satisfies the long-time Dirichlet input and the mesoscopic heat-trace law used in the abstract results, with explicit leading constant
\[
  \cG_\theta=\frac{2}{\pi(1+\theta)}.
\]
\end{theorem}

\begin{proof}
The lazy transition operator is
\[
  P_R^{(\theta),1/2}
  =
  \frac12 I+\frac12 P_R^{(\theta),0},
\]
so its symbol is
\[
  F_\theta^{\mathrm{lazy}}(x,y)
  :=
  \frac12+\frac12 Q_\theta(x,y).
\]
Since $Q_\theta(x,y)\in[-1,1]$ on $[-2,2]^2$, one has
\[
  0\le F_\theta^{\mathrm{lazy}}(x,y)\le 1.
\]
Moreover,
\[
  1-F_\theta^{\mathrm{lazy}}(2\cos a,2\cos b)
  =
  \frac12\bigl(1-Q_\theta(2\cos a,2\cos b)\bigr)
  =
  \frac{1+\theta}{8}(a^2+b^2)+O(a^4+b^4).
\]
Thus \Cref{thm:poly_square} and \Cref{prop:poly_square_LT} apply with
\[
  \sigma_1=\sigma_2=\frac{1+\theta}{4},
\]
which gives item~(1). The same choice of $(\sigma_1,\sigma_2)$ in \Cref{thm:poly_square_heat_trace} gives item~(2), since
\[
  \frac{1}{2\pi\sqrt{\sigma_1\sigma_2}}
  =
  \frac{2}{\pi(1+\theta)}.
\]
Finally, item~(3) is exactly the lazy part of \Cref{cor:axial_diagonal_square}.
\end{proof}

\begin{remark}
\Cref{thm:Z2_square} is the special case $\theta=0$, while the king walk corresponds to $\theta=\frac12$ and yields
\[
  \tr\!\bigl((\mathcal L_R^{(1/2),0})^{-1}\bigr)
  =
  \frac{2}{3\pi}R^2\log R^2+O(R^2).
\]
Equivalently,
\begin{equation}\label{eq:King_pi}
\boxed{\;\displaystyle \pi=\frac{2}{3}\;\lim_{R\to\infty}\frac{R^{2}\log R^{2}}{\tr\!\bigl((\mathcal{L}_R^{(1/2),0})^{-1}\bigr)}\;}
\end{equation}
\end{remark}

\begin{remark}
\Cref{thm:poly_square} isolates the actual mechanism behind the square computations: once the Dirichlet transition operator is a fixed polynomial in the horizontal and vertical path adjacencies, the square-domain spectral zeta asymptotic is governed solely by the quadratic part of its symbol at $(2,2)$. \Cref{cor:axial_diagonal_square} turns this into a concrete verified family interpolating between the square-lattice and king-walk cases.
\end{remark}
